     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react';
     4→import { motion } from 'framer-motion';
     5→import { Film, ArrowLeft, Sparkles, Check, AlertCircle, RefreshCw } from 'lucide-react';
     6→import { toast } from 'sonner';
     7→import { z } from 'zod';
     8→import { useForm } from 'react-hook-form';
     9→import { zodResolver } from '@hookform/resolvers/zod';
    10→
    11→import { useAppStore } from '@/lib/store';
    12→import { generateScript } from '@/lib/gemini';
    13→import type { Project, VideoTheme, WritingStyle, TargetAudience, VideoLanguage, VideoDuration } from '@/lib/types';
    14→import {
    15→  DURATION_LABELS,
    16→  THEME_LABELS,
    17→  STYLE_LABELS,
    18→  AUDIENCE_LABELS,
    19→  LANGUAGE_LABELS,
    20→} from '@/lib/types';
    21→import { useRouter } from 'next/navigation';
    22→
    23→import { Button } from '@/components/ui/button';
    24→import { Input } from '@/components/ui/input';
    25→import { Textarea } from '@/components/ui/textarea';
    26→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    27→import { Separator } from '@/components/ui/separator';
    28→import {
    29→  Form,
    30→  FormControl,
    31→  FormField,
    32→  FormItem,
    33→  FormLabel,
    34→  FormMessage,
    35→} from '@/components/ui/form';
    36→import {
    37→  Select,
    38→  SelectContent,
    39→  SelectItem,
    40→  SelectTrigger,
    41→  SelectValue,
    42→} from '@/components/ui/select';
    43→
    44→// ─── Schema ───────────────────────────────────────────────────────────────────
    45→
    46→const formSchema = z.object({
    47→  title: z.string().min(1, 'Title is required').max(100),
    48→  topic: z.string().min(1, 'Topic is required').max(200),
    49→  description: z.string().max(1000).optional().default(''),
    50→  duration: z.enum(['short', 'medium', 'long']),
    51→  theme: z.enum(['realistic', 'anime', 'cinematic', 'cartoon', '3d-render', 'watercolor', 'pixel-art']),
    52→  language: z.enum(['english', 'spanish', 'french', 'german', 'portuguese', 'japanese', 'korean', 'chinese', 'hindi', 'arabic']),
    53→  writingStyle: z.enum(['conversational', 'professional', 'dramatic', 'educational', 'storytelling', 'humorous']),
    54→  targetAudience: z.enum(['beginners', 'general', 'intermediate', 'experts', 'kids', 'teens']),
    55→});
    56→
    57→type FormValues = z.infer<typeof formSchema>;
    58→
    59→// ─── Generation Stages ────────────────────────────────────────────────────────
    60→
    61→const GENERATION_STAGES = [
    62→  { active: 'Researching topic...', completed: 'Research complete' },
    63→  { active: 'Building story structure...', completed: 'Story structured' },
    64→  { active: 'Generating scenes with AI...', completed: 'Scenes generated' },
    65→  { active: 'Finalizing...', completed: 'Finalized' },
    66→] as const;
    67→
    68→const STAGE_INTERVAL_MS = 5000;
    69→
    70→// ─── Component ────────────────────────────────────────────────────────────────
    71→
    72→export default function CreateProject() {
    73→  const router = useRouter();
    74→  const {
    75→    addProject,
    76→    setActiveProjectId,
    77→    addScenes,
    78→    updateProject,
    79→    loadScenes,
    80→  } = useAppStore();
    81→
    82→  // Phase: 'form' | 'generating' | 'error'
    83→  const [phase, setPhase] = useState<'form' | 'generating' | 'error'>('form');
    84→  const [currentStage, setCurrentStage] = useState(-1);
    85→  const [errorMessage, setErrorMessage] = useState('');
    86→  const [elapsed, setElapsed] = useState(0);
    87→  const stageTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    88→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    89→  const projectIdRef = useRef<string | null>(null);
    90→  const formValuesRef = useRef<FormValues | null>(null);
    91→
    92→  const form = useForm<FormValues>({
    93→    resolver: zodResolver(formSchema),
    94→    defaultValues: {
    95→      title: '',
    96→      topic: '',
    97→      description: '',
    98→      duration: 'medium',
    99→      theme: 'cinematic',
   100→      language: 'english',
   101→      writingStyle: 'conversational',
   102→      targetAudience: 'general',
   103→    },
   104→  });
   105→
   106→  // Cleanup timers on unmount
   107→  useEffect(() => {
   108→    return () => {
   109→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   110→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   111→    };
   112→  }, []);
   113→
   114→  const advanceStage = useCallback(() => {
   115→    setCurrentStage((prev) => {
   116→      if (prev >= GENERATION_STAGES.length - 1) return prev;
   117→      return prev + 1;
   118→    });
   119→  }, []);
   120→
   121→  const startGeneration = useCallback(async (values: FormValues) => {
   122→    const now = Date.now();
   123→
   124→    const project: Project = {
   125→      id: '',
   126→      title: values.title,
   127→      topic: values.topic,
   128→      description: values.description ?? '',
   129→      settings: {
   130→        duration: values.duration as VideoDuration,
   131→        theme: values.theme as VideoTheme,
   132→        language: values.language as VideoLanguage,
   133→        writingStyle: values.writingStyle as WritingStyle,
   134→        targetAudience: values.targetAudience as TargetAudience,
   135→      },
   136→      status: 'generating',
   137→      createdAt: now,
   138→      updatedAt: now,
   139→    };
   140→
   141→    // Save project to MongoDB and get the real ID back
   142→    const mongoId = await addProject(project);
   143→    setActiveProjectId(mongoId);
   144→    projectIdRef.current = mongoId;
   145→
   146→    // Switch to generating phase (stays on this page)
   147→    setPhase('generating');
   148→    setCurrentStage(0);
   149→    setElapsed(0);
   150→    stageTimerRef.current = setInterval(advanceStage, STAGE_INTERVAL_MS);
   151→    elapsedTimerRef.current = setInterval(() => setElapsed((t) => t + 1), 1000);
   152→
   153→    try {
   154→      // Use the real MongoDB project from store for generation
   155→      const realProject = useAppStore.getState().projects.find((p) => p.id === mongoId);
   156→      const projectToGenerate = realProject || { ...project, id: mongoId };
   157→      const scenes = await generateScript(projectToGenerate);
   158→      const scenesWithCorrectId = scenes.map((s) => ({ ...s, projectId: mongoId }));
   159→
   160→      // Advance to "saving" stage
   161→      setCurrentStage(GENERATION_STAGES.length - 1);
   162→
   163→      // Save scenes to DB
   164→      await addScenes(scenesWithCorrectId);
   165→      await updateProject(mongoId, { status: 'completed' });
   166→
   167→      // Brief pause to show "Finalized" stage
   168→      await new Promise((r) => setTimeout(r, 800));
   169→
   170→      // Clean up timers
   171→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   172→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   173→
   174→      // Pre-load scenes into store, then auto-redirect to editor
   175→      await loadScenes(mongoId);
   176→      router.push(`/project/${mongoId}`);
   177→    } catch (error) {
   178→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   179→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   180→
   181→      const message = error instanceof Error ? error.message : 'An unexpected error occurred';
   182→      setErrorMessage(message);
   183→      await updateProject(mongoId, { status: 'error' });
   184→      setPhase('error');
   185→    }
   186→  }, [addProject, setActiveProjectId, advanceStage, addScenes, updateProject, loadScenes, router]);
   187→
   188→  const onRetry = useCallback(() => {
   189→    if (formValuesRef.current) {
   190→      setPhase('form');
   191→      setCurrentStage(-1);
   192→      setElapsed(0);
   193→      setErrorMessage('');
   194→    }
   195→  }, []);
   196→
   197→  const onSubmit = useCallback((values: FormValues) => {
   198→    formValuesRef.current = values;
   199→    startGeneration(values);
   200→  }, [startGeneration]);
   201→
   202→  const onBackToDashboard = useCallback(() => {
   203→    router.push('/dashboard');
   204→  }, [router]);
   205→
   206→  const descriptionValue = form.watch('description') ?? '';
   207→
   208→  const formatTime = (s: number) => {
   209→    const m = Math.floor(s / 60);
   210→    const sec = s % 60;
   211→    return `${m}:${sec.toString().padStart(2, '0')}`;
   212→  };
   213→
   214→  // ─── ERROR VIEW ──────────────────────────────────────────────────────────
   215→  if (phase === 'error') {
   216→    return (
   217→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   218→        <motion.div
   219→          initial={{ opacity: 0, scale: 0.95 }}
   220→          animate={{ opacity: 1, scale: 1 }}
   221→          transition={{ duration: 0.4 }}
   222→          className="w-full max-w-md text-center space-y-6"
   223→        >
   224→          <motion.div
   225→            initial={{ scale: 0 }}
   226→            animate={{ scale: 1 }}
   227→            transition={{ type: 'spring', stiffness: 300, damping: 20, delay: 0.1 }}
   228→            className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-destructive/10"
   229→          >
   230→            <AlertCircle className="h-8 w-8 text-destructive" />
   231→          </motion.div>
   232→
   233→          <div className="space-y-2">
   234→            <h2 className="text-2xl font-bold tracking-tight">Generation Failed</h2>
   235→            <p className="text-sm text-muted-foreground">
   236→              {errorMessage || 'Something went wrong. Please try again.'}
   237→            </p>
   238→          </div>
   239→
   240→          <div className="flex flex-col gap-3">
   241→            <Button
   242→              size="lg"
   243→              onClick={onRetry}
   244→              className="w-full text-base font-semibold gap-2"
   245→            >
   246→              <RefreshCw className="h-4 w-4" />
   247→              Try Again
   248→            </Button>
   249→            <Button
   250→              variant="outline"
   251→              onClick={onBackToDashboard}
   252→              className="w-full"
   253→            >
   254→              Back to Dashboard
   255→            </Button>
   256→          </div>
   257→        </motion.div>
   258→      </div>
   259→    );
   260→  }
   261→
   262→  // ─── GENERATING VIEW ─────────────────────────────────────────────────────
   263→  if (phase === 'generating') {
   264→    return (
   265→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   266→        <motion.div
   267→          initial={{ opacity: 0, y: 20 }}
   268→          animate={{ opacity: 1, y: 0 }}
   269→          transition={{ duration: 0.4, ease: 'easeOut' }}
   270→          className="w-full max-w-lg space-y-8"
   271→        >
   272→          {/* Header */}
   273→          <div className="text-center space-y-2">
   274→            <motion.div
   275→              animate={{ scale: [1, 1.08, 1] }}
   276→              transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   277→              className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-2xl bg-primary/10"
   278→            >
   279→              <Sparkles className="h-7 w-7 text-primary" />
   280→            </motion.div>
   281→            <h2 className="text-2xl font-bold tracking-tight">Generating Your Script</h2>
   282→            <p className="text-sm text-muted-foreground">
   283→              {formValuesRef.current?.title && `Crafting scenes for "${formValuesRef.current.title}"`}
   284→            </p>
   285→          </div>
   286→
   287→          {/* Elapsed time */}
   288→          <div className="text-center">
   289→            <span className="text-xs text-muted-foreground font-mono">
   290→              Elapsed {formatTime(elapsed)}
   291→            </span>
   292→          </div>
   293→
   294→          {/* Progress bar */}
   295→          <div className="relative h-2 w-full overflow-hidden rounded-full bg-primary/20">
   296→            <div className="absolute inset-0 animate-indeterminate-progress rounded-full bg-primary" />
   297→          </div>
   298→
   299→          {/* Stages */}
   300→          <div className="space-y-3">
   301→            {GENERATION_STAGES.map((stage, index) => {
   302→              const isCompleted = index < currentStage;
   303→              const isActive = index === currentStage;
   304→
   305→              return (
   306→                <motion.div
   307→                  key={index}
   308→                  initial={{ opacity: 0, x: -10 }}
   309→                  animate={{ opacity: 1, x: 0 }}
   310→                  transition={{ delay: index * 0.05 }}
   311→                  className="flex items-center gap-3 text-sm rounded-lg px-3 py-2 transition-colors"
   312→                  style={{
   313→                    backgroundColor: isActive ? 'hsl(var(--primary) / 0.06)' : 'transparent',
   314→                  }}
   315→                >
   316→                  <div className="flex h-6 w-6 shrink-0 items-center justify-center">
   317→                    {isCompleted ? (
   318→                      <motion.div
   319→                        initial={{ scale: 0 }}
   320→                        animate={{ scale: 1 }}
   321→                        transition={{ type: 'spring', stiffness: 500, damping: 30 }}
   322→                      >
   323→                        <Check className="h-4 w-4 text-emerald-500" />
   324→                      </motion.div>
   325→                    ) : isActive ? (
   326→                      <motion.div
   327→                        className="h-2.5 w-2.5 rounded-full bg-primary"
   328→                        animate={{ scale: [1, 1.5, 1], opacity: [0.5, 1, 0.5] }}
   329→                        transition={{ repeat: Infinity, duration: 1.2 }}
   330→                      />
   331→                    ) : (
   332→                      <div className="h-2.5 w-2.5 rounded-full bg-muted-foreground/30" />
   333→                    )}
   334→                  </div>
   335→                  <span
   336→                    className={
   337→                      isCompleted
   338→                        ? 'text-emerald-600 dark:text-emerald-400'
   339→                        : isActive
   340→                          ? 'font-medium text-foreground'
   341→                          : 'text-muted-foreground'
   342→                    }
   343→                  >
   344→                    {isCompleted ? stage.completed : stage.active}
   345→                  </span>
   346→                </motion.div>
   347→              );
   348→            })}
   349→          </div>
   350→
   351→          {/* Tip */}
   352→          <p className="text-center text-xs text-muted-foreground/60">
   353→            AI is generating narration, image prompts, and animation prompts for each scene. This may take a moment for complex topics.
   354→          </p>
   355→        </motion.div>
   356→
   357→        {/* Indeterminate progress bar animation */}
   358→        <style jsx global>{`
   359→          @keyframes indeterminate-progress {
   360→            0% { transform: translateX(-100%); width: 40%; }
   361→            50% { transform: translateX(100%); width: 60%; }
   362→            100% { transform: translateX(200%); width: 40%; }
   363→          }
   364→          .animate-indeterminate-progress {
   365→            animation: indeterminate-progress 1.8s ease-in-out infinite;
   366→          }
   367→        `}</style>
   368→      </div>
   369→    );
   370→  }
   371→
   372→  // ─── FORM VIEW (default) ─────────────────────────────────────────────────
   373→  return (
   374→    <motion.div
   375→      initial={{ opacity: 0, y: 20 }}
   376→      animate={{ opacity: 1, y: 0 }}
   377→      transition={{ duration: 0.4, ease: 'easeOut' }}
   378→      className="flex min-h-full items-start justify-center px-4 py-8 md:items-center md:py-12"
   379→    >
   380→      <Card className="w-full max-w-2xl shadow-lg">
   381→        <CardHeader className="pb-6">
   382→          <div className="flex items-center gap-3">
   383→            <Button
   384→              variant="ghost"
   385→              size="icon"
   386→              onClick={() => router.push('/dashboard')}
   387→              className="shrink-0"
   388→              aria-label="Back to dashboard"
   389→            >
   390→              <ArrowLeft className="h-5 w-5" />
   391→            </Button>
   392→            <div className="flex items-center gap-3">
   393→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10">
   394→                <Film className="h-5 w-5 text-primary" />
   395→              </div>
   396→              <div>
   397→                <CardTitle className="text-xl font-semibold tracking-tight">
   398→                  Create New Project
   399→                </CardTitle>
   400→                <p className="mt-0.5 text-sm text-muted-foreground">
   401→                  Transform your video idea into a production script
   402→                </p>
   403→              </div>
   404→            </div>
   405→          </div>
   406→        </CardHeader>
   407→
   408→        <CardContent className="px-6 pb-8">
   409→          <Form {...form}>
   410→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-8">
   411→              {/* ── Basic Information ────────────────────────────────── */}
   412→              <section>
   413→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   414→                  Basic Information
   415→                </h2>
   416→
   417→                <div className="rounded-lg border bg-muted/30 p-4 space-y-4">
   418→                  <FormField
   419→                    control={form.control}
   420→                    name="title"
   421→                    render={({ field }) => (
   422→                      <FormItem>
   423→                        <FormLabel>Project Title</FormLabel>
   424→                        <FormControl>
   425→                          <Input
   426→                            placeholder="e.g. The Future of Renewable Energy"
   427→                            {...field}
   428→                          />
   429→                        </FormControl>
   430→                        <FormMessage />
   431→                      </FormItem>
   432→                    )}
   433→                  />
   434→
   435→                  <FormField
   436→                    control={form.control}
   437→                    name="topic"
   438→                    render={({ field }) => (
   439→                      <FormItem>
   440→                        <FormLabel>Video Topic</FormLabel>
   441→                        <FormControl>
   442→                          <Input
   443→                            placeholder="e.g. How solar panels are changing the world"
   444→                            {...field}
   445→                          />
   446→                        </FormControl>
   447→                        <FormMessage />
   448→                      </FormItem>
   449→                    )}
   450→                  />
   451→
   452→                  <FormField
   453→                    control={form.control}
   454→                    name="description"
   455→                    render={({ field }) => (
   456→                      <FormItem>
   457→                        <div className="flex items-center justify-between">
   458→                          <FormLabel>Description</FormLabel>
   459→                          <span className="text-xs text-muted-foreground">
   460→                            {descriptionValue.length}/1000
   461→                          </span>
   462→                        </div>
   463→                        <FormControl>
   464→                          <Textarea
   465→                            placeholder="Add any extra details, context, or specific requirements for the video..."
   466→                            rows={3}
   467→                            maxLength={1000}
   468→                            {...field}
   469→                            value={field.value ?? ''}
   470→                          />
   471→                        </FormControl>
   472→                        <FormMessage />
   473→                      </FormItem>
   474→                    )}
   475→                  />
   476→                </div>
   477→              </section>
   478→
   479→              <Separator />
   480→
   481→              {/* ── Production Settings ──────────────────────────────── */}
   482→              <section>
   483→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   484→                  Production Settings
   485→                </h2>
   486→
   487→                <div className="rounded-lg border bg-muted/30 p-4">
   488→                  <div className="grid gap-4 sm:grid-cols-2">
   489→                    <FormField
   490→                      control={form.control}
   491→                      name="duration"
   492→                      render={({ field }) => (
   493→                        <FormItem>
   494→                          <FormLabel>Video Duration</FormLabel>
   495→                          <Select
   496→                            onValueChange={field.onChange}
   497→                            defaultValue={field.value}
   498→                          >
   499→                            <FormControl>
   500→                              <SelectTrigger>
   501→                                <SelectValue placeholder="Select duration" />
   502→                              </SelectTrigger>
   503→                            </FormControl>
   504→                            <SelectContent>
   505→                              {(Object.entries(DURATION_LABELS) as [VideoDuration, string][]).map(
   506→                                ([value, label]) => (
   507→                                  <SelectItem key={value} value={value}>
   508→                                    {label}
   509→                                  </SelectItem>
   510→                                )
   511→                              )}
   512→                            </SelectContent>
   513→                          </Select>
   514→                          <FormMessage />
   515→                        </FormItem>
   516→                      )}
   517→                    />
   518→
   519→                    <FormField
   520→                      control={form.control}
   521→                      name="theme"
   522→                      render={({ field }) => (
   523→                        <FormItem>
   524→                          <FormLabel>Visual Theme</FormLabel>
   525→                          <Select
   526→                            onValueChange={field.onChange}
   527→                            defaultValue={field.value}
   528→                          >
   529→                            <FormControl>
   530→                              <SelectTrigger>
   531→                                <SelectValue placeholder="Select theme" />
   532→                              </SelectTrigger>
   533→                            </FormControl>
   534→                            <SelectContent>
   535→                              {(Object.entries(THEME_LABELS) as [VideoTheme, string][]).map(
   536→                                ([value, label]) => (
   537→                                  <SelectItem key={value} value={value}>
   538→                                    {label}
   539→                                  </SelectItem>
   540→                                )
   541→                              )}
   542→                            </SelectContent>
   543→                          </Select>
   544→                          <FormMessage />
   545→                        </FormItem>
   546→                      )}
   547→                    />
   548→
   549→                    <FormField
   550→                      control={form.control}
   551→                      name="language"
   552→                      render={({ field }) => (
   553→                        <FormItem>
   554→                          <FormLabel>Language</FormLabel>
   555→                          <Select
   556→                            onValueChange={field.onChange}
   557→                            defaultValue={field.value}
   558→                          >
   559→                            <FormControl>
   560→                              <SelectTrigger>
   561→                                <SelectValue placeholder="Select language" />
   562→                              </SelectTrigger>
   563→                            </FormControl>
   564→                            <SelectContent>
   565→                              {(Object.entries(LANGUAGE_LABELS) as [VideoLanguage, string][]).map(
   566→                                ([value, label]) => (
   567→                                  <SelectItem key={value} value={value}>
   568→                                    {label}
   569→                                  </SelectItem>
   570→                                )
   571→                              )}
   572→                            </SelectContent>
   573→                          </Select>
   574→                          <FormMessage />
   575→                        </FormItem>
   576→                      )}
   577→                    />
   578→
   579→                    <FormField
   580→                      control={form.control}
   581→                      name="writingStyle"
   582→                      render={({ field }) => (
   583→                        <FormItem>
   584→                          <FormLabel>Writing Style</FormLabel>
   585→                          <Select
   586→                            onValueChange={field.onChange}
   587→                            defaultValue={field.value}
   588→                          >
   589→                            <FormControl>
   590→                              <SelectTrigger>
   591→                                <SelectValue placeholder="Select style" />
   592→                              </SelectTrigger>
   593→                            </FormControl>
