    52→function formatDate(timestamp: string | number): string {
    53→  if (!timestamp) return '—';
    54→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    55→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    56→}
    57→
    58→function formatExpiryDate(timestamp: number): string {
    59→  if (!timestamp || timestamp <= 0) return '—';
    60→  const date = new Date(timestamp);
    61→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    62→}
