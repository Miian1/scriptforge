     5→import { motion, AnimatePresence } from 'framer-motion';

    10→} from 'lucide-react';

    24→} from '@/components/ui/dialog';

    30→interface AdminUser {
    31→  id: string;
    32→  name: string;
    33→  email: string;
    34→  provider: string;
    35→  role: string;
    36→  plan: string;
    37→  isVerified: boolean;
    38→  isCustomPlan: boolean;
    39→  customPlan: { isCustom: boolean; customLabel: string; customDays: number };
    40→  planExpiresAt: number;
    41→  planDaysLeft: number;

    43→  dailyUsage: { date: string; projectsCreated: number; aiGenerations: number };
    44→  createdAt: string;
    45→  updatedAt: string;

    47→
    48→function formatDate(timestamp: string | number): string {
    49→  if (!timestamp) return '—';
    50→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    51→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    52→}
    53→
    54→function formatExpiryDate(timestamp: number): string {
    55→  if (!timestamp || timestamp <= 0) return '—';
    56→  const date = new Date(timestamp);
    57→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    58→}

    60→// ── Days editing mode ──
    61→type DaysEditMode = 'add' | 'reduce' | 'set';
    62→
    63→export default function AdminPage() {
    64→  const router = useRouter();
    65→  const user = useAuthStore((s) => s.user);
    66→  const [users, setUsers] = useState<AdminUser[]>([]);
    67→  const [loading, setLoading] = useState(true);
    68→  const [search, setSearch] = useState('');
    69→  const [editUser, setEditUser] = useState<AdminUser | null>(null);
    70→  const [saving, setSaving] = useState(false);
    71→  const [deleteUser, setDeleteUser] = useState<AdminUser | null>(null);
    72→  const [deleting, setDeleting] = useState(false);
    73→  const [notifyUser, setNotifyUser] = useState<AdminUser | null>(null);
    74→  const [sendingNotif, setSendingNotif] = useState(false);
    75→  const [notifTitle, setNotifTitle] = useState('');
    76→  const [notifMessage, setNotifMessage] = useState('');
    77→  const [notifType, setNotifType] = useState<'info' | 'warning' | 'urgent'>('info');
    78→
    79→  // Edit form state
    80→  const [editPlan, setEditPlan] = useState('free');
    81→  const [editIsCustom, setEditIsCustom] = useState(false);
    82→  const [editCustomLabel, setEditCustomLabel] = useState('');
    83→  const [editCustomDays, setEditCustomDays] = useState(0);
    84→  const [editRole, setEditRole] = useState('user');
    85→  const [daysEditMode, setDaysEditMode] = useState<DaysEditMode>('add');
    86→  const [daysValue, setDaysValue] = useState(0);
    87→
    88→  // Redirect non-admin
    89→  useEffect(() => {
    90→    if (user && user.role !== 'admin') {
    91→      router.push('/dashboard');
    92→    }
    93→  }, [user, router]);
    94→
    95→  const fetchUsers = useCallback(async () => {
    96→    setLoading(true);
    97→    try {
    98→      const res = await fetch('/api/admin/users');
    99→      if (res.ok) {
   100→        const data = await res.json();
   101→        setUsers(data.users || []);
   102→      } else {
   103→        const data = await res.json();
   104→        toast.error(data.error || 'Failed to load users');
   105→      }
   106→    } catch {
   107→      toast.error('Network error. Please try again.');
   108→    } finally {
   109→      setLoading(false);
   110→    }
   111→  }, []);
   112→
   113→  useEffect(() => {
   114→    if (user?.role === 'admin') {
   115→      fetchUsers();
   116→    }
   117→  }, [user, fetchUsers]);
   118→
   119→  // Filter users by search
   120→  const filteredUsers = users.filter((u) => {
   121→    if (!search.trim()) return true;
   122→    const q = search.toLowerCase();
   123→    return u.name.toLowerCase().includes(q) || u.email.toLowerCase().includes(q);
   124→  });
   125→
   126→  // Stats
   127→  const totalUsers = users.length;
   128→  const proUsers = users.filter((u) => u.plan === 'pro').length;
   129→  const customUsers = users.filter((u) => u.isCustomPlan).length;
   130→  const expiredUsers = users.filter((u) => u.plan === 'pro' && u.planDaysLeft <= 0).length;
   131→
   132→  // Open edit dialog
   133→  const openEdit = (u: AdminUser) => {
   134→    setEditUser(u);
   135→    setEditPlan(u.plan);
   136→    setEditIsCustom(u.isCustomPlan);
   137→    setEditCustomLabel(u.customPlan?.customLabel || '');
   138→    setEditCustomDays(u.customPlan?.customDays || 0);
   139→    setEditRole(u.role);
   140→    setDaysEditMode('add');
   141→    setDaysValue(0);
   142→  };
   143→
   144→  // ── Compute preview of what days will be after save ──
   145→  const previewDaysLeft = (): number => {
   146→    if (!editUser) return 0;
   147→    if (editPlan === 'free') return 0;
   148→
   149→    // If custom is ON and has custom days, those take priority
   150→    if (editIsCustom && editCustomDays > 0) {
   151→      return editCustomDays;
   152→    }
   153→
   154→    // Base expiry
   155→    let baseExpiry = editUser.planExpiresAt && editUser.planExpiresAt > Date.now()
   156→      ? editUser.planExpiresAt
   157→      : Date.now();
   158→
   159→    if (daysEditMode === 'add' && daysValue > 0) {
   160→      baseExpiry = baseExpiry + daysValue * 24 * 60 * 60 * 1000;
   161→    } else if (daysEditMode === 'reduce' && daysValue > 0) {
   162→      baseExpiry = Math.max(Date.now(), baseExpiry - daysValue * 24 * 60 * 60 * 1000);
   163→    } else if (daysEditMode === 'set') {
   164→      if (daysValue <= 0) return 0;
   165→      baseExpiry = Date.now() + daysValue * 24 * 60 * 60 * 1000;
   166→    }
   167→
   168→    const diff = baseExpiry - Date.now();
   169→    return Math.max(0, Math.ceil(diff / (1000 * 60 * 60 * 24)));
   170→  };
   171→
   172→  const previewExpiryDate = (): string => {
   173→    if (!editUser) return '—';
   174→    if (editPlan === 'free') return 'N/A';
   175→
   176→    if (editIsCustom && editCustomDays > 0) {
   177→      const d = new Date(Date.now() + editCustomDays * 24 * 60 * 60 * 1000);
   178→      return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   179→    }
   180→
   181→    let baseExpiry = editUser.planExpiresAt && editUser.planExpiresAt > Date.now()
   182→      ? editUser.planExpiresAt
   183→      : Date.now();
   184→
   185→    if (daysEditMode === 'add' && daysValue > 0) {
   186→      baseExpiry = baseExpiry + daysValue * 24 * 60 * 60 * 1000;
   187→    } else if (daysEditMode === 'reduce' && daysValue > 0) {
   188→      baseExpiry = Math.max(Date.now(), baseExpiry - daysValue * 24 * 60 * 60 * 1000);
   189→    } else if (daysEditMode === 'set') {
   190→      if (daysValue <= 0) return 'N/A';
   191→      baseExpiry = Date.now() + daysValue * 24 * 60 * 60 * 1000;
   192→    }
   193→
   194→    return new Date(baseExpiry).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   195→  };
   196→
   197→  // Save user
   198→  const handleSave = async () => {
   199→    if (!editUser) return;
   200→    setSaving(true);
   201→    try {
   202→      const body: Record<string, unknown> = {
   203→        plan: editPlan,
   204→        role: editRole,
   205→        isCustomPlan: editIsCustom,
   206→        customPlan: {
   207→          isCustom: editIsCustom,
   208→          customLabel: editCustomLabel,
   209→          customDays: editCustomDays,
   210→        },
   211→      };
   212→
   213→      // Days operations (only when not custom, since custom takes priority)
   214→      if (!editIsCustom || editCustomDays <= 0) {
   215→        if (daysEditMode === 'add' && daysValue > 0) {
   216→          body.addDays = daysValue;
   217→        } else if (daysEditMode === 'reduce' && daysValue > 0) {
   218→          body.reduceDays = daysValue;
   219→        } else if (daysEditMode === 'set') {
   220→          body.setDays = daysValue;
   221→        }
   222→      }
   223→
   224→      const res = await fetch(`/api/admin/users/${editUser.id}`, {
   225→        method: 'PUT',
   226→        headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify(body),
   228→      });
   229→      const data = await res.json();
   230→      if (res.ok) {
   231→        toast.success(`Updated ${editUser.name} successfully`);
   232→        setEditUser(null);
   233→        fetchUsers();
   234→      } else {
   235→        toast.error(data.error || 'Failed to update user');
   236→      }
   237→    } catch {
   238→      toast.error('Network error. Please try again.');
   239→    } finally {
   240→      setSaving(false);
   241→    }
   242→  };
   243→
   244→  // Delete user
   245→  const handleDelete = async () => {
   246→    if (!deleteUser) return;
   247→    setDeleting(true);
   248→    try {
   249→      const res = await fetch(`/api/admin/users/${deleteUser.id}`, { method: 'DELETE' });
   250→      const data = await res.json();
   251→      if (res.ok) {
   252→        toast.success(`Deleted ${deleteUser.email}`);
   253→        setDeleteUser(null);
   254→        fetchUsers();
   255→      } else {
   256→        toast.error(data.error || 'Failed to delete user');
   257→      }
   258→    } catch {
   259→      toast.error('Network error. Please try again.');
   260→    } finally {
   261→      setDeleting(false);
   262→    }
   263→  };
   264→
   265→  // Quick add 30 days
   266→  const quickAddDays = async (u: AdminUser, days: number) => {
   267→    try {
   268→      const res = await fetch(`/api/admin/users/${u.id}`, {
   269→        method: 'PUT',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({ addDays: days }),
   272→      });
   273→      const data = await res.json();
   274→      if (res.ok) {
   275→        toast.success(`Added ${days} days to ${u.name}`);
   276→        fetchUsers();
   277→      } else {
   278→        toast.error(data.error || 'Failed to add days');
   279→      }
   280→    } catch {
   281→      toast.error('Network error');
   282→    }
   283→  };
   284→
   285→  // Send notification to user
   286→  const handleSendNotification = async () => {
   287→    if (!notifyUser || !notifTitle.trim() || !notifMessage.trim()) return;
   288→    setSendingNotif(true);
   289→    try {
   290→      const res = await fetch('/api/admin/notify', {
   291→        method: 'POST',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({
   294→          userId: notifyUser.id,
   295→          title: notifTitle.trim(),
   296→          message: notifMessage.trim(),
   297→          type: notifType,
   298→        }),
   299→      });
   300→      const data = await res.json();
   301→      if (res.ok) {
   302→        toast.success(`Notification sent to ${notifyUser.name}`);
   303→        setNotifyUser(null);
   304→        setNotifTitle('');
   305→        setNotifMessage('');
   306→        setNotifType('info');
   307→      } else {
   308→        toast.error(data.error || 'Failed to send notification');
   309→      }
   310→    } catch {
   311→      toast.error('Network error');
   312→    } finally {
   313→      setSendingNotif(false);
   314→    }
   315→  };
   316→
   317→  if (!user || user.role !== 'admin') return null;
   318→
   319→  return (
   320→    <div className="max-w-6xl mx-auto space-y-6">
   321→      {/* Header */}
   322→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   323→        <div className="flex items-center gap-3">
   324→          <Button variant="ghost" size="icon" onClick={() => router.push('/dashboard')} className="shrink-0">
   325→            <ChevronDown className="h-5 w-5 rotate-90" />
   326→          </Button>
   327→          <div>
   328→            <div className="flex items-center gap-2">
   329→              <ShieldCheck className="size-5 text-primary" />
   330→              <h1 className="text-2xl font-bold tracking-tight">Admin Panel</h1>
   331→            </div>
   332→            <p className="text-sm text-muted-foreground">Manage users, plans, and subscriptions</p>
   333→          </div>
   334→        </div>
   335→        <Button variant="outline" className="gap-2" onClick={fetchUsers} disabled={loading}>
   336→          <RefreshCw className={cn('size-4', loading && 'animate-spin')} />
   337→          Refresh
   338→        </Button>
   339→      </div>
   340→
   341→      {/* Stats Cards */}
   342→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   343→        {[
   344→          { label: 'Total Users', value: totalUsers, icon: Users, color: 'text-blue-500', bg: 'bg-blue-500/5' },
   345→          { label: 'Pro Users', value: proUsers, icon: Crown, color: 'text-primary', bg: 'bg-primary/5' },
   346→          { label: 'Custom Plans', value: customUsers, icon: Star, color: 'text-amber-500', bg: 'bg-amber-500/5' },
   347→          { label: 'Expired Plans', value: expiredUsers, icon: AlertTriangle, color: 'text-red-500', bg: 'bg-red-500/5' },
   348→        ].map((stat) => (
   349→          <Card key={stat.label}>
   350→            <CardContent className="pt-4 pb-3 px-4">
   351→              <div className="flex items-center justify-between">
   352→                <div>
   353→                  <p className="text-xs text-muted-foreground">{stat.label}</p>
   354→                  <p className="text-2xl font-bold mt-0.5">{stat.value}</p>
   355→                </div>
   356→                <div className={cn('size-9 rounded-lg flex items-center justify-center', stat.bg)}>
   357→                  <stat.icon className={cn('size-4', stat.color)} />
   358→                </div>
   359→              </div>
   360→            </CardContent>
   361→          </Card>
   362→        ))}
   363→      </div>
   364→
   365→      {/* Search */}
   366→      <Card>
   367→        <CardContent className="pt-4">
   368→          <div className="relative">
   369→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   370→            <Input
   371→              type="search"
   372→              placeholder="Search by name or email..."
   373→              className="pl-10 h-10"
   374→              value={search}
   375→              onChange={(e) => setSearch(e.target.value)}
   376→            />
   377→          </div>
   378→        </CardContent>
   379→      </Card>
   380→
   381→      {/* Users Table */}
   382→      <Card>
   383→        <CardHeader className="pb-3">
   384→          <CardTitle className="text-base flex items-center gap-2">
   385→            <Users className="size-4" />
   386→            All Users
   387→            <Badge variant="secondary" className="text-[10px]">{filteredUsers.length}</Badge>
   388→          </CardTitle>
   389→          <CardDescription>Click edit to change plans, add/reduce days, or set custom plans</CardDescription>
   390→        </CardHeader>
   391→        <CardContent className="p-0">
   392→          {loading ? (
   393→            <div className="flex items-center justify-center py-12">
   394→              <Loader2 className="size-6 animate-spin text-muted-foreground" />
   395→              <span className="ml-2 text-sm text-muted-foreground">Loading users...</span>
   396→            </div>
   397→          ) : filteredUsers.length === 0 ? (
   398→            <div className="text-center py-12 text-sm text-muted-foreground">
   399→              {search ? 'No users match your search.' : 'No users found.'}
   400→            </div>
   401→          ) : (
   402→            <div className="overflow-x-auto">
   403→              <table className="w-full text-sm">
   404→                <thead>
   405→                  <tr className="border-b bg-muted/30">
   406→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">User</th>
   407→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Plan</th>
   408→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Days Left</th>
   409→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Custom</th>
   410→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Usage</th>
   411→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Joined</th>
   412→                    <th className="text-right px-4 py-2.5 font-medium text-muted-foreground">Actions</th>
   413→                  </tr>
   414→                </thead>
   415→                <tbody>
   416→                  {filteredUsers.map((u) => (
   417→                    <motion.tr
   418→                      key={u.id}
   419→                      layout
   420→                      initial={{ opacity: 0 }}
   421→                      animate={{ opacity: 1 }}
   422→                      className="border-b hover:bg-muted/20 transition-colors"
   423→                    >
   424→                      <td className="px-4 py-3">
   425→                        <div className="flex flex-col gap-0.5">
   426→                          <div className="flex items-center gap-2">
   427→                            <span className="font-medium text-sm">{u.name}</span>
   428→                            {u.role === 'admin' && (
   429→                              <Badge className="text-[9px] px-1.5 py-0 h-4">Admin</Badge>
   430→                            )}
   431→                            {!u.isVerified && (
   432→                              <Badge variant="secondary" className="text-[9px] px-1.5 py-0 h-4">Unverified</Badge>
   433→                            )}
   434→                          </div>
   435→                          <span className="text-xs text-muted-foreground">{u.email}</span>
   436→                        </div>
   437→                      </td>
   438→                      <td className="px-4 py-3">
   439→                        <div className="flex items-center gap-1.5">
   440→                          {u.plan === 'pro' ? (
   441→                            <>
   442→                              <Crown className="size-3.5 text-primary" />
   443→                              <span className="font-medium text-primary">Pro</span>
   444→                            </>
   445→                          ) : (
   446→                            <>
   447→                              <Zap className="size-3.5 text-muted-foreground" />
   448→                              <span className="text-muted-foreground">Free</span>
   449→                            </>
   450→                          )}
   451→                        </div>
   452→                        {u.stripe?.subscriptionId && (
   453→                          <span className="text-[10px] text-muted-foreground">Stripe</span>
   454→                        )}
   455→                      </td>
   456→                      <td className="px-4 py-3">
   457→                        {u.plan === 'pro' ? (
   458→                          <div className="flex flex-col gap-0.5">
   459→                            <span className={cn(
   460→                              'text-sm font-semibold',
   461→                              u.planDaysLeft <= 3 ? 'text-red-500' : u.planDaysLeft <= 7 ? 'text-amber-500' : 'text-foreground'
   462→                            )}>
   463→                              {u.planDaysLeft}d
   464→                            </span>
   465→                            {u.planExpiresAt > 0 && (
   466→                              <span className="text-[10px] text-muted-foreground">
   467→                                {formatExpiryDate(u.planExpiresAt)}
   468→                              </span>
   469→                            )}
   470→                          </div>
   471→                        ) : (
   472→                          <span className="text-xs text-muted-foreground">—</span>
   473→                        )}
   474→                      </td>
   475→                      <td className="px-4 py-3">
   476→                        {u.isCustomPlan ? (
   477→                          <div className="flex flex-col gap-0.5">
   478→                            <Badge className="text-[10px] gap-1 w-fit" variant="outline">
   479→                              <Star className="size-2.5" />
   480→                              {u.customPlan?.customLabel || 'Custom'}
   481→                            </Badge>
   482→                            {u.customPlan?.customDays > 0 && (
   483→                              <span className="text-[10px] text-muted-foreground">
   484→                                {u.customPlan.customDays}d plan
   485→                              </span>
   486→                            )}
   487→                          </div>
   488→                        ) : (
   489→                          <span className="text-xs text-muted-foreground">—</span>
   490→                        )}
   491→                      </td>
   492→                      <td className="px-4 py-3">
   493→                        <div className="text-xs text-muted-foreground">
   494→                          <span>{u.dailyUsage?.projectsCreated ?? 0} proj</span>
   495→                          <span className="mx-1">·</span>
   496→                          <span>{u.dailyUsage?.aiGenerations ?? 0} AI</span>
   497→                        </div>
   498→                      </td>
   499→                      <td className="px-4 py-3 text-xs text-muted-foreground">
   500→                        {formatDate(u.createdAt)}
   501→                      </td>
   502→                      <td className="px-4 py-3">
   503→                        <div className="flex items-center justify-end gap-1">
   504→                          {/* Quick add 30 days */}
   505→                          <Button
   506→                            variant="ghost"
   507→                            size="icon"
   508→                            className="size-7"
   509→                            title="Add 30 days"
   510→                            onClick={() => quickAddDays(u, 30)}
   511→                          >
   512→                            <Plus className="size-3" />
   513→                          </Button>
   514→                          {/* Send notification */}
   515→                          <Button
   516→                            variant="ghost"
   517→                            size="icon"
   518→                            className="size-7 text-blue-500 hover:text-blue-500"
   519→                            title="Send notification"
   520→                            onClick={() => { setNotifyUser(u); setNotifTitle(''); setNotifMessage(''); setNotifType('info'); }}
   521→                          >
   522→                            <Bell className="size-3" />
   523→                          </Button>
   524→                          {/* Edit */}
   525→                          <Button
   526→                            variant="ghost"
   527→                            size="icon"
   528→                            className="size-7"
