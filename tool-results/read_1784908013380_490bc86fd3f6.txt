    58→    58→  writingStyle: z.enum(['conversational', 'professional', 'dramatic', 'educational', 'storytelling', 'humorous']),

    65→    65→
    66→    66→const GENERATION_STAGES = [
    67→    67→  { active: 'Researching topic...', completed: 'Research complete' },
    68→    68→  { active: 'Building story structure...', completed: 'Story structured' },
    69→    69→  { active: 'Generating Phase 1 scenes...', completed: 'Phase 1 generated' },
    70→    70→  { active: 'Finalizing...', completed: 'Finalized' },
    71→    71→] as const;
    72→    72→

    74→    74→
    75→    75→// ─── Component ────────────────────────────────────────────────────────────────
    76→    76→
    77→    77→export default function CreateProject() {
    78→    78→  const router = useRouter();
    79→    79→  const {
    80→    80→    addProject,
    81→    81→    setActiveProjectId,
    82→    82→    addScenes,
    83→    83→    updateProject,
    84→    84→    loadScenes,
    85→    85→  } = useAppStore();
    86→    86→
    87→    87→  // Phase: 'form' | 'generating' | 'error'
    88→    88→  const [phase, setPhase] = useState<'form' | 'generating' | 'error'>('form');
    89→    89→  const [currentStage, setCurrentStage] = useState(-1);
    90→    90→  const [errorMessage, setErrorMessage] = useState('');
    91→    91→  const [elapsed, setElapsed] = useState(0);
    92→    92→  const stageTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    93→    93→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    94→    94→  const projectIdRef = useRef<string | null>(null);
    95→    95→  const formValuesRef = useRef<FormValues | null>(null);
    96→    96→
    97→    97→  const form = useForm<FormValues>({
    98→    98→    resolver: zodResolver(formSchema),
    99→    99→    defaultValues: {
   100→   100→      title: '',
   101→   101→      topic: '',
   102→   102→      description: '',
   103→   103→      duration: 'medium',
   104→   104→      sceneLengthMode: 'default',
   105→   105→      customSceneLength: undefined,
   106→   106→      theme: 'cinematic',
   107→   107→      language: 'english',
   108→   108→      writingStyle: 'conversational',
   109→   109→      targetAudience: 'general',
   110→   110→    },
   111→   111→  });
   112→   112→
   113→   113→  // Watch duration and scene length for calculations
   114→   114→  const watchDuration = form.watch('duration');
   115→   115→  const watchSceneLengthMode = form.watch('sceneLengthMode');
   116→   116→  const watchCustomSceneLength = form.watch('customSceneLength');
   117→   117→
   118→   118→  // Computed scene length
   119→   119→  const sceneLength = watchSceneLengthMode === 'custom' && watchCustomSceneLength
   120→   120→    ? watchCustomSceneLength
   121→   121→    : 8;
   122→   122→
   123→   123→  // Computed totals
   124→   124→  const totalSeconds = DURATION_SECONDS[watchDuration as VideoDuration];
   125→   125→  const totalScenes = Math.ceil(totalSeconds / sceneLength);
   126→   126→  const totalPhases = Math.ceil(totalScenes / SCENES_PER_PHASE);
   127→   127→  const firstPhaseScenes = Math.min(SCENES_PER_PHASE, totalScenes);
   128→   128→
   129→   129→  // Cleanup timers on unmount
   130→   130→  useEffect(() => {
   131→   131→    return () => {
   132→   132→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   133→   133→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   134→   134→    };
   135→   135→  }, []);
   136→   136→
   137→   137→  const advanceStage = useCallback(() => {
   138→   138→    setCurrentStage((prev) => {
   139→   139→      if (prev >= GENERATION_STAGES.length - 1) return prev;
   140→   140→      return prev + 1;
   141→   141→    });
   142→   142→  }, []);
   143→   143→
   144→   144→  const startGeneration = useCallback(async (values: FormValues) => {
   145→   145→    const now = Date.now();
   146→   146→    const sLen = values.sceneLengthMode === 'custom' && values.customSceneLength
   147→   147→      ? values.customSceneLength
   148→   148→      : 8;
   149→   149→    const totalSec = DURATION_SECONDS[values.duration as VideoDuration];
   150→   150→    const tScenes = Math.ceil(totalSec / sLen);
   151→   151→    const tPhases = Math.ceil(tScenes / SCENES_PER_PHASE);
   152→   152→
   153→   153→    const project: Project = {
   154→   154→      id: '',
   155→   155→      title: values.title,
   156→   156→      topic: values.topic,
   157→   157→      description: values.description ?? '',
   158→   158→      thumbnailPrompt: '',
   159→   159→      tags: [],
   160→   160→      settings: {
   161→   161→        duration: values.duration as VideoDuration,
   162→   162→        theme: values.theme as VideoTheme,
   163→   163→        language: values.language as VideoLanguage,
   164→   164→        writingStyle: values.writingStyle as WritingStyle,
   165→   165→        targetAudience: values.targetAudience as TargetAudience,
   166→   166→        sceneLength: sLen,
   167→   167→        totalScenes: tScenes,
   168→   168→        scenesPerPhase: SCENES_PER_PHASE,
   169→   169→      },
   170→   170→      status: 'generating',
   171→   171→      scoreHistory: [],
   172→   172→      createdAt: now,
   173→   173→      updatedAt: now,
   174→   174→    };
   175→   175→
   176→   176→    // Save project to MongoDB and get the real ID back
   177→   177→    const mongoId = await addProject(project);
   178→   178→    setActiveProjectId(mongoId);
   179→   179→    projectIdRef.current = mongoId;
   180→   180→
   181→   181→    // Switch to generating phase
   182→   182→    setPhase('generating');
   183→   183→    setCurrentStage(0);
   184→   184→    setElapsed(0);
   185→   185→    stageTimerRef.current = setInterval(advanceStage, STAGE_INTERVAL_MS);
   186→   186→    elapsedTimerRef.current = setInterval(() => setElapsed((t) => t + 1), 1000);
   187→   187→
   188→   188→    try {
   189→   189→      const realProject = useAppStore.getState().projects.find((p) => p.id === mongoId);
   190→   190→      const projectToGenerate = realProject || { ...project, id: mongoId };
   191→   191→
   192→   192→      // Generate Phase 1 only
   193→   193→      const phase1ScenesCount = Math.min(SCENES_PER_PHASE, tScenes);
   194→   194→      const result = await generatePhase(projectToGenerate, {
   195→   195→        phaseNumber: 1,
   196→   196→        totalPhases: tPhases,
   197→   197→        sceneStart: 1,
   198→   198→        sceneEnd: phase1ScenesCount,
   199→   199→        previousPhaseTitles: [],
   200→   200→      });
   201→   201→
   202→   202→      const scenesWithCorrectId = result.scenes.map((s) => ({ ...s, projectId: mongoId }));
   203→   203→
   204→   204→      // Save AI-generated metadata
   205→   205→      if (result.metadata) {
   206→   206→        await updateProject(mongoId, {
   207→   207→          description: result.metadata.videoDescription || projectToGenerate.description || '',
   208→   208→          thumbnailPrompt: result.metadata.thumbnailPrompt || '',
   209→   209→          tags: result.metadata.tags || [],
   210→   210→        });
   211→   211→      }
   212→   212→
   213→   213→      // Advance to "saving" stage
   214→   214→      setCurrentStage(GENERATION_STAGES.length - 1);
   215→   215→
   216→   216→      // Save scenes to DB
   217→   217→      await addScenes(scenesWithCorrectId);
   218→   218→      await updateProject(mongoId, { status: 'completed' });
   219→   219→
   220→   220→      // Brief pause
   221→   221→      await new Promise((r) => setTimeout(r, 800));
   222→   222→
   223→   223→      // Clean up timers
   224→   224→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   225→   225→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   226→   226→
   227→   227→      // Pre-load scenes into store, then redirect to editor
   228→   228→      await loadScenes(mongoId);
   229→   229→      router.push(`/project/${mongoId}`);
   230→   230→    } catch (error) {
   231→   231→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   232→   232→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   233→   233→
   234→   234→      const message = error instanceof Error ? error.message : 'An unexpected error occurred';
   235→   235→      setErrorMessage(message);
   236→   236→      await updateProject(mongoId, { status: 'error' });
   237→   237→      setPhase('error');
   238→   238→    }
   239→   239→  }, [addProject, setActiveProjectId, advanceStage, addScenes, updateProject, loadScenes, router]);
   240→   240→
   241→   241→  const onRetry = useCallback(() => {
   242→   242→    if (formValuesRef.current) {
   243→   243→      setPhase('form');
   244→   244→      setCurrentStage(-1);
   245→   245→      setElapsed(0);
   246→   246→      setErrorMessage('');
   247→   247→    }
   248→   248→  }, []);
   249→   249→
   250→   250→  const onSubmit = useCallback((values: FormValues) => {
   251→   251→    formValuesRef.current = values;
   252→   252→    startGeneration(values);
   253→   253→  }, [startGeneration]);
   254→   254→
   255→   255→  const onBackToDashboard = useCallback(() => {
   256→   256→    router.push('/dashboard');
   257→   257→  }, [router]);
   258→   258→
   259→   259→  const descriptionValue = form.watch('description') ?? '';
   260→   260→
   261→   261→  const formatTime = (s: number) => {
   262→   262→    const m = Math.floor(s / 60);
   263→   263→    const sec = s % 60;
   264→   264→    return `${m}:${sec.toString().padStart(2, '0')}`;
   265→   265→  };
   266→   266→
   267→   267→  // ─── ERROR VIEW ──────────────────────────────────────────────────────────
   268→   268→  if (phase === 'error') {
   269→   269→    return (
   270→   270→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   271→   271→        <motion.div
   272→   272→          initial={{ opacity: 0, scale: 0.95 }}
   273→   273→          animate={{ opacity: 1, scale: 1 }}
   274→   274→          transition={{ duration: 0.4 }}
   275→   275→          className="w-full max-w-md text-center space-y-6"
   276→   276→        >
   277→   277→          <motion.div
   278→   278→            initial={{ scale: 0 }}
   279→   279→            animate={{ scale: 1 }}
   280→   280→            transition={{ type: 'spring', stiffness: 300, damping: 20, delay: 0.1 }}
   281→   281→            className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-destructive/10"
   282→   282→          >
   283→   283→            <AlertCircle className="h-8 w-8 text-destructive" />
   284→   284→          </motion.div>
   285→   285→
   286→   286→          <div className="space-y-2">
   287→   287→            <h2 className="text-2xl font-bold tracking-tight">Generation Failed</h2>
   288→   288→            <p className="text-sm text-muted-foreground">
   289→   289→              {errorMessage || 'Something went wrong. Please try again.'}
   290→   290→            </p>
   291→   291→          </div>
   292→   292→
   293→   293→          <div className="flex flex-col gap-3">
   294→   294→            <Button
   295→   295→              size="lg"
   296→   296→              onClick={onRetry}
   297→   297→              className="w-full text-base font-semibold gap-2"
   298→   298→            >
   299→   299→              <RefreshCw className="h-4 w-4" />
   300→   300→              Try Again
   301→   301→            </Button>
   302→   302→            <Button
   303→   303→              variant="outline"
   304→   304→              onClick={onBackToDashboard}
   305→   305→              className="w-full"
   306→   306→            >
   307→   307→              Back to Dashboard
   308→   308→            </Button>
   309→   309→          </div>
   310→   310→        </motion.div>
   311→   311→      </div>
   312→   312→    );
   313→   313→  }
   314→   314→
   315→   315→  // ─── GENERATING VIEW ─────────────────────────────────────────────────────
   316→   316→  if (phase === 'generating') {
   317→   317→    return (
   318→   318→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   319→   319→        <motion.div
   320→   320→          initial={{ opacity: 0, y: 20 }}
   321→   321→          animate={{ opacity: 1, y: 0 }}
   322→   322→          transition={{ duration: 0.4, ease: 'easeOut' }}
   323→   323→          className="w-full max-w-lg space-y-8"
   324→   324→        >
   325→   325→          {/* Header */}
   326→   326→          <div className="text-center space-y-2">
   327→   327→            <motion.div
   328→   328→              animate={{ scale: [1, 1.08, 1] }}
   329→   329→              transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   330→   330→              className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-2xl bg-primary/10"
   331→   331→            >
   332→   332→              <Sparkles className="h-7 w-7 text-primary" />
   333→   333→            </motion.div>
   334→   334→            <h2 className="text-2xl font-bold tracking-tight">Generating Phase 1</h2>
   335→   335→            <p className="text-sm text-muted-foreground">
   336→   336→              {formValuesRef.current?.title && `Crafting first ${SCENES_PER_PHASE} scenes for "${formValuesRef.current.title}"`}
   337→   337→            </p>
   338→   338→          </div>
   339→   339→
   340→   340→          {/* Elapsed time */}
   341→   341→          <div className="text-center">
   342→   342→            <span className="text-xs text-muted-foreground font-mono">
   343→   343→              Elapsed {formatTime(elapsed)}
   344→   344→            </span>
   345→   345→          </div>
   346→   346→
   347→   347→          {/* Progress bar */}
   348→   348→          <div className="relative h-2 w-full overflow-hidden rounded-full bg-primary/20">
   349→   349→            <div className="absolute inset-0 animate-indeterminate-progress rounded-full bg-primary" />
   350→   350→          </div>
   351→   351→
   352→   352→          {/* Stages */}
   353→   353→          <div className="space-y-3">
   354→   354→            {GENERATION_STAGES.map((stage, index) => {
   355→   355→              const isCompleted = index < currentStage;
   356→   356→              const isActive = index === currentStage;
   357→   357→
   358→   358→              return (
   359→   359→                <motion.div
   360→   360→                  key={index}
   361→   361→                  initial={{ opacity: 0, x: -10 }}
   362→   362→                  animate={{ opacity: 1, x: 0 }}
   363→   363→                  transition={{ delay: index * 0.05 }}
   364→   364→                  className="flex items-center gap-3 text-sm rounded-lg px-3 py-2 transition-colors"
   365→   365→                  style={{
   366→   366→                    backgroundColor: isActive ? 'hsl(var(--primary) / 0.06)' : 'transparent',
   367→   367→                  }}
   368→   368→                >
   369→   369→                  <div className="flex h-6 w-6 shrink-0 items-center justify-center">
   370→   370→                    {isCompleted ? (
   371→   371→                      <motion.div
   372→   372→                        initial={{ scale: 0 }}
   373→   373→                        animate={{ scale: 1 }}
   374→   374→                        transition={{ type: 'spring', stiffness: 500, damping: 30 }}
   375→   375→                      >
   376→   376→                        <Check className="h-4 w-4 text-emerald-500" />
   377→   377→                      </motion.div>
   378→   378→                    ) : isActive ? (
   379→   379→                      <motion.div
   380→   380→                        className="h-2.5 w-2.5 rounded-full bg-primary"
   381→   381→                        animate={{ scale: [1, 1.5, 1], opacity: [0.5, 1, 0.5] }}
   382→   382→                        transition={{ repeat: Infinity, duration: 1.2 }}
   383→   383→                      />
   384→   384→                    ) : (
   385→   385→                      <div className="h-2.5 w-2.5 rounded-full bg-muted-foreground/30" />
   386→   386→                    )}
   387→   387→                  </div>
   388→   388→                  <span
   389→   389→                    className={
   390→   390→                      isCompleted
   391→   391→                        ? 'text-emerald-600 dark:text-emerald-400'
   392→   392→                        : isActive
   393→   393→                          ? 'font-medium text-foreground'
   394→   394→                          : 'text-muted-foreground'
   395→   395→                    }
   396→   396→                  >
   397→   397→                    {isCompleted ? stage.completed : stage.active}
   398→   398→                  </span>
   399→   399→                </motion.div>
   400→   400→              );
   401→   401→            })}
   402→   402→          </div>
   403→   403→
   404→   404→          {/* Tip */}
   405→   405→          <p className="text-center text-xs text-muted-foreground/60">
   406→   406→            Generating Phase 1 of {totalPhases}. Remaining phases can be generated from the editor.
   407→   407→          </p>
   408→   408→        </motion.div>
   409→   409→
   410→   410→        {/* Indeterminate progress bar animation */}
   411→   411→        <style jsx global>{`
   412→   412→          @keyframes indeterminate-progress {
   413→   413→            0% { transform: translateX(-100%); width: 40%; }
   414→   414→            50% { transform: translateX(100%); width: 60%; }
   415→   415→            100% { transform: translateX(200%); width: 40%; }
   416→   416→          }
   417→   417→          .animate-indeterminate-progress {
   418→   418→            animation: indeterminate-progress 1.8s ease-in-out infinite;
   419→   419→          }
   420→   420→        `}</style>
   421→   421→      </div>
   422→   422→    );
   423→   423→  }
   424→   424→
   425→   425→  // ─── FORM VIEW (default) ─────────────────────────────────────────────────
   426→   426→  return (
   427→   427→    <motion.div
   428→   428→      initial={{ opacity: 0, y: 20 }}
   429→   429→      animate={{ opacity: 1, y: 0 }}
   430→   430→      transition={{ duration: 0.4, ease: 'easeOut' }}
   431→   431→      className="flex min-h-full items-start justify-center px-4 py-8 md:items-center md:py-12"
   432→   432→    >
   433→   433→      <Card className="w-full max-w-2xl shadow-lg">
   434→   434→        <CardHeader className="pb-6">
   435→   435→          <div className="flex items-center gap-3">
   436→   436→            <Button
   437→   437→              variant="ghost"
   438→   438→              size="icon"
   439→   439→              onClick={() => router.push('/dashboard')}
   440→   440→              className="shrink-0"
   441→   441→              aria-label="Back to dashboard"
   442→   442→            >
   443→   443→              <ArrowLeft className="h-5 w-5" />
   444→   444→            </Button>
   445→   445→            <div className="flex items-center gap-3">
   446→   446→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10">
   447→   447→                <Film className="h-5 w-5 text-primary" />
   448→   448→              </div>
   449→   449→              <div>
   450→   450→                <CardTitle className="text-xl font-semibold tracking-tight">
   451→   451→                  Create New Project
   452→   452→                </CardTitle>
   453→   453→                <p className="mt-0.5 text-sm text-muted-foreground">
   454→   454→                  Transform your video idea into a production script
   455→   455→                </p>
   456→   456→              </div>
   457→   457→            </div>
   458→   458→          </div>
   459→   459→        </CardHeader>
   460→   460→
   461→   461→        <CardContent className="px-6 pb-8">
   462→   462→          <Form {...form}>
