     2→

     6→import {

    10→} from 'lucide-react';
    11→import { Button } from '@/components/ui/button';
    12→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    13→import { Input } from '@/components/ui/input';
    14→import { Badge } from '@/components/ui/badge';
    15→import { Separator } from '@/components/ui/separator';
    16→import {
    17→  Dialog,
    18→  DialogContent,
    19→  DialogHeader,
    20→  DialogTitle,
    21→  DialogDescription,
    22→  DialogFooter,
    23→} from '@/components/ui/dialog';
    24→import { Select, SelectTrigger, SelectValue, SelectContent, SelectItem } from '@/components/ui/select';
    25→import { useAuthStore } from '@/lib/auth-store';

    28→
    29→interface AdminUser {
    30→  id: string;
    31→  name: string;
    32→  email: string;
    33→  provider: string;
    34→  role: string;
    35→  plan: string;
    36→  isVerified: boolean;
    37→  isCustomPlan: boolean;
    38→  customPlan: { isCustom: boolean; customLabel: string; customDays: number };
    39→  planExpiresAt: number;
    40→  planDaysLeft: number;
    41→  stripe: { customerId: string; subscriptionId: string; currentPeriodEnd: number; cancelAtPeriodEnd: boolean };
    42→  dailyUsage: { date: string; projectsCreated: number; aiGenerations: number };
    43→  createdAt: string;

    46→
    47→function formatDate(timestamp: string | number): string {
    48→  if (!timestamp) return '—';
    49→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    50→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    51→}
    52→
    53→export default function AdminPage() {
    54→  const router = useRouter();
    55→  const user = useAuthStore((s) => s.user);
    56→  const [users, setUsers] = useState<AdminUser[]>([]);
    57→  const [loading, setLoading] = useState(true);
    58→  const [search, setSearch] = useState('');
    59→  const [editUser, setEditUser] = useState<AdminUser | null>(null);
    60→  const [saving, setSaving] = useState(false);
    61→  const [deleteUser, setDeleteUser] = useState<AdminUser | null>(null);
    62→  const [deleting, setDeleting] = useState(false);
    63→
    64→  // Edit form state
    65→  const [editPlan, setEditPlan] = useState('free');
    66→  const [editAddDays, setEditAddDays] = useState(0);
    67→  const [editIsCustom, setEditIsCustom] = useState(false);
    68→  const [editCustomLabel, setEditCustomLabel] = useState('');
    69→  const [editCustomDays, setEditCustomDays] = useState(0);
    70→  const [editRole, setEditRole] = useState('user');
    71→
    72→  // Redirect non-admin
    73→  useEffect(() => {
    74→    if (user && user.role !== 'admin') {
    75→      router.push('/dashboard');
    76→    }
    77→  }, [user, router]);
    78→
    79→  const fetchUsers = useCallback(async () => {
    80→    setLoading(true);
    81→    try {
    82→      const res = await fetch('/api/admin/users');
    83→      if (res.ok) {
    84→        const data = await res.json();
    85→        setUsers(data.users || []);
    86→      } else {
    87→        const data = await res.json();
    88→        toast.error(data.error || 'Failed to load users');
    89→      }
    90→    } catch {
    91→      toast.error('Network error. Please try again.');
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, []);
    96→
    97→  useEffect(() => {
    98→    if (user?.role === 'admin') {
    99→      fetchUsers();
   100→    }
   101→  }, [user, fetchUsers]);
   102→
   103→  // Filter users by search
   104→  const filteredUsers = users.filter((u) => {
   105→    if (!search.trim()) return true;
   106→    const q = search.toLowerCase();
   107→    return u.name.toLowerCase().includes(q) || u.email.toLowerCase().includes(q);
   108→  });
   109→
   110→  // Stats
   111→  const totalUsers = users.length;
   112→  const proUsers = users.filter((u) => u.plan === 'pro').length;
   113→  const customUsers = users.filter((u) => u.isCustomPlan).length;

   117→  const openEdit = (u: AdminUser) => {
   118→    setEditUser(u);
   119→    setEditPlan(u.plan);
   120→    setEditAddDays(0);
   121→    setEditIsCustom(u.isCustomPlan);
   122→    setEditCustomLabel(u.customPlan?.customLabel || '');
   123→    setEditCustomDays(u.customPlan?.customDays || 0);
   124→    setEditRole(u.role);
   125→  };
   126→
   127→  // Save user
   128→  const handleSave = async () => {
   129→    if (!editUser) return;
   130→    setSaving(true);
   131→    try {
   132→      const body: Record<string, unknown> = {
   133→        plan: editPlan,
   134→        role: editRole,
   135→        isCustomPlan: editIsCustom,
   136→        customPlan: {
   137→          isCustom: editIsCustom,
   138→          customLabel: editCustomLabel,
   139→          customDays: editCustomDays,
   140→        },
   141→      };
   142→      if (editAddDays > 0) {
   143→        body.addDays = editAddDays;
   144→      }
   145→
   146→      const res = await fetch(`/api/admin/users/${editUser.id}`, {
   147→        method: 'PUT',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify(body),
   150→      });
   151→      const data = await res.json();
   152→      if (res.ok) {
   153→        toast.success(`Updated ${editUser.name} successfully`);
   154→        setEditUser(null);
   155→        fetchUsers();
   156→      } else {
   157→        toast.error(data.error || 'Failed to update user');
   158→      }
   159→    } catch {
   160→      toast.error('Network error. Please try again.');
   161→    } finally {
   162→      setSaving(false);
   163→    }
   164→  };
   165→
   166→  // Delete user
   167→  const handleDelete = async () => {
   168→    if (!deleteUser) return;

   170→    try {
   171→      const res = await fetch(`/api/admin/users/${deleteUser.id}`, { method: 'DELETE' });
   172→      const data = await res.json();
   173→      if (res.ok) {
   174→        toast.success(`Deleted ${deleteUser.email}`);
   175→        setDeleteUser(null);
   176→        fetchUsers();
   177→      } else {
   178→        toast.error(data.error || 'Failed to delete user');
   179→      }
   180→    } catch {
   181→      toast.error('Network error. Please try again.');
   182→    } finally {
   183→      setDeleting(false);
   184→    }
   185→  };
   186→
   187→  // Quick add 30 days
   188→  const quickAddDays = async (u: AdminUser, days: number) => {

   190→      const res = await fetch(`/api/admin/users/${u.id}`, {
   191→        method: 'PUT',
   192→        headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ addDays: days }),
   194→      });
   195→      const data = await res.json();

   197→        toast.success(`Added ${days} days to ${u.name}`);
   198→        fetchUsers();
   199→      } else {
   200→        toast.error(data.error || 'Failed to add days');
   201→      }
   202→    } catch {
   203→      toast.error('Network error');
   204→    }
   205→  };
   206→
   207→  if (!user || user.role !== 'admin') return null;
   208→
   209→  return (
   210→    <div className="max-w-6xl mx-auto space-y-6">
   211→      {/* Header */}
   212→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   213→        <div className="flex items-center gap-3">
   214→          <Button variant="ghost" size="icon" onClick={() => router.push('/dashboard')} className="shrink-0">
   215→            <ChevronDown className="h-5 w-5 rotate-90" />
   216→          </Button>
   217→          <div>
   218→            <div className="flex items-center gap-2">
   219→              <ShieldCheck className="size-5 text-primary" />
   220→              <h1 className="text-2xl font-bold tracking-tight">Admin Panel</h1>
   221→            </div>
   222→            <p className="text-sm text-muted-foreground">Manage users, plans, and subscriptions</p>
   223→          </div>
   224→        </div>
   225→        <Button variant="outline" className="gap-2" onClick={fetchUsers} disabled={loading}>
   226→          <RefreshCw className={cn('size-4', loading && 'animate-spin')} />
   227→          Refresh
   228→        </Button>
   229→      </div>
   230→
   231→      {/* Stats Cards */}
   232→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   233→        {[
   234→          { label: 'Total Users', value: totalUsers, icon: Users, color: 'text-blue-500', bg: 'bg-blue-500/5' },
   235→          { label: 'Pro Users', value: proUsers, icon: Crown, color: 'text-primary', bg: 'bg-primary/5' },
   236→          { label: 'Custom Plans', value: customUsers, icon: Star, color: 'text-amber-500', bg: 'bg-amber-500/5' },
   237→          { label: 'Expired Plans', value: expiredUsers, icon: AlertTriangle, color: 'text-red-500', bg: 'bg-red-500/5' },
   238→        ].map((stat) => (
   239→          <Card key={stat.label}>
   240→            <CardContent className="pt-4 pb-3 px-4">
   241→              <div className="flex items-center justify-between">
   242→                <div>
   243→                  <p className="text-xs text-muted-foreground">{stat.label}</p>
   244→                  <p className="text-2xl font-bold mt-0.5">{stat.value}</p>
   245→                </div>
   246→                <div className={cn('size-9 rounded-lg flex items-center justify-center', stat.bg)}>
   247→                  <stat.icon className={cn('size-4', stat.color)} />
   248→                </div>
   249→              </div>
   250→            </CardContent>
   251→          </Card>
   252→        ))}
   253→      </div>
   254→
   255→      {/* Search */}
   256→      <Card>
   257→        <CardContent className="pt-4">
   258→          <div className="relative">
   259→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   260→            <Input
   261→              type="search"
   262→              placeholder="Search by name or email..."
   263→              className="pl-10 h-10"
   264→              value={search}
   265→              onChange={(e) => setSearch(e.target.value)}
   266→            />
   267→          </div>
   268→        </CardContent>
   269→      </Card>
   270→
   271→      {/* Users Table */}
   272→      <Card>
   273→        <CardHeader className="pb-3">
   274→          <CardTitle className="text-base flex items-center gap-2">
   275→            <Users className="size-4" />
   276→            All Users
   277→            <Badge variant="secondary" className="text-[10px]">{filteredUsers.length}</Badge>
   278→          </CardTitle>
   279→          <CardDescription>Click edit to change plans, add days, or set custom plans</CardDescription>
   280→        </CardHeader>
   281→        <CardContent className="p-0">
   282→          {loading ? (
   283→            <div className="flex items-center justify-center py-12">
   284→              <Loader2 className="size-6 animate-spin text-muted-foreground" />
   285→              <span className="ml-2 text-sm text-muted-foreground">Loading users...</span>
   286→            </div>
   287→          ) : filteredUsers.length === 0 ? (
   288→            <div className="text-center py-12 text-sm text-muted-foreground">
   289→              {search ? 'No users match your search.' : 'No users found.'}
   290→            </div>
   291→          ) : (
   292→            <div className="overflow-x-auto">
   293→              <table className="w-full text-sm">
   294→                <thead>
   295→                  <tr className="border-b bg-muted/30">
   296→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">User</th>
   297→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Plan</th>
   298→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Days Left</th>
   299→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Custom</th>
   300→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Usage</th>
   301→                    <th className="text-left px-4 py-2.5 font-medium text-muted-foreground">Joined</th>
   302→                    <th className="text-right px-4 py-2.5 font-medium text-muted-foreground">Actions</th>
   303→                  </tr>
   304→                </thead>
   305→                <tbody>
   306→                  {filteredUsers.map((u) => (
   307→                    <motion.tr
   308→                      key={u.id}
   309→                      layout
   310→                      initial={{ opacity: 0 }}
   311→                      animate={{ opacity: 1 }}
   312→                      className="border-b hover:bg-muted/20 transition-colors"
   313→                    >
   314→                      <td className="px-4 py-3">
   315→                        <div className="flex flex-col gap-0.5">
   316→                          <div className="flex items-center gap-2">
   317→                            <span className="font-medium text-sm">{u.name}</span>
   318→                            {u.role === 'admin' && (
   319→                              <Badge className="text-[9px] px-1.5 py-0 h-4">Admin</Badge>
   320→                            )}
   321→                            {!u.isVerified && (
   322→                              <Badge variant="secondary" className="text-[9px] px-1.5 py-0 h-4">Unverified</Badge>
   323→                            )}
   324→                          </div>
   325→                          <span className="text-xs text-muted-foreground">{u.email}</span>
   326→                        </div>
   327→                      </td>
   328→                      <td className="px-4 py-3">
   329→                        <div className="flex items-center gap-1.5">
   330→                          {u.plan === 'pro' ? (
   331→                            <>
   332→                              <Crown className="size-3.5 text-primary" />
   333→                              <span className="font-medium text-primary">Pro</span>
   334→                            </>
   335→                          ) : (
   336→                            <>
   337→                              <Zap className="size-3.5 text-muted-foreground" />
   338→                              <span className="text-muted-foreground">Free</span>
   339→                            </>
   340→                          )}
   341→                        </div>
   342→                        {u.stripe?.subscriptionId && (
   343→                          <span className="text-[10px] text-muted-foreground">Stripe</span>
   344→                        )}
   345→                      </td>
   346→                      <td className="px-4 py-3">
   347→                        {u.plan === 'pro' ? (
   348→                          <div className="flex items-center gap-1.5">
   349→                            <span className={cn(
   350→                              'text-sm font-semibold',
   351→                              u.planDaysLeft <= 3 ? 'text-red-500' : u.planDaysLeft <= 7 ? 'text-amber-500' : 'text-foreground'
   352→                            )}>
   353→                              {u.planDaysLeft}d
   354→                            </span>
   355→                            {u.planExpiresAt > 0 && (
   356→                              <span className="text-[10px] text-muted-foreground">
   357→                                ({formatDate(u.planExpiresAt)})
   358→                              </span>
   359→                            )}
   360→                          </div>
   361→                        ) : (
   362→                          <span className="text-xs text-muted-foreground">—</span>
   363→                        )}
   364→                      </td>
   365→                      <td className="px-4 py-3">
   366→                        {u.isCustomPlan ? (
   367→                          <Badge className="text-[10px] gap-1" variant="outline">
   368→                            <Star className="size-2.5" />
   369→                            {u.customPlan?.customLabel || 'Custom'}
   370→                          </Badge>
   371→                        ) : (
   372→                          <span className="text-xs text-muted-foreground">—</span>
   373→                        )}
   374→                      </td>
   375→                      <td className="px-4 py-3">
   376→                        <div className="text-xs text-muted-foreground">
   377→                          <span>{u.dailyUsage?.projectsCreated ?? 0} proj</span>
   378→                          <span className="mx-1">·</span>
   379→                          <span>{u.dailyUsage?.aiGenerations ?? 0} AI</span>
   380→                        </div>
   381→                      </td>
   382→                      <td className="px-4 py-3 text-xs text-muted-foreground">
   383→                        {formatDate(u.createdAt)}
   384→                      </td>
   385→                      <td className="px-4 py-3">
   386→                        <div className="flex items-center justify-end gap-1">
   387→                          {/* Quick add 30 days */}
   388→                          <Button
   389→                            variant="ghost"
   390→                            size="icon"
   391→                            className="size-7"
   392→                            title="Add 30 days"
   393→                            onClick={() => quickAddDays(u, 30)}
   394→                          >
   395→                            <Plus className="size-3" />
   396→                          </Button>
   397→                          {/* Edit */}
   398→                          <Button
   399→                            variant="ghost"
   400→                            size="icon"
   401→                            className="size-7"
   402→                            title="Edit user"
   403→                            onClick={() => openEdit(u)}
   404→                          >
   405→                            <Edit3 className="size-3" />
   406→                          </Button>
   407→                          {/* Delete */}
   408→                          <Button
   409→                            variant="ghost"
   410→                            size="icon"
   411→                            className="size-7 text-destructive hover:text-destructive"
   412→                            title="Delete user"
   413→                            onClick={() => setDeleteUser(u)}
   414→                            disabled={u.role === 'admin'}
   415→                          >
   416→                            <Trash2 className="size-3" />
   417→                          </Button>
   418→                        </div>
   419→                      </td>
   420→                    </motion.tr>
   421→                  ))}
   422→                </tbody>
   423→              </table>
   424→            </div>
   425→          )}
   426→        </CardContent>
   427→      </Card>
   428→
   429→      {/* ── Edit User Dialog ── */}
   430→      <Dialog open={!!editUser} onOpenChange={() => setEditUser(null)}>
   431→        <DialogContent className="max-w-md">
   432→          <DialogHeader>
   433→            <DialogTitle className="flex items-center gap-2">
   434→              <UserCog className="size-4" />
   435→              Edit User
   436→            </DialogTitle>
   437→            <DialogDescription>
   438→              Modify plan, add days, or set custom plan for {editUser?.name}
   439→            </DialogDescription>
   440→          </DialogHeader>
   441→
   442→          {editUser && (
   443→            <div className="space-y-4 py-2">
   444→              {/* Current info */}
   445→              <div className="rounded-lg bg-muted/50 p-3 space-y-1">
   446→                <div className="flex justify-between text-sm">
   447→                  <span className="text-muted-foreground">Email</span>
   448→                  <span className="font-medium text-xs">{editUser.email}</span>
   449→                </div>
   450→                <div className="flex justify-between text-sm">
   451→                  <span className="text-muted-foreground">Current Plan</span>
   452→                  <span className="font-medium">{editUser.plan === 'pro' ? 'Pro' : 'Free'}</span>
   453→                </div>
   454→                <div className="flex justify-between text-sm">
   455→                  <span className="text-muted-foreground">Days Remaining</span>
   456→                  <span className={cn('font-semibold', editUser.planDaysLeft <= 3 ? 'text-red-500' : 'text-foreground')}>
   457→                    {editUser.plan === 'pro' ? `${editUser.planDaysLeft} days` : '—'}
   458→                  </span>
   459→                </div>
   460→              </div>
   461→
   462→              <Separator />
   463→
   464→              {/* Plan selector */}
   465→              <div className="space-y-2">
   466→                <label className="text-sm font-medium">Plan</label>
   467→                <Select value={editPlan} onValueChange={setEditPlan}>
   468→                  <SelectTrigger className="w-full">
   469→                    <SelectValue />
   470→                  </SelectTrigger>
   471→                  <SelectContent>
   472→                    <SelectItem value="free">
   473→                      <div className="flex items-center gap-2">
   474→                        <Zap className="size-3.5 text-muted-foreground" />
   475→                        Free
   476→                      </div>
   477→                    </SelectItem>
   478→                    <SelectItem value="pro">
   479→                      <div className="flex items-center gap-2">
   480→                        <Crown className="size-3.5 text-primary" />

   486→              </div>

   488→              {/* Add days */}
   489→              <div className="space-y-2">
   490→                <label className="text-sm font-medium">Add Days</label>
   491→                <div className="flex items-center gap-2">
   492→                  <Button
   493→                    variant="outline"
   494→                    size="icon"
   495→                    className="size-8"
   496→                    onClick={() => setEditAddDays(Math.max(0, editAddDays - 7))}
   497→                  >
   498→                    <Minus className="size-3" />
   499→                  </Button>
   500→                  <Input
   501→                    type="number"
   502→                    min={0}
   503→                    max={365}
   504→                    value={editAddDays}
   505→                    onChange={(e) => setEditAddDays(Math.max(0, parseInt(e.target.value) || 0))}
   506→                    className="w-24 text-center"
   507→                  />
   508→                  <span className="text-xs text-muted-foreground">days</span>
   509→                  <Button
   510→                    variant="outline"
   511→                    size="icon"
   512→                    className="size-8"
   513→                    onClick={() => setEditAddDays(Math.min(365, editAddDays + 7))}
   514→                  >
   515→                    <Plus className="size-3" />
   516→                  </Button>

   518→                <p className="text-[11px] text-muted-foreground">

   520→                </p>

   523→              {/* Custom plan toggle */}
