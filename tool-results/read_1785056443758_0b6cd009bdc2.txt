    43→    43→function loadSettings(): AppSettings {
    44→    44→  if (typeof window === 'undefined') return DEFAULT_SETTINGS;
    45→    45→  try {
    46→    46→    const raw = localStorage.getItem(SETTINGS_KEY);
    47→    47→    if (raw) return { ...DEFAULT_SETTINGS, ...JSON.parse(raw) };
    48→    48→  } catch {}
    49→    49→  return DEFAULT_SETTINGS;
    50→    50→}

    52→    52→function persistSettings(settings: AppSettings) {
    53→    53→  if (typeof window === 'undefined') return;
    54→    54→  try {
    55→    55→    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
    56→    56→  } catch {}
    57→    57→}
