    52→    52→function formatDate(timestamp: string | number): string {
    53→    53→  if (!timestamp) return '—';
    54→    54→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    55→    55→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    56→    56→}
