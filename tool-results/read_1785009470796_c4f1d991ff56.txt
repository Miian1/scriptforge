    78→  const s = seconds % 60;

   363→// ---- Main ScriptEditor ----
   364→export default function ScriptEditor() {
   365→  const projects = useAppStore((s) => s.projects);
   366→  const activeProjectId = useAppStore((s) => s.activeProjectId);
   367→  const scenes = useAppStore((s) => s.scenes);
   368→  const addScene = useAppStore((s) => s.addScene);
   369→  const addScenes = useAppStore((s) => s.addScenes);
   370→  const loadScenes = useAppStore((s) => s.loadScenes);
   371→  const loadCharacters = useAppStore((s) => s.loadCharacters);
   372→  const reorderScenes = useAppStore((s) => s.reorderScenes);
   373→  const updateProject = useAppStore((s) => s.updateProject);
   374→  const setActiveProjectId = useAppStore((s) => s.setActiveProjectId);
   375→  const generatingProjectId = useAppStore((s) => s.generatingProjectId);
   376→  const setGeneratingProjectId = useAppStore((s) => s.setGeneratingProjectId);
   377→  const router = useRouter();
   378→
   379→  const listEndRef = useRef<HTMLDivElement>(null);
   380→
   381→  // Local generating state for phases
   382→  const [isGeneratingPhase, setIsGeneratingPhase] = useState(false);
   383→  const [generatingPhaseLabel, setGeneratingPhaseLabel] = useState('');
   384→
   385→  // Batch voice generation dialog state
   386→  const [voiceDialogOpen, setVoiceDialogOpen] = useState(false);
   387→  const [batchGenerating, setBatchGenerating] = useState(false);
   388→  const [batchProgress, setBatchProgress] = useState({ current: 0, total: 0 });
   389→  const [batchSceneAudio, setBatchSceneAudio] = useState<Record<string, string>>({});
   390→
   391→  const project = projects.find((p) => p.id === activeProjectId) ?? null;
   392→
   393→  // Computed phase info from project settings
   394→  const totalScenes = project?.settings?.totalScenes || 60;
   395→  const scenesPerPhase = project?.settings?.scenesPerPhase || SCENES_PER_PHASE;
   396→  const totalPhases = Math.ceil(totalScenes / scenesPerPhase);
   397→  const currentPhaseNumber = Math.min(
   398→    Math.ceil(scenes.length / scenesPerPhase) + (scenes.length % scenesPerPhase === 0 ? 0 : 1),
   399→    totalPhases
   400→  );
   401→  // Actually: how many full phases are done?
   402→  const completedPhases = Math.floor(scenes.length / scenesPerPhase);
   403→  const isOnPartialPhase = scenes.length % scenesPerPhase !== 0;
   404→  const activePhaseNum = isOnPartialPhase ? completedPhases + 1 : completedPhases;
   405→  const allPhasesDone = scenes.length >= totalScenes;
   406→  const hasNextPhase = !allPhasesDone && scenes.length > 0;
   407→  const nextPhaseNumber = hasNextPhase
   408→    ? (scenes.length % scenesPerPhase === 0
   409→        ? scenes.length / scenesPerPhase + 1
   410→        : Math.floor(scenes.length / scenesPerPhase) + 1)
   411→    : 0;
   412→
   413→  // Group scenes into phases for display
   414→  const phases = useMemo(() => {
   415→    const grouped: { phase: number; scenes: Scene[] }[] = [];
   416→    const sorted = [...scenes].sort((a, b) => a.sceneNumber - b.sceneNumber);
   417→    for (let i = 0; i < sorted.length; i++) {
   418→      const phaseNum = Math.floor(sorted[i].sceneNumber / scenesPerPhase) + 1;
   419→      const lastGroup = grouped[grouped.length - 1];
   420→      if (lastGroup && lastGroup.phase === phaseNum) {
   421→        lastGroup.scenes.push(sorted[i]);
   422→      } else {
   423→        grouped.push({ phase: phaseNum, scenes: [sorted[i]] });
   424→      }
   425→    }
   426→    return grouped;
   427→  }, [scenes, scenesPerPhase]);
   428→
   429→  // Load scenes and characters on mount
   430→  useEffect(() => {
   431→    if (activeProjectId) {
   432→      loadScenes(activeProjectId);
   433→      loadCharacters(activeProjectId);
   434→    }
   435→  }, [activeProjectId, loadScenes, loadCharacters]);
   436→
   437→  // DnD sensors
   438→  const sensors = useSensors(
   439→    useSensor(PointerSensor, {
   440→      activationConstraint: { distance: 8 },
   441→    }),
   442→    useSensor(KeyboardSensor)
   443→  );
   444→
   445→  // Handle drag end
   446→  const handleDragEnd = useCallback(
   447→    (event: DragEndEvent) => {
   448→      const { active, over } = event;
   449→      if (!over || active.id === over.id) return;
   450→
   451→      const oldIndex = scenes.findIndex((s) => s.id === active.id);
   452→      const newIndex = scenes.findIndex((s) => s.id === over.id);
   453→      if (oldIndex === -1 || newIndex === -1) return;
   454→
   455→      const reordered = arrayMove(scenes, oldIndex, newIndex);
   456→      reorderScenes(reordered);
   457→    },
   458→    [scenes, reorderScenes]
   459→  );
   460→
   461→  // Add new scene
   462→  const handleAddScene = useCallback(async () => {
   463→    if (!project) return;
   464→    const now = Date.now();
   465→    const sceneLen = project.settings.sceneLength || 8;
   466→    const newScene: Scene = {
   467→      id: uuidv4(),
   468→      projectId: project.id,
   469→      sceneNumber: scenes.length + 1,
   470→      title: 'New Scene',
   471→      estimatedDuration: sceneLen,
   472→      goal: '',
   473→      narration: '',
   474→      imagePrompt: '',
   475→      animationPrompt: '',
   476→      characterIds: [],
   477→      notes: {
   478→        emotion: '',
   479→        visualFocus: '',
   480→        transitionSuggestion: '',
   481→        importantDetails: '',
   482→      },
   483→      createdAt: now,
   484→      updatedAt: now,
   485→    };
   486→    await addScene(newScene);
   487→
   488→    requestAnimationFrame(() => {
   489→      listEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   490→    });
   491→
   492→    toast.success('Scene added');
   493→  }, [project, scenes.length, addScene]);
   494→
   495→  // Generate next phase
   496→  const handleGenerateNextPhase = useCallback(async () => {
   497→    if (!project || isGeneratingPhase) return;
   498→
   499→    const phaseNum = nextPhaseNumber;
   500→    const sceneStart = scenes.length + 1;
   501→    const scenesInPhase = Math.min(scenesPerPhase, totalScenes - scenes.length);
   502→    const sceneEnd = sceneStart + scenesInPhase - 1;
   503→
   504→    // Get titles of existing scenes for continuity
   505→    const existingTitles = scenes
   506→      .sort((a, b) => a.sceneNumber - b.sceneNumber)
   507→      .map((s) => s.title);
   508→
   509→    setIsGeneratingPhase(true);
   510→    setGeneratingPhaseLabel(`Phase ${phaseNum} (Scenes ${sceneStart}-${sceneEnd})`);
   511→    setGeneratingProjectId(project.id);
   512→    await updateProject(project.id, { status: 'generating' });
   513→
   514→    try {
   515→      const channelNiche = useAuthStore.getState().user?.channelNiche as ChannelNiche | undefined;
   516→
   517→      const result = await generatePhase(project, {
   518→        phaseNumber: phaseNum,
   519→        totalPhases,
   520→        sceneStart,
   521→        sceneEnd,
   522→        previousPhaseTitles: existingTitles,
   523→      }, channelNiche || null);
   524→
   525→      const scenesWithId = result.scenes.map((s) => ({ ...s, projectId: project.id }));
   526→
   527→      // Save metadata if first phase somehow missed it
   528→      if (result.metadata) {
   529→        await updateProject(project.id, {
   530→          description: result.metadata.videoDescription || project.description || '',
   531→          thumbnailPrompt: result.metadata.thumbnailPrompt || '',
   532→          tags: result.metadata.tags || [],
   533→        });
   534→      }
   535→
   536→      await addScenes(scenesWithId);
   537→      await updateProject(project.id, { status: 'completed' });
   538→      await loadScenes(project.id);
   539→      toast.success(`Phase ${phaseNum} generated! ${scenesInPhase} scenes added.`);
   540→
   541→      // Scroll to the new scenes
   542→      requestAnimationFrame(() => {
   543→        listEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   544→      });
   545→    } catch (err) {
   546→      await updateProject(project.id, { status: 'error' });
   547→      toast.error((err as Error).message || `Failed to generate Phase ${phaseNum}`);
   548→    } finally {
   549→      setIsGeneratingPhase(false);
   550→      setGeneratingPhaseLabel('');
   551→      setGeneratingProjectId(null);
   552→    }
   553→  }, [project, scenes, nextPhaseNumber, scenesPerPhase, totalScenes, totalPhases, isGeneratingPhase, addScenes, updateProject, loadScenes, setGeneratingProjectId]);
   554→
   555→  // Export all as JSON
   556→  const handleExport = useCallback(() => {
   557→    if (!project || scenes.length === 0) return;
   558→    const exportData = {
   559→      project: {
   560→        title: project.title,
   561→        topic: project.topic,
   562→        description: project.description,
   563→        settings: project.settings,

   566→      scenes: scenes

   625→      setIsGeneratingPhase(false);

   629→

   673→          {scenes.length > 0 && !isGenerating && (

   717→                    variant="outline"

   719→                    onClick={() => setVoiceDialogOpen(true)}

   760→                    </Badge>
   761→                  </div>

   768→                </div>

   823→                          </div>
   824→                        </SortableContext>
   825→                      </DndContext>
   826→                    </div>
   827→                  ))}
   828→                </div>
   829→
   830→                {/* ---- Bottom actions ---- */}
   831→                <div className="mt-8">
   832→                  {hasNextPhase && !allPhasesDone && (
   833→                    <motion.div
   834→                      initial={{ opacity: 0, y: 12 }}
   835→                      animate={{ opacity: 1, y: 0 }}
   836→                      className="flex flex-col items-center gap-3"
   837→                    >
   838→                      <div className="rounded-xl border-2 border-dashed border-primary/30 bg-primary/[0.03] p-6 text-center w-full max-w-md space-y-4">
   839→                        <div className="flex items-center justify-center gap-2">
   840→                          <Zap className="size-5 text-primary" />
   841→                          <span className="text-sm font-semibold">
   842→                            Generate Next Phase

   872→
   873→                  {allPhasesDone && (
   874→                    <motion.div
   875→                      initial={{ opacity: 0, y: 12 }}
   876→                      animate={{ opacity: 1, y: 0 }}
   877→                      className="flex flex-col items-center gap-3"
   878→                    >
   879→                      <div className="rounded-xl border border-emerald-500/30 bg-emerald-500/5 p-4 text-center max-w-md">

   887→                      </div>
   888→                      <Button
   889→                        variant="outline"
   890→                        size="sm"
   891→                        onClick={handleAddScene}
   892→                        className="gap-1.5 text-muted-foreground"
   893→                      >
   894→                        <Plus className="size-3.5" />
   895→                        Add Scene
   896→                      </Button>
   897→                    </motion.div>
   898→                  )}
   899→                </div>
   900→
   901→                <div ref={listEndRef} />
   902→              </>
   903→            )}
   904→          </>
   905→        )}
   906→      </main>
   907→
   908→      {/* ── Batch Voice Generation Dialog ── */}
   909→      <Dialog open={voiceDialogOpen} onOpenChange={setVoiceDialogOpen}>
   910→        <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   911→          <DialogHeader>
   912→            <DialogTitle className="flex items-center gap-2">
   913→              <Volume2 className="size-5 text-primary" />
   914→              Generate All Narration Audio
   915→            </DialogTitle>
   916→            <DialogDescription>
   917→              Generate voice-over audio for all scene narrations in this project. Select a voice and settings below.
   918→            </DialogDescription>
   919→          </DialogHeader>
   920→
   921→          <div className="space-y-4 mt-2">
   922→            {/* Voice selector for all scenes */}
   923→            <div className="space-y-3">
   924→              <VoiceGenerator
   925→                text={scenes[0]?.narration || ''}
   926→                sceneId="preview"
   927→                label="Voice Preview"
   928→              />
   929→            </div>
   930→
   931→            <div className="rounded-lg border bg-muted/30 p-3 text-xs text-muted-foreground space-y-2">

   933→                <strong className="text-foreground">{scenes.length}</strong> scenes with narration will be generated.
   934→                Each scene&apos;s narration text will be converted to speech using the selected voice and settings above.
   935→              </p>
   936→              <p>
   937→                The same voice and settings will be applied to all scenes. You can also generate individual scene audio
   938→                from the voice controls on each scene card.
   939→              </p>
   940→            </div>
   941→
   942→            {/* Progress */}
   943→            {batchGenerating && (
   944→              <div className="space-y-2">
   945→                <div className="flex items-center justify-between text-xs">

   980→                        <Download className="size-3" />
   981→                        Scene {scene?.sceneNumber || '?'}

  1019→                      headers: { 'Content-Type': 'application/json' },

  1022→                        text: cleanedText,
  1023→                        instructions: voiceInstructions || undefined,

  1035→                  } catch {

  1038→                  setBatchProgress({ current: i + 1, total: narrationScenes.length });
  1039→                }
  1040→
  1041→                setBatchGenerating(false);
  1042→                toast.success(`${Object.keys(generated).length} audio files generated and saved!`);
  1043→              }}
  1044→              disabled={batchGenerating || scenes.length === 0}
  1045→              className="w-full gap-2"
  1046→            >
  1047→              {batchGenerating ? (
  1048→                <>
  1049→                  <Loader2 className="size-4 animate-spin" />
  1050→                  Generating All Audio ({batchProgress.current}/{batchProgress.total})...
  1051→                </>
  1052→              ) : (
  1053→                <>
  1054→                  <Sparkles className="size-4" />
  1055→                  Generate Audio for All {scenes.filter((s) => s.narration?.trim()).length} Scenes
  1056→                </>
  1057→              )}
  1058→            </Button>
  1059→          </div>
  1060→        </DialogContent>
  1061→      </Dialog>
  1062→    </div>
  1063→  );
