    50→function formatDate(timestamp: string | number): string {
    51→  if (!timestamp) return '—';
    52→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    53→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    54→}
    55→
    56→function formatExpiryDate(timestamp: number): string {
    57→  if (!timestamp || timestamp <= 0) return '—';
    58→  const date = new Date(timestamp);
    59→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    60→}
