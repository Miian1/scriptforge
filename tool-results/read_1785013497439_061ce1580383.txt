     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback, useRef, useMemo } from 'react';
     4→import {
     5→  Volume2,
     6→  Play,
     7→  Pause,
     8→  Square,
     9→  Loader2,
    10→  Settings2,
    11→  Mic,
    12→  Download,
    13→  Sparkles,
    14→  MessageSquare,
    15→  Filter,
    16→  Palette,
    17→  Gauge,
    18→  Globe,
    19→  ChevronDown,
    20→  X,
    21→} from 'lucide-react';
    22→import { motion, AnimatePresence } from 'framer-motion';
    23→import { toast } from 'sonner';
    24→
    25→import { Button } from '@/components/ui/button';
    26→import { Card, CardContent } from '@/components/ui/card';
    27→import { Label } from '@/components/ui/label';
    28→import { Separator } from '@/components/ui/separator';
    29→import {
    30→  Select,
    31→  SelectContent,
    32→  SelectItem,
    33→  SelectTrigger,
    34→  SelectValue,
    35→} from '@/components/ui/select';
    36→import { Textarea } from '@/components/ui/textarea';
    37→import { Badge } from '@/components/ui/badge';
    38→import {
    39→  Popover,
    40→  PopoverContent,
    41→  PopoverTrigger,
    42→} from '@/components/ui/popover';
    43→
    44→// ── Types ─────────────────────────────────────────────
    45→
    46→interface Voice {
    47→  name: string;
    48→  description: string;
    49→  category: string;
    50→}
    51→
    52→interface VoiceGeneratorProps {
    53→  text: string;
    54→  sceneId: string;
    55→  narrationAudioPath?: string;
    56→  label?: string;
    57→}
    58→
    59→// ── Strip stage directions client-side (for preview) ──
    60→
    61→function stripBrackets(text: string): string {
    62→  return text
    63→    .replace(/\[.*?\]/g, '')
    64→    .replace(/\(.*?\)/g, '')
    65→    .replace(/\*.*?\*/g, '')
    66→    .replace(/[\[\]\(\)]/g, '')
    67→    .replace(/\s{2,}/g, ' ')
    68→    .trim();
    69→}
    70→
    71→// ── Voice Generator Component ────────────────────────
    72→
    73→export default function VoiceGenerator({
    74→  text,
    75→  sceneId,
    76→  narrationAudioPath,
    77→  label = 'Narration',
    78→}: VoiceGeneratorProps) {
    79→  // Voice data from API
    80→  const [voices, setVoices] = useState<Voice[]>([]);
    81→  const [categories, setCategories] = useState<string[]>([]);
    82→  const [styles, setStyles] = useState<{ value: string; label: string }[]>([]);
    83→  const [paces, setPaces] = useState<{ value: string; label: string }[]>([]);
    84→  const [accents, setAccents] = useState<{ value: string; label: string }[]>([]);
    85→  const [voicesLoading, setVoicesLoading] = useState(false);
    86→
    87→  // Voice selection
    88→  const [selectedVoiceName, setSelectedVoiceName] = useState<string>('');
    89→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
    90→
    91→  // Style / Pace / Accent
    92→  const [selectedStyle, setSelectedStyle] = useState('');
    93→  const [selectedPace, setSelectedPace] = useState('');
    94→  const [selectedAccent, setSelectedAccent] = useState('');
    95→  const [customInstructions, setCustomInstructions] = useState('');
    96→  const [settingsOpen, setSettingsOpen] = useState(false);
    97→
    98→  // Playback
    99→  const [generating, setGenerating] = useState(false);
   100→  const [playing, setPlaying] = useState(false);
   101→  const [audioUrl, setAudioUrl] = useState<string | null>(null);
   102→  const [audioProgress, setAudioProgress] = useState(0);
   103→  const [audioDuration, setAudioDuration] = useState(0);
   104→  const [hasSavedAudio, setHasSavedAudio] = useState(false);
   105→  const audioRef = useRef<HTMLAudioElement | null>(null);
   106→
   107→  // Save voice preference per browser session
   108→  useEffect(() => {
   109→    try {
   110→      const saved = sessionStorage.getItem('scriptforge_voice_settings');
   111→      if (saved) {
   112→        const parsed = JSON.parse(saved);
   113→        if (parsed.voiceName) setSelectedVoiceName(parsed.voiceName);
   114→        if (parsed.style) setSelectedStyle(parsed.style);
   115→        if (parsed.pace) setSelectedPace(parsed.pace);
   116→        if (parsed.accent) setSelectedAccent(parsed.accent);
   117→        if (parsed.customInstructions) setCustomInstructions(parsed.customInstructions);
   118→      }
   119→    } catch {}
   120→  }, []);
   121→
   122→  // Load saved audio on mount
   123→  useEffect(() => {
   124→    if (narrationAudioPath) {
   125→      setHasSavedAudio(true);
   126→      setAudioUrl(narrationAudioPath);
   127→    }
   128→  }, [narrationAudioPath]);
   129→
   130→  // Cleanup audio on unmount
   131→  useEffect(() => {
   132→    return () => {
   133→      if (audioRef.current) {
   134→        audioRef.current.pause();
   135→      }
   136→    };
   137→  }, []);
   138→
   139→  const savePreference = useCallback((settings: Record<string, string>) => {
   140→    try {
   141→      sessionStorage.setItem('scriptforge_voice_settings', JSON.stringify(settings));
   142→    } catch {}
   143→  }, []);
   144→
   145→  // Fetch voices on mount
   146→  useEffect(() => {
   147→    async function fetchVoices() {
   148→      setVoicesLoading(true);
   149→      try {
   150→        const res = await fetch('/api/tts/voices');
   151→        if (res.ok) {
   152→          const data = await res.json();
   153→          const voiceList: Voice[] = data.voices || [];
   154→          setVoices(voiceList);
   155→          setCategories(data.categories || []);
   156→          setStyles(data.styles || []);
   157→          setPaces(data.paces || []);
   158→          setAccents(data.accents || []);
   159→          if (!selectedVoiceName && voiceList.length > 0) {
   160→            setSelectedVoiceName(voiceList[0].name);
   161→          }
   162→        } else {
   163→          const data = await res.json().catch(() => ({}));
   164→          toast.error((data as Record<string, string>).error || 'Failed to load voices');
   165→        }
   166→      } catch {
   167→        toast.error('Failed to connect to voice service');
   168→      } finally {
   169→        setVoicesLoading(false);
   170→      }
   171→    }
   172→    fetchVoices();
   173→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   174→
   175→  // Build instructions from all dropdown selections + custom text
   176→  const builtInstructions = useMemo(() => {
   177→    const parts: string[] = [];
   178→    if (selectedStyle) parts.push(selectedStyle);
   179→    if (selectedPace === 'very_slow') parts.push('speak very slowly');
   180→    else if (selectedPace === 'slow') parts.push('speak slowly');
   181→    else if (selectedPace === 'moderate') parts.push('speak at a moderate pace');
   182→    else if (selectedPace === 'fast') parts.push('speak quickly');
   183→    else if (selectedPace === 'very_fast') parts.push('speak very quickly');
   184→    if (selectedAccent === 'american') parts.push('with an American accent');
   185→    else if (selectedAccent === 'british') parts.push('with a British accent');
   186→    else if (selectedAccent === 'australian') parts.push('with an Australian accent');
   187→    else if (selectedAccent === 'neutral') parts.push('with a neutral accent');
   188→    if (customInstructions.trim()) parts.push(customInstructions.trim());
   189→    return parts.join(', ');
   190→  }, [selectedStyle, selectedPace, selectedAccent, customInstructions]);
   191→
   192→  // Filter voices by category
   193→  const filteredVoices = useMemo(() => {
   194→    if (!categoryFilter || categoryFilter === 'all') return voices;
   195→    return voices.filter(v => v.category === categoryFilter);
   196→  }, [voices, categoryFilter]);
   197→
   198→  // Group filtered voices by description
   199→  const groupedVoices = useMemo(() => {
   200→    const descriptions = [...new Set(filteredVoices.map(v => v.description))].sort();
   201→    return descriptions.map(desc => ({
   202→      description: desc,
   203→      category: filteredVoices.find(v => v.description === desc)?.category || '',
   204→      voices: filteredVoices.filter(v => v.description === desc),
   205→    }));
   206→  }, [filteredVoices]);
   207→
   208→  // Active filters count
   209→  const activeFilterCount = useMemo(() => {
   210→    let count = 0;
   211→    if (selectedStyle) count++;
   212→    if (selectedPace) count++;
   213→    if (selectedAccent) count++;
   214→    return count;
   215→  }, [selectedStyle, selectedPace, selectedAccent]);
   216→
   217→  // Setup audio player from URL
   218→  const setupAudioPlayer = useCallback((url: string, autoPlay = false) => {
   219→    if (audioRef.current) {
   220→      audioRef.current.pause();
   221→    }
   222→
   223→    const audio = new Audio(url);
   224→    audioRef.current = audio;
   225→
   226→    audio.addEventListener('loadedmetadata', () => {
   227→      setAudioDuration(audio.duration);
   228→    });
   229→    audio.addEventListener('timeupdate', () => {
   230→      setAudioProgress(audio.currentTime / (audio.duration || 1));
   231→    });
   232→    audio.addEventListener('ended', () => {
   233→      setPlaying(false);
   234→      setAudioProgress(0);
   235→    });
   236→    audio.addEventListener('error', () => {
   237→      toast.error('Audio playback error');
   238→      setPlaying(false);
   239→    });
   240→
   241→    if (autoPlay) {
   242→      audio.play().then(() => setPlaying(true)).catch(() => {});
   243→    }
   244→  }, []);
   245→
   246→  // Generate speech
   247→  const handleGenerate = useCallback(async () => {
   248→    if (!selectedVoiceName) {
   249→      toast.error('Please select a voice first');
   250→      return;
   251→    }
   252→    if (!text.trim()) {
   253→      toast.error('No text to generate speech from');
   254→      return;
   255→    }
   256→
   257→    const cleanText = stripBrackets(text);
   258→    if (!cleanText) {
   259→      toast.error('No spoken text after removing stage directions');
   260→      return;
   261→    }
   262→
   263→    // Stop any current playback
   264→    if (audioRef.current) {
   265→      audioRef.current.pause();
   266→      audioRef.current = null;
   267→    }
   268→    setPlaying(false);
   269→    setAudioProgress(0);
   270→    setAudioDuration(0);
   271→
   272→    setGenerating(true);
   273→    try {
   274→      const res = await fetch('/api/tts/generate', {
   275→        method: 'POST',
   276→        headers: { 'Content-Type': 'application/json' },
   277→        body: JSON.stringify({
   278→          voiceName: selectedVoiceName,
   279→          text: cleanText,
   280→          instructions: builtInstructions || undefined,
   281→          saveAudio: sceneId !== 'preview',
   282→          sceneId: sceneId !== 'preview' ? sceneId : undefined,
   283→        }),
   284→      });
   285→
   286→      if (!res.ok) {
   287→        const data = await res.json().catch(() => ({}));
   288→        throw new Error((data as Record<string, string>).error || 'Generation failed');
   289→      }
   290→
   291→      const serverAudioPath = res.headers.get('X-Audio-Path');
   292→
   293→      let url: string;
   294→      if (serverAudioPath) {
   295→        url = serverAudioPath;
   296→        setHasSavedAudio(true);
   297→        toast.success('Voice generated and saved!');
   298→      } else {
   299→        const blob = await res.blob();
   300→        url = URL.createObjectURL(blob);
   301→        toast.success('Voice generated!');
   302→      }
   303→
   304→      setAudioUrl(url);
   305→      setupAudioPlayer(url, true);
   306→    } catch (err) {
   307→      toast.error((err as Error).message || 'Failed to generate voice');
   308→    } finally {
   309→      setGenerating(false);
   310→    }
   311→  }, [selectedVoiceName, text, builtInstructions, sceneId, setupAudioPlayer]);
   312→
   313→  // Playback controls
   314→  const togglePlayback = useCallback(() => {
   315→    if (!audioUrl) return;
   316→    if (audioRef.current) {
   317→      if (playing) {
   318→        audioRef.current.pause();
   319→        setPlaying(false);
   320→      } else {
   321→        audioRef.current.play();
   322→        setPlaying(true);
   323→      }
   324→    } else {
   325→      setupAudioPlayer(audioUrl, true);
   326→    }
   327→  }, [playing, audioUrl, setupAudioPlayer]);
   328→
   329→  const stopPlayback = useCallback(() => {
   330→    if (!audioRef.current) return;
   331→    audioRef.current.pause();
   332→    audioRef.current.currentTime = 0;
   333→    setPlaying(false);
   334→    setAudioProgress(0);
   335→  }, []);
   336→
   337→  const handleDownload = useCallback(() => {
   338→    if (!audioUrl) return;
   339→    const a = document.createElement('a');
   340→    a.href = audioUrl;
   341→    a.download = `narration-${sceneId}.wav`;
   342→    document.body.appendChild(a);
   343→    a.click();
   344→    document.body.removeChild(a);
   345→    toast.success('Audio downloaded');
   346→  }, [audioUrl, sceneId]);
   347→
   348→  // Handlers that save preferences
   349→  const handleVoiceChange = useCallback((voiceName: string) => {
   350→    setSelectedVoiceName(voiceName);
   351→    savePreference({
   352→      voiceName,
   353→      style: selectedStyle,
   354→      pace: selectedPace,
   355→      accent: selectedAccent,
   356→      customInstructions,
   357→    });
   358→  }, [selectedStyle, selectedPace, selectedAccent, customInstructions, savePreference]);
   359→
   360→  const handleStyleChange = useCallback((value: string) => {
   361→    setSelectedStyle(value);
   362→    savePreference({
   363→      voiceName: selectedVoiceName,
   364→      style: value,
   365→      pace: selectedPace,
   366→      accent: selectedAccent,
   367→      customInstructions,
   368→    });
   369→  }, [selectedVoiceName, selectedPace, selectedAccent, customInstructions, savePreference]);
   370→
   371→  const handlePaceChange = useCallback((value: string) => {
   372→    setSelectedPace(value);
   373→    savePreference({
   374→      voiceName: selectedVoiceName,
   375→      style: selectedStyle,
   376→      pace: value,
   377→      accent: selectedAccent,
   378→      customInstructions,
   379→    });
   380→  }, [selectedVoiceName, selectedStyle, selectedAccent, customInstructions, savePreference]);
   381→
   382→  const handleAccentChange = useCallback((value: string) => {
   383→    setSelectedAccent(value);
   384→    savePreference({
   385→      voiceName: selectedVoiceName,
   386→      style: selectedStyle,
   387→      pace: selectedPace,
   388→      accent: value,
   389→      customInstructions,
   390→    });
   391→  }, [selectedVoiceName, selectedStyle, selectedPace, customInstructions, savePreference]);
   392→
   393→  const handleCustomInstructionsChange = useCallback((value: string) => {
   394→    setCustomInstructions(value);
   395→    savePreference({
   396→      voiceName: selectedVoiceName,
   397→      style: selectedStyle,
   398→      pace: selectedPace,
   399→      accent: selectedAccent,
   400→      customInstructions: value,
   401→    });
   402→  }, [selectedVoiceName, selectedStyle, selectedPace, selectedAccent, savePreference]);
   403→
   404→  // Format time
   405→  const formatTime = (seconds: number) => {
   406→    const m = Math.floor(seconds / 60);
   407→    const s = Math.floor(seconds % 60);
   408→    return `${m}:${s.toString().padStart(2, '0')}`;
   409→  };
   410→
   411→  const hasAudio = !!audioUrl;
   412→  const noText = !text.trim();
   413→
   414→  return (
   415→    <div className="space-y-3">
   416→      {/* Voice Selection + Controls Row */}
   417→      <div className="flex items-center gap-2 flex-wrap">
   418→        {/* Category Filter */}
   419→        <Popover>
   420→          <PopoverTrigger asChild>
   421→            <Button
   422→              variant="outline"
   423→              size="sm"
   424→              className="h-8 text-xs gap-1.5 shrink-0"
   425→              disabled={voicesLoading}
   426→            >
   427→              <Filter className="size-3" />
   428→              {categoryFilter === 'all' ? 'All Voices' : categoryFilter}
   429→            </Button>
   430→          </PopoverTrigger>
   431→          <PopoverContent className="w-48 p-2" align="start">
   432→            <div className="space-y-1">
   433→              <button
   434→                onClick={() => setCategoryFilter('all')}
   435→                className={`w-full text-left px-2 py-1.5 rounded text-xs transition-colors ${
   436→                  categoryFilter === 'all' ? 'bg-primary text-primary-foreground' : 'hover:bg-muted'
   437→                }`}
   438→              >
   439→                All Voices ({voices.length})
   440→              </button>
   441→              {categories.map(cat => {
   442→                const count = voices.filter(v => v.category === cat).length;
   443→                return (
   444→                  <button
   445→                    key={cat}
   446→                    onClick={() => setCategoryFilter(cat)}
   447→                    className={`w-full text-left px-2 py-1.5 rounded text-xs transition-colors ${
   448→                      categoryFilter === cat ? 'bg-primary text-primary-foreground' : 'hover:bg-muted'
   449→                    }`}
   450→                  >
   451→                    {cat} ({count})
   452→                  </button>
   453→                );
   454→              })}
   455→            </div>
   456→          </PopoverContent>
   457→        </Popover>
   458→
   459→        {/* Voice Dropdown */}
   460→        <div className="flex-1 min-w-[200px]">
   461→          <Select value={selectedVoiceName} onValueChange={handleVoiceChange} disabled={voicesLoading}>
   462→            <SelectTrigger className="h-8 text-xs">
   463→              <Mic className="size-3 mr-1.5 shrink-0 text-muted-foreground" />
   464→              <SelectValue placeholder={voicesLoading ? 'Loading voices...' : 'Select voice'} />
   465→            </SelectTrigger>
   466→            <SelectContent>
   467→              {groupedVoices.map(group => (
   468→                <React.Fragment key={group.description}>
   469→                  <div className="px-2 py-1.5 text-[10px] font-semibold text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
   470→                    {group.description}
   471→                    <Badge variant="secondary" className="text-[9px] px-1 py-0">
   472→                      {group.category}
   473→                    </Badge>
   474→                  </div>
   475→                  {group.voices.map(voice => (
   476→                    <SelectItem key={voice.name} value={voice.name} className="text-xs">
   477→                      <span className="font-medium">{voice.name}</span>
   478→                    </SelectItem>
   479→                  ))}
   480→                </React.Fragment>
   481→              ))}
   482→              {filteredVoices.length === 0 && (
   483→                <div className="px-2 py-3 text-xs text-muted-foreground text-center">
   484→                  No voices in this category
   485→                </div>
   486→              )}
   487→            </SelectContent>
   488→          </Select>
   489→        </div>
   490→
   491→        {/* Generate Button */}
   492→        <Button
   493→          size="sm"
   494→          onClick={handleGenerate}
   495→          disabled={generating || noText || !selectedVoiceName}
   496→          className="gap-1.5 h-8 text-xs shrink-0"
   497→        >
   498→          {generating ? (
   499→            <>
   500→              <Loader2 className="size-3 animate-spin" />
   501→              Generating...
   502→            </>
   503→          ) : hasAudio ? (
   504→            <>
   505→              <Sparkles className="size-3" />
   506→              Regenerate
   507→            </>
   508→          ) : (
   509→            <>
   510→              <Volume2 className="size-3" />
   511→              Generate Voice
   512→            </>
   513→          )}
   514→        </Button>
   515→
   516→        {/* Playback Controls */}
   517→        {hasAudio && (
   518→          <div className="flex items-center gap-1">
   519→            <Button variant="outline" size="icon" className="size-8 shrink-0" onClick={togglePlayback} disabled={!hasAudio}>
   520→              {playing ? <Pause className="size-3" /> : <Play className="size-3 ml-0.5" />}
   521→            </Button>
   522→            <Button variant="outline" size="icon" className="size-8 shrink-0" onClick={stopPlayback} disabled={!hasAudio}>
   523→              <Square className="size-3" />
   524→            </Button>
   525→            <Button variant="outline" size="icon" className="size-8 shrink-0" onClick={handleDownload} disabled={!hasAudio}>
   526→              <Download className="size-3" />
   527→            </Button>
   528→          </div>
   529→        )}
   530→
   531→        {/* Saved indicator */}
   532→        {hasSavedAudio && (
   533→          <span className="text-[10px] text-emerald-500 font-medium flex items-center gap-0.5 shrink-0">
   534→            <Volume2 className="size-3" /> Saved
   535→          </span>
   536→        )}
   537→
   538→        {/* Settings Toggle */}
   539→        <Popover open={settingsOpen} onOpenChange={setSettingsOpen}>
   540→          <PopoverTrigger asChild>
   541→            <Button
   542→              variant="ghost"
   543→              size="icon"
   544→              className="size-8 shrink-0 relative"
   545→            >
   546→              <Settings2 className="size-3.5 text-muted-foreground" />
   547→              {activeFilterCount > 0 && (
   548→                <span className="absolute -top-0.5 -right-0.5 size-3 bg-primary rounded-full flex items-center justify-center text-[8px] text-primary-foreground font-bold">
   549→                  {activeFilterCount}
   550→                </span>
   551→              )}
   552→            </Button>
   553→          </PopoverTrigger>
   554→          <PopoverContent className="w-80 p-0" align="end">
   555→            <div className="p-3 border-b flex items-center justify-between">
   556→              <span className="text-sm font-medium flex items-center gap-1.5">
   557→                <Settings2 className="size-3.5" />
   558→                Voice Settings
   559→              </span>
   560→              {activeFilterCount > 0 && (
   561→                <Button
   562→                  variant="ghost"
   563→                  size="sm"
   564→                  className="h-6 text-[10px] text-muted-foreground"
   565→                  onClick={() => {
   566→                    setSelectedStyle('');
   567→                    setSelectedPace('');
   568→                    setSelectedAccent('');
   569→                    setCustomInstructions('');
   570→                    savePreference({ voiceName: selectedVoiceName, style: '', pace: '', accent: '', customInstructions: '' });
   571→                  }}
   572→                >
   573→                  Reset All
   574→                </Button>
   575→              )}
   576→            </div>
   577→
   578→            <div className="p-3 space-y-3">
   579→              {/* Style Dropdown */}
   580→              <div className="space-y-1.5">
   581→                <Label className="text-xs font-medium flex items-center gap-1.5">
   582→                  <Palette className="size-3 text-purple-500" />
   583→                  Style
   584→                </Label>
   585→                <Select value={selectedStyle} onValueChange={handleStyleChange}>
   586→                  <SelectTrigger className="h-8 text-xs">
   587→                    <SelectValue placeholder="Select style" />
   588→                  </SelectTrigger>
   589→                  <SelectContent>
   590→                    {styles.map(s => (
   591→                      <SelectItem key={s.value || 'default'} value={s.value || 'default'}>
   592→                        {s.label}
   593→                      </SelectItem>
   594→                    ))}
   595→                  </SelectContent>
   596→                </Select>
   597→              </div>
   598→
   599→              {/* Pace Dropdown */}
   600→              <div className="space-y-1.5">
   601→                <Label className="text-xs font-medium flex items-center gap-1.5">
   602→                  <Gauge className="size-3 text-blue-500" />
   603→                  Pace
