     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback, useMemo } from 'react';
     4→import { motion } from 'framer-motion';
     5→import { Film, ArrowLeft, Sparkles, Check, AlertCircle, RefreshCw, Layers, Clock, Calculator } from 'lucide-react';
     6→import { toast } from 'sonner';
     7→import { z } from 'zod';
     8→import { useForm } from 'react-hook-form';
     9→import { zodResolver } from '@hookform/resolvers/zod';
    10→
    11→import { useAppStore } from '@/lib/store';
    12→import { generatePhase, type ChannelNiche } from '@/lib/gemini';
    13→import type { Project, VideoTheme, WritingStyle, TargetAudience, VideoLanguage, VideoDuration } from '@/lib/types';
    14→import { useAuthStore } from '@/lib/auth-store';
    15→import {
    16→  DURATION_LABELS,
    17→  DURATION_SECONDS,
    18→  SCENES_PER_PHASE,
    19→  THEME_LABELS,
    20→  STYLE_LABELS,
    21→  AUDIENCE_LABELS,
    22→  LANGUAGE_LABELS,
    23→} from '@/lib/types';
    24→import { useRouter } from 'next/navigation';
    25→
    26→import { Button } from '@/components/ui/button';
    27→import { Input } from '@/components/ui/input';
    28→import { Textarea } from '@/components/ui/textarea';
    29→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    30→import { Separator } from '@/components/ui/separator';
    31→import { Badge } from '@/components/ui/badge';
    32→import {
    33→  Form,
    34→  FormControl,
    35→  FormField,
    36→  FormItem,
    37→  FormLabel,
    38→  FormMessage,
    39→} from '@/components/ui/form';
    40→import {
    41→  Select,
    42→  SelectContent,
    43→  SelectItem,
    44→  SelectTrigger,
    45→  SelectValue,
    46→} from '@/components/ui/select';
    47→
    48→// ─── Schema ───────────────────────────────────────────────────────────────────
    49→
    50→const formSchema = z.object({
    51→  title: z.string().min(1, 'Title is required').max(100),
    52→  topic: z.string().min(1, 'Topic is required').max(200),
    53→  description: z.string().max(1000).optional().default(''),
    54→  duration: z.enum(['short', 'medium', 'long']),
    55→  sceneLengthMode: z.enum(['default', 'custom']),
    56→  customSceneLength: z.number().min(3).max(60).optional().default(undefined),
    57→  theme: z.enum(['realistic', 'anime', 'cinematic', 'cartoon', '3d-render', 'watercolor', 'pixel-art']),
    58→  language: z.enum(['english', 'spanish', 'french', 'german', 'portuguese', 'japanese', 'korean', 'chinese', 'hindi', 'arabic']),
    59→  writingStyle: z.enum(['conversational', 'professional', 'dramatic', 'educational', 'storytelling', 'humorous']),
    60→  targetAudience: z.enum(['beginners', 'general', 'intermediate', 'experts', 'kids', 'teens']),
    61→});
    62→
    63→type FormValues = z.infer<typeof formSchema>;
    64→
    65→// ─── Generation Stages ────────────────────────────────────────────────────────
    66→
    67→const GENERATION_STAGES = [
    68→  { active: 'Researching topic...', completed: 'Research complete' },
    69→  { active: 'Building story structure...', completed: 'Story structured' },
    70→  { active: 'Generating Phase 1 scenes...', completed: 'Phase 1 generated' },
    71→  { active: 'Finalizing...', completed: 'Finalized' },
    72→] as const;
    73→
    74→const STAGE_INTERVAL_MS = 5000;
    75→
    76→// ─── Component ────────────────────────────────────────────────────────────────
    77→
    78→export default function CreateProject() {
    79→  const router = useRouter();
    80→  const {
    81→    addProject,
    82→    setActiveProjectId,
    83→    addScenes,
    84→    updateProject,
    85→    loadScenes,
    86→  } = useAppStore();
    87→
    88→  // Phase: 'form' | 'generating' | 'error'
    89→  const [phase, setPhase] = useState<'form' | 'generating' | 'error'>('form');
    90→  const [currentStage, setCurrentStage] = useState(-1);
    91→  const [errorMessage, setErrorMessage] = useState('');
    92→  const [elapsed, setElapsed] = useState(0);
    93→  const stageTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    94→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    95→  const projectIdRef = useRef<string | null>(null);
    96→  const formValuesRef = useRef<FormValues | null>(null);
    97→  const isSubmittingRef = useRef(false);
    98→
    99→  const form = useForm<FormValues>({
   100→    resolver: zodResolver(formSchema),
   101→    defaultValues: {
   102→      title: '',
   103→      topic: '',
   104→      description: '',
   105→      duration: 'medium',
   106→      sceneLengthMode: 'default',
   107→      customSceneLength: undefined,
   108→      theme: 'cinematic',
   109→      language: 'english',
   110→      writingStyle: 'conversational',
   111→      targetAudience: 'general',
   112→    },
   113→  });
   114→
   115→  // Watch duration and scene length for calculations
   116→  const watchDuration = form.watch('duration');
   117→  const watchSceneLengthMode = form.watch('sceneLengthMode');
   118→  const watchCustomSceneLength = form.watch('customSceneLength');
   119→
   120→  // Computed scene length
   121→  const sceneLength = watchSceneLengthMode === 'custom' && watchCustomSceneLength
   122→    ? watchCustomSceneLength
   123→    : 8;
   124→
   125→  // Computed totals
   126→  const totalSeconds = DURATION_SECONDS[watchDuration as VideoDuration];
   127→  const totalScenes = Math.ceil(totalSeconds / sceneLength);
   128→  const totalPhases = Math.ceil(totalScenes / SCENES_PER_PHASE);
   129→  const firstPhaseScenes = Math.min(SCENES_PER_PHASE, totalScenes);
   130→
   131→  // Cleanup timers on unmount
   132→  useEffect(() => {
   133→    return () => {
   134→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   135→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   136→    };
   137→  }, []);
   138→
   139→  const advanceStage = useCallback(() => {
   140→    setCurrentStage((prev) => {
   141→      if (prev >= GENERATION_STAGES.length - 1) return prev;
   142→      return prev + 1;
   143→    });
   144→  }, []);
   145→
   146→  const startGeneration = useCallback(async (values: FormValues) => {
   147→    // Block double submissions
   148→    if (isSubmittingRef.current) return;
   149→    isSubmittingRef.current = true;
   150→
   151→    const now = Date.now();
   152→    const sLen = values.sceneLengthMode === 'custom' && values.customSceneLength
   153→      ? values.customSceneLength
   154→      : 8;
   155→    const totalSec = DURATION_SECONDS[values.duration as VideoDuration];
   156→    const tScenes = Math.ceil(totalSec / sLen);
   157→    const tPhases = Math.ceil(tScenes / SCENES_PER_PHASE);
   158→
   159→    const project: Project = {
   160→      id: '',
   161→      title: values.title,
   162→      topic: values.topic,
   163→      description: values.description ?? '',
   164→      thumbnailPrompt: '',
   165→      tags: [],
   166→      settings: {
   167→        duration: values.duration as VideoDuration,
   168→        theme: values.theme as VideoTheme,
   169→        language: values.language as VideoLanguage,
   170→        writingStyle: values.writingStyle as WritingStyle,
   171→        targetAudience: values.targetAudience as TargetAudience,
   172→        sceneLength: sLen,
   173→        totalScenes: tScenes,
   174→        scenesPerPhase: SCENES_PER_PHASE,
   175→      },
   176→      status: 'generating',
   177→      scoreHistory: [],
   178→      createdAt: now,
   179→      updatedAt: now,
   180→    };
   181→
   182→    // Save project to MongoDB and get the real ID back
   183→    const mongoId = await addProject(project);
   184→    setActiveProjectId(mongoId);
   185→    projectIdRef.current = mongoId;
   186→
   187→    // Switch to generating phase
   188→    setPhase('generating');
   189→    setCurrentStage(0);
   190→    setElapsed(0);
   191→    stageTimerRef.current = setInterval(advanceStage, STAGE_INTERVAL_MS);
   192→    elapsedTimerRef.current = setInterval(() => setElapsed((t) => t + 1), 1000);
   193→
   194→    try {
   195→      const realProject = useAppStore.getState().projects.find((p) => p.id === mongoId);
   196→      const projectToGenerate = realProject || { ...project, id: mongoId };
   197→
   198→      // Get channel niche from auth store
   199→      const channelNiche = useAuthStore.getState().user?.channelNiche as ChannelNiche | undefined;
   200→
   201→      // Generate Phase 1 only
   202→      const phase1ScenesCount = Math.min(SCENES_PER_PHASE, tScenes);
   203→      const result = await generatePhase(projectToGenerate, {
   204→        phaseNumber: 1,
   205→        totalPhases: tPhases,
   206→        sceneStart: 1,
   207→        sceneEnd: phase1ScenesCount,
   208→        previousPhaseTitles: [],
   209→      }, channelNiche || null);
   210→
   211→      const scenesWithCorrectId = result.scenes.map((s) => ({ ...s, projectId: mongoId }));
   212→
   213→      // Save AI-generated metadata
   214→      if (result.metadata) {
   215→        await updateProject(mongoId, {
   216→          description: result.metadata.videoDescription || projectToGenerate.description || '',
   217→          thumbnailPrompt: result.metadata.thumbnailPrompt || '',
   218→          tags: result.metadata.tags || [],
   219→        });
   220→      }
   221→
   222→      // Advance to "saving" stage
   223→      setCurrentStage(GENERATION_STAGES.length - 1);
   224→
   225→      // Save scenes to DB
   226→      await addScenes(scenesWithCorrectId);
   227→      await updateProject(mongoId, { status: 'completed' });
   228→
   229→      // Brief pause
   230→      await new Promise((r) => setTimeout(r, 800));
   231→
   232→      // Clean up timers
   233→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   234→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   235→
   236→      // Pre-load scenes into store, then redirect to editor
   237→      await loadScenes(mongoId);
   238→      router.push(`/project/${mongoId}`);
   239→    } catch (error) {
   240→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   241→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   242→
   243→      const message = error instanceof Error ? error.message : 'An unexpected error occurred';
   244→      setErrorMessage(message);
   245→      await updateProject(mongoId, { status: 'error' });
   246→      setPhase('error');
   247→      isSubmittingRef.current = false;
   248→    }
   249→  }, [addProject, setActiveProjectId, advanceStage, addScenes, updateProject, loadScenes, router]);
   250→
   251→  const onRetry = useCallback(() => {
   252→    if (formValuesRef.current) {
   253→      setPhase('form');
   254→      setCurrentStage(-1);
   255→      setElapsed(0);
   256→      setErrorMessage('');
   257→    }
   258→  }, []);
   259→
   260→  const onSubmit = useCallback((values: FormValues) => {
   261→    formValuesRef.current = values;
   262→    startGeneration(values);
   263→  }, [startGeneration]);
   264→
   265→  const onBackToProjects = useCallback(() => {
   266→    router.push('/projects');
   267→  }, [router]);
   268→
   269→  const descriptionValue = form.watch('description') ?? '';
   270→
   271→  const formatTime = (s: number) => {
   272→    const m = Math.floor(s / 60);
   273→    const sec = s % 60;
   274→    return `${m}:${sec.toString().padStart(2, '0')}`;
   275→  };
   276→
   277→  // ─── ERROR VIEW ──────────────────────────────────────────────────────────
   278→  if (phase === 'error') {
   279→    return (
   280→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   281→        <motion.div
   282→          initial={{ opacity: 0, scale: 0.95 }}
   283→          animate={{ opacity: 1, scale: 1 }}
   284→          transition={{ duration: 0.4 }}
   285→          className="w-full max-w-md text-center space-y-6"
   286→        >
   287→          <motion.div
   288→            initial={{ scale: 0 }}
   289→            animate={{ scale: 1 }}
   290→            transition={{ type: 'spring', stiffness: 300, damping: 20, delay: 0.1 }}
   291→            className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-destructive/10"
   292→          >
   293→            <AlertCircle className="h-8 w-8 text-destructive" />
   294→          </motion.div>
   295→
   296→          <div className="space-y-2">
   297→            <h2 className="text-2xl font-bold tracking-tight">Generation Failed</h2>
   298→            <p className="text-sm text-muted-foreground">
   299→              {errorMessage || 'Something went wrong. Please try again.'}
   300→            </p>
   301→          </div>
   302→
   303→          <div className="flex flex-col gap-3">
   304→            <Button
   305→              size="lg"
   306→              onClick={onRetry}
   307→              className="w-full text-base font-semibold gap-2"
   308→            >
   309→              <RefreshCw className="h-4 w-4" />
   310→              Try Again
   311→            </Button>
   312→            <Button
   313→              variant="outline"
   314→              onClick={onBackToDashboard}
   315→              className="w-full"
   316→            >
   317→              Back to Dashboard
   318→            </Button>
   319→          </div>
   320→        </motion.div>
   321→      </div>
   322→    );
   323→  }
   324→
   325→  // ─── GENERATING VIEW ─────────────────────────────────────────────────────
   326→  if (phase === 'generating') {
   327→    return (
   328→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   329→        <motion.div
   330→          initial={{ opacity: 0, y: 20 }}
   331→          animate={{ opacity: 1, y: 0 }}
   332→          transition={{ duration: 0.4, ease: 'easeOut' }}
   333→          className="w-full max-w-lg space-y-8"
   334→        >
   335→          {/* Header */}
   336→          <div className="text-center space-y-2">
   337→            <motion.div
   338→              animate={{ scale: [1, 1.08, 1] }}
   339→              transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   340→              className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-2xl bg-primary/10"
   341→            >
   342→              <Sparkles className="h-7 w-7 text-primary" />
   343→            </motion.div>
   344→            <h2 className="text-2xl font-bold tracking-tight">Generating Phase 1</h2>
   345→            <p className="text-sm text-muted-foreground">
   346→              {formValuesRef.current?.title && `Crafting first ${SCENES_PER_PHASE} scenes for "${formValuesRef.current.title}"`}
   347→            </p>
   348→          </div>
   349→
   350→          {/* Elapsed time */}
   351→          <div className="text-center">
   352→            <span className="text-xs text-muted-foreground font-mono">
   353→              Elapsed {formatTime(elapsed)}
   354→            </span>
   355→          </div>
   356→
   357→          {/* Progress bar */}
   358→          <div className="relative h-2 w-full overflow-hidden rounded-full bg-primary/20">
   359→            <div className="absolute inset-0 animate-indeterminate-progress rounded-full bg-primary" />
   360→          </div>
   361→
   362→          {/* Stages */}
   363→          <div className="space-y-3">
   364→            {GENERATION_STAGES.map((stage, index) => {
   365→              const isCompleted = index < currentStage;
   366→              const isActive = index === currentStage;
   367→
   368→              return (
   369→                <motion.div
   370→                  key={index}
   371→                  initial={{ opacity: 0, x: -10 }}
   372→                  animate={{ opacity: 1, x: 0 }}
   373→                  transition={{ delay: index * 0.05 }}
   374→                  className="flex items-center gap-3 text-sm rounded-lg px-3 py-2 transition-colors"
   375→                  style={{
   376→                    backgroundColor: isActive ? 'hsl(var(--primary) / 0.06)' : 'transparent',
   377→                  }}
   378→                >
   379→                  <div className="flex h-6 w-6 shrink-0 items-center justify-center">
   380→                    {isCompleted ? (
   381→                      <motion.div
   382→                        initial={{ scale: 0 }}
   383→                        animate={{ scale: 1 }}
   384→                        transition={{ type: 'spring', stiffness: 500, damping: 30 }}
   385→                      >
   386→                        <Check className="h-4 w-4 text-emerald-500" />
   387→                      </motion.div>
   388→                    ) : isActive ? (
   389→                      <motion.div
   390→                        className="h-2.5 w-2.5 rounded-full bg-primary"
   391→                        animate={{ scale: [1, 1.5, 1], opacity: [0.5, 1, 0.5] }}
   392→                        transition={{ repeat: Infinity, duration: 1.2 }}
   393→                      />
   394→                    ) : (
   395→                      <div className="h-2.5 w-2.5 rounded-full bg-muted-foreground/30" />
   396→                    )}
   397→                  </div>
   398→                  <span
   399→                    className={
   400→                      isCompleted
   401→                        ? 'text-emerald-600 dark:text-emerald-400'
   402→                        : isActive
   403→                          ? 'font-medium text-foreground'
   404→                          : 'text-muted-foreground'
   405→                    }
   406→                  >
   407→                    {isCompleted ? stage.completed : stage.active}
   408→                  </span>
   409→                </motion.div>
   410→              );
   411→            })}
   412→          </div>
   413→
   414→          {/* Tip */}
   415→          <p className="text-center text-xs text-muted-foreground/60">
   416→            Generating Phase 1 of {totalPhases}. Remaining phases can be generated from the editor.
   417→          </p>
   418→        </motion.div>
   419→
   420→        {/* Indeterminate progress bar animation */}
   421→        <style jsx global>{`
   422→          @keyframes indeterminate-progress {
   423→            0% { transform: translateX(-100%); width: 40%; }
   424→            50% { transform: translateX(100%); width: 60%; }
   425→            100% { transform: translateX(200%); width: 40%; }
   426→          }
   427→          .animate-indeterminate-progress {
   428→            animation: indeterminate-progress 1.8s ease-in-out infinite;
   429→          }
   430→        `}</style>
   431→      </div>
   432→    );
   433→  }
   434→
   435→  // ─── FORM VIEW (default) ─────────────────────────────────────────────────
   436→  return (
   437→    <motion.div
   438→      initial={{ opacity: 0, y: 20 }}
   439→      animate={{ opacity: 1, y: 0 }}
   440→      transition={{ duration: 0.4, ease: 'easeOut' }}
   441→      className="flex min-h-full items-start justify-center px-4 py-8 md:items-center md:py-12"
   442→    >
   443→      <Card className="w-full max-w-2xl shadow-lg">
   444→        <CardHeader className="pb-6">
   445→          <div className="flex items-center gap-3">
   446→            <Button
   447→              variant="ghost"
   448→              size="icon"
   449→              onClick={() => router.push('/projects')}
   450→              className="shrink-0"
   451→              aria-label="Back to projects"
   452→            >
   453→              <ArrowLeft className="h-5 w-5" />
   454→            </Button>
   455→            <div className="flex items-center gap-3">
   456→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10">
   457→                <Film className="h-5 w-5 text-primary" />
   458→              </div>
   459→              <div>
   460→                <CardTitle className="text-xl font-semibold tracking-tight">
   461→                  Create New Project
   462→                </CardTitle>
   463→                <p className="mt-0.5 text-sm text-muted-foreground">
   464→                  Transform your video idea into a production script
   465→                </p>
   466→              </div>
   467→            </div>
   468→          </div>
   469→        </CardHeader>
   470→
   471→        <CardContent className="px-6 pb-8">
   472→          <Form {...form}>
   473→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-8">
   474→              {/* ── Basic Information ────────────────────────────────── */}
   475→              <section>
   476→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   477→                  Basic Information
   478→                </h2>
   479→
   480→                <div className="rounded-lg border bg-muted/30 p-4 space-y-4">
   481→                  <FormField
   482→                    control={form.control}
   483→                    name="title"
   484→                    render={({ field }) => (
   485→                      <FormItem>
   486→                        <FormLabel>Project Title</FormLabel>
   487→                        <FormControl>
   488→                          <Input
   489→                            placeholder="e.g. The Future of Renewable Energy"
   490→                            {...field}
   491→                          />
   492→                        </FormControl>
   493→                        <FormMessage />
   494→                      </FormItem>
   495→                    )}
   496→                  />
   497→
   498→                  <FormField
   499→                    control={form.control}
   500→                    name="topic"
   501→                    render={({ field }) => (
   502→                      <FormItem>
   503→                        <FormLabel>Video Topic</FormLabel>
   504→                        <FormControl>
   505→                          <Input
   506→                            placeholder="e.g. How solar panels are changing the world"
   507→                            {...field}
   508→                          />
   509→                        </FormControl>
   510→                        <FormMessage />
   511→                      </FormItem>
   512→                    )}
   513→                  />
   514→
   515→                  <FormField
   516→                    control={form.control}
   517→                    name="description"
   518→                    render={({ field }) => (
   519→                      <FormItem>
   520→                        <div className="flex items-center justify-between">
   521→                          <FormLabel>Description</FormLabel>
   522→                          <span className="text-xs text-muted-foreground">
   523→                            {descriptionValue.length}/1000
   524→                          </span>
   525→                        </div>
   526→                        <FormControl>
   527→                          <Textarea
   528→                            placeholder="Add any extra details, context, or specific requirements for the video..."
   529→                            rows={3}
   530→                            maxLength={1000}
   531→                            {...field}
   532→                            value={field.value ?? ''}
   533→                          />
   534→                        </FormControl>
   535→                        <FormMessage />
   536→                      </FormItem>
   537→                    )}
   538→                  />
   539→                </div>
   540→              </section>
   541→
   542→              <Separator />
   543→
   544→              {/* ── Video Structure ──────────────────────────────────── */}
   545→              <section>
   546→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   547→                  Video Structure
   548→                </h2>
   549→
   550→                <div className="rounded-lg border bg-muted/30 p-4 space-y-4">
   551→                  <div className="grid gap-4 sm:grid-cols-2">
   552→                    <FormField
   553→                      control={form.control}
   554→                      name="duration"
   555→                      render={({ field }) => (
   556→                        <FormItem>
   557→                          <FormLabel>Video Duration</FormLabel>
   558→                          <Select
   559→                            onValueChange={field.onChange}
   560→                            defaultValue={field.value}
   561→                          >
   562→                            <FormControl>
   563→                              <SelectTrigger>
   564→                                <SelectValue placeholder="Select duration" />
   565→                              </SelectTrigger>
   566→                            </FormControl>
   567→                            <SelectContent>
   568→                              {(Object.entries(DURATION_LABELS) as [VideoDuration, string][]).map(
   569→                                ([value, label]) => (
   570→                                  <SelectItem key={value} value={value}>
   571→                                    {label}
   572→                                  </SelectItem>
   573→                                )
   574→                              )}
   575→                            </SelectContent>
