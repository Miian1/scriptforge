   365→   365→
   366→   366→  const project = projects.find((p) => p.id === activeProjectId) ?? null;
   367→   367→
   368→   368→  // Computed phase info from project settings
   369→   369→  const totalScenes = project?.settings?.totalScenes || 60;
   370→   370→  const scenesPerPhase = project?.settings?.scenesPerPhase || SCENES_PER_PHASE;
   371→   371→  const totalPhases = Math.ceil(totalScenes / scenesPerPhase);
   372→   372→  const currentPhaseNumber = Math.min(
   373→   373→    Math.ceil(scenes.length / scenesPerPhase) + (scenes.length % scenesPerPhase === 0 ? 0 : 1),
   374→   374→    totalPhases
   375→   375→  );
   376→   376→  // Actually: how many full phases are done?
   377→   377→  const completedPhases = Math.floor(scenes.length / scenesPerPhase);
   378→   378→  const isOnPartialPhase = scenes.length % scenesPerPhase !== 0;
   379→   379→  const activePhaseNum = isOnPartialPhase ? completedPhases + 1 : completedPhases;
   380→   380→  const allPhasesDone = scenes.length >= totalScenes;
   381→   381→  const hasNextPhase = !allPhasesDone && scenes.length > 0;
   382→   382→  const nextPhaseNumber = hasNextPhase
   383→   383→    ? (scenes.length % scenesPerPhase === 0
   384→   384→        ? scenes.length / scenesPerPhase + 1
   385→   385→        : Math.floor(scenes.length / scenesPerPhase) + 1)
   386→   386→    : 0;
   387→   387→
   388→   388→  // Group scenes into phases for display
   389→   389→  const phases = useMemo(() => {
   390→   390→    const grouped: { phase: number; scenes: Scene[] }[] = [];
   391→   391→    const sorted = [...scenes].sort((a, b) => a.sceneNumber - b.sceneNumber);
   392→   392→    for (let i = 0; i < sorted.length; i++) {
   393→   393→      const phaseNum = Math.floor(sorted[i].sceneNumber / scenesPerPhase) + 1;
   394→   394→      const lastGroup = grouped[grouped.length - 1];
   395→   395→      if (lastGroup && lastGroup.phase === phaseNum) {
   396→   396→        lastGroup.scenes.push(sorted[i]);
   397→   397→      } else {
   398→   398→        grouped.push({ phase: phaseNum, scenes: [sorted[i]] });
   399→   399→      }
   400→   400→    }
   401→   401→    return grouped;
   402→   402→  }, [scenes, scenesPerPhase]);
   403→   403→
   404→   404→  // Load scenes on mount
   405→   405→  useEffect(() => {
   406→   406→    if (activeProjectId) {
   407→   407→      loadScenes(activeProjectId);

   410→   410→

   413→   413→    useSensor(PointerSensor, {

   419→   419→  // Handle drag end
   420→   420→  const handleDragEnd = useCallback(
   421→   421→    (event: DragEndEvent) => {
   422→   422→      const { active, over } = event;
   423→   423→      if (!over || active.id === over.id) return;
   424→   424→
   425→   425→      const oldIndex = scenes.findIndex((s) => s.id === active.id);
   426→   426→      const newIndex = scenes.findIndex((s) => s.id === over.id);
   427→   427→      if (oldIndex === -1 || newIndex === -1) return;
   428→   428→
   429→   429→      const reordered = arrayMove(scenes, oldIndex, newIndex);
   430→   430→      reorderScenes(reordered);
   431→   431→    },
   432→   432→    [scenes, reorderScenes]
   433→   433→  );
   434→   434→
   435→   435→  // Add new scene
   436→   436→  const handleAddScene = useCallback(async () => {
   437→   437→    if (!project) return;
   438→   438→    const now = Date.now();
   439→   439→    const sceneLen = project.settings.sceneLength || 8;
   440→   440→    const newScene: Scene = {
   441→   441→      id: uuidv4(),
   442→   442→      projectId: project.id,
   443→   443→      sceneNumber: scenes.length + 1,
   444→   444→      title: 'New Scene',
   445→   445→      estimatedDuration: sceneLen,
   446→   446→      goal: '',
   447→   447→      narration: '',
   448→   448→      imagePrompt: '',
   449→   449→      animationPrompt: '',
   450→   450→      notes: {
   451→   451→        emotion: '',
   452→   452→        visualFocus: '',
   453→   453→        transitionSuggestion: '',
   454→   454→        importantDetails: '',
   455→   455→      },
   456→   456→      createdAt: now,
   457→   457→      updatedAt: now,
   458→   458→    };
   459→   459→    await addScene(newScene);
   460→   460→
   461→   461→    requestAnimationFrame(() => {
   462→   462→      listEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   463→   463→    });
   464→   464→
   465→   465→    toast.success('Scene added');

   543→   543→          title: s.title,
   544→   544→          estimatedDuration: s.estimatedDuration,

   567→   567→  // Retry generation (regenerate Phase 1)

   578→   578→
   579→   579→      const { scenes: newScenes, metadata } = await generateScript(project, channelNiche || null);
   580→   580→      for (const scene of newScenes) {
   581→   581→        await addScene(scene);
   582→   582→      }
   583→   583→      if (metadata) {
   584→   584→        await updateProject(project.id, {
   585→   585→          description: metadata.videoDescription || project.description || '',
   586→   586→          thumbnailPrompt: metadata.thumbnailPrompt || '',
   587→   587→          tags: metadata.tags || [],
   588→   588→        });
   589→   589→      }
   590→   590→      await updateProject(project.id, { status: 'completed' });
   591→   591→      toast.success('Script generated successfully!');

   593→   593→    } catch (err) {
   594→   594→      await updateProject(project.id, { status: 'error' });
   595→   595→      toast.error((err as Error).message || 'Generation failed');
   596→   596→    } finally {
   597→   597→      setGeneratingProjectId(null);
   598→   598→      setIsGeneratingPhase(false);
   599→   599→      setGeneratingPhaseLabel('');
   600→   600→    }
   601→   601→  }, [project, addScene, updateProject, loadScenes, setGeneratingProjectId]);
   602→   602→
   603→   603→  // Back to dashboard
   604→   604→  const handleBack = useCallback(() => {
   605→   605→    setActiveProjectId(null);
   606→   606→    router.push('/dashboard');
   607→   607→  }, [setActiveProjectId, router]);
   608→   608→
   609→   609→  // ---- Render ----
   610→   610→  if (!project) {
   611→   611→    return (
   612→   612→      <div className="flex items-center justify-center min-h-[60vh]">

   630→   630→                <ArrowLeft className="size-4" />
   631→   631→              </Button>

   637→   637→            <h1 className="text-lg sm:text-xl font-bold truncate">{project.title}</h1>

   666→   666→                      <span className="hidden sm:inline">Add Scene</span>
   667→   667→                    </Button>

   669→   669→                  <TooltipContent>Add a new scene</TooltipContent>

   677→   677→                    onClick={handleExport}
   678→   678→                    disabled={scenes.length === 0}

   733→   733→
   734→   734→                {/* Scenes grouped by phase */}
   735→   735→                <div className="space-y-6">

   749→   749→                        </span>
   750→   750→                        <div className="flex-1" />
   751→   751→                        <span className="text-xs text-muted-foreground">

   766→   766→                          strategy={verticalListSortingStrategy}
   767→   767→                        >
   768→   768→                          <div className="space-y-4">
   769→   769→                            <AnimatePresence mode="popLayout">
   770→   770→                              {phaseScenes.map((scene) => (
   771→   771→                                <SceneCard
   772→   772→                                  key={scene.id}
   773→   773→                                  scene={scene}
   774→   774→                                  project={project}

   785→   785→

   790→   790→                      initial={{ opacity: 0, y: 12 }}
   791→   791→                      animate={{ opacity: 1, y: 0 }}
   792→   792→                      className="flex flex-col items-center gap-3"
   793→   793→                    >
   794→   794→                      <div className="rounded-xl border-2 border-dashed border-primary/30 bg-primary/[0.03] p-6 text-center w-full max-w-md space-y-4">
   795→   795→                        <div className="flex items-center justify-center gap-2">
   796→   796→                          <Zap className="size-5 text-primary" />
   797→   797→                          <span className="text-sm font-semibold">
   798→   798→                            Generate Next Phase
   799→   799→                          </span>

   812→   812→                        >

   824→   824→                        </Button>

   828→   828→
   829→   829→                  {allPhasesDone && (
   830→   830→                    <motion.div
   831→   831→                      initial={{ opacity: 0, y: 12 }}
   832→   832→                      animate={{ opacity: 1, y: 0 }}
   833→   833→                      className="flex flex-col items-center gap-3"
   834→   834→                    >
