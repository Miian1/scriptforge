     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react';
     4→import { motion } from 'framer-motion';
     5→import { Film, ArrowLeft, Sparkles, Check, AlertCircle, RefreshCw } from 'lucide-react';
     6→import { toast } from 'sonner';
     7→import { z } from 'zod';
     8→import { useForm } from 'react-hook-form';
     9→import { zodResolver } from '@hookform/resolvers/zod';
    10→
    11→import { useAppStore } from '@/lib/store';
    12→import { generateScript } from '@/lib/gemini';
    13→import type { Project, VideoTheme, WritingStyle, TargetAudience, VideoLanguage, VideoDuration } from '@/lib/types';
    14→import {
    15→  DURATION_LABELS,
    16→  THEME_LABELS,
    17→  STYLE_LABELS,
    18→  AUDIENCE_LABELS,
    19→  LANGUAGE_LABELS,
    20→} from '@/lib/types';
    21→
    22→import { Button } from '@/components/ui/button';
    23→import { Input } from '@/components/ui/input';
    24→import { Textarea } from '@/components/ui/textarea';
    25→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    26→import { Separator } from '@/components/ui/separator';
    27→import {
    28→  Form,
    29→  FormControl,
    30→  FormField,
    31→  FormItem,
    32→  FormLabel,
    33→  FormMessage,
    34→} from '@/components/ui/form';
    35→import {
    36→  Select,
    37→  SelectContent,
    38→  SelectItem,
    39→  SelectTrigger,
    40→  SelectValue,
    41→} from '@/components/ui/select';
    42→
    43→// ─── Schema ───────────────────────────────────────────────────────────────────
    44→
    45→const formSchema = z.object({
    46→  title: z.string().min(1, 'Title is required').max(100),
    47→  topic: z.string().min(1, 'Topic is required').max(200),
    48→  description: z.string().max(1000).optional().default(''),
    49→  duration: z.enum(['short', 'medium', 'long']),
    50→  theme: z.enum(['realistic', 'anime', 'cinematic', 'cartoon', '3d-render', 'watercolor', 'pixel-art']),
    51→  language: z.enum(['english', 'spanish', 'french', 'german', 'portuguese', 'japanese', 'korean', 'chinese', 'hindi', 'arabic']),
    52→  writingStyle: z.enum(['conversational', 'professional', 'dramatic', 'educational', 'storytelling', 'humorous']),
    53→  targetAudience: z.enum(['beginners', 'general', 'intermediate', 'experts', 'kids', 'teens']),
    54→});
    55→
    56→type FormValues = z.infer<typeof formSchema>;
    57→
    58→// ─── Generation Stages ────────────────────────────────────────────────────────
    59→
    60→const GENERATION_STAGES = [
    61→  { active: 'Researching topic...', completed: 'Research complete' },
    62→  { active: 'Building story structure...', completed: 'Story structured' },
    63→  { active: 'Generating scenes with AI...', completed: 'Scenes generated' },
    64→  { active: 'Finalizing...', completed: 'Finalized' },
    65→] as const;
    66→
    67→const STAGE_INTERVAL_MS = 5000;
    68→
    69→// ─── Component ────────────────────────────────────────────────────────────────
    70→
    71→export default function CreateProject() {
    72→  const {
    73→    setCurrentView,
    74→    addProject,
    75→    setActiveProjectId,
    76→    addScenes,
    77→    updateProject,
    78→    loadScenes,
    79→  } = useAppStore();
    80→
    81→  // Phase: 'form' | 'generating' | 'error'
    82→  const [phase, setPhase] = useState<'form' | 'generating' | 'error'>('form');
    83→  const [currentStage, setCurrentStage] = useState(-1);
    84→  const [errorMessage, setErrorMessage] = useState('');
    85→  const [elapsed, setElapsed] = useState(0);
    86→  const stageTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    87→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    88→  const projectIdRef = useRef<string | null>(null);
    89→  const formValuesRef = useRef<FormValues | null>(null);
    90→
    91→  const form = useForm<FormValues>({
    92→    resolver: zodResolver(formSchema),
    93→    defaultValues: {
    94→      title: '',
    95→      topic: '',
    96→      description: '',
    97→      duration: 'medium',
    98→      theme: 'cinematic',
    99→      language: 'english',
   100→      writingStyle: 'conversational',
   101→      targetAudience: 'general',
   102→    },
   103→  });
   104→
   105→  // Cleanup timers on unmount
   106→  useEffect(() => {
   107→    return () => {
   108→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   109→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   110→    };
   111→  }, []);
   112→
   113→  const advanceStage = useCallback(() => {
   114→    setCurrentStage((prev) => {
   115→      if (prev >= GENERATION_STAGES.length - 1) return prev;
   116→      return prev + 1;
   117→    });
   118→  }, []);
   119→
   120→  const startGeneration = useCallback(async (values: FormValues) => {
   121→    const now = Date.now();
   122→
   123→    const project: Project = {
   124→      id: '',
   125→      title: values.title,
   126→      topic: values.topic,
   127→      description: values.description ?? '',
   128→      settings: {
   129→        duration: values.duration as VideoDuration,
   130→        theme: values.theme as VideoTheme,
   131→        language: values.language as VideoLanguage,
   132→        writingStyle: values.writingStyle as WritingStyle,
   133→        targetAudience: values.targetAudience as TargetAudience,
   134→      },
   135→      status: 'generating',
   136→      createdAt: now,
   137→      updatedAt: now,
   138→    };
   139→
   140→    // Save project to MongoDB and get the real ID back
   141→    const mongoId = await addProject(project);
   142→    setActiveProjectId(mongoId);
   143→    projectIdRef.current = mongoId;
   144→
   145→    // Switch to generating phase (stays on this page)
   146→    setPhase('generating');
   147→    setCurrentStage(0);
   148→    setElapsed(0);
   149→    stageTimerRef.current = setInterval(advanceStage, STAGE_INTERVAL_MS);
   150→    elapsedTimerRef.current = setInterval(() => setElapsed((t) => t + 1), 1000);
   151→
   152→    try {
   153→      // Use the real MongoDB project from store for generation
   154→      const realProject = useAppStore.getState().projects.find((p) => p.id === mongoId);
   155→      const projectToGenerate = realProject || { ...project, id: mongoId };
   156→      const scenes = await generateScript(projectToGenerate);
   157→      const scenesWithCorrectId = scenes.map((s) => ({ ...s, projectId: mongoId }));
   158→
   159→      // Advance to "saving" stage
   160→      setCurrentStage(GENERATION_STAGES.length - 1);
   161→
   162→      // Save scenes to DB
   163→      await addScenes(scenesWithCorrectId);
   164→      await updateProject(mongoId, { status: 'completed' });
   165→
   166→      // Brief pause to show "Finalized" stage
   167→      await new Promise((r) => setTimeout(r, 800));
   168→
   169→      // Clean up timers
   170→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   171→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   172→
   173→      // Pre-load scenes into store, then auto-redirect to editor
   174→      await loadScenes(mongoId);
   175→      setCurrentView('editor');
   176→    } catch (error) {
   177→      if (stageTimerRef.current) clearInterval(stageTimerRef.current);
   178→      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
   179→
   180→      const message = error instanceof Error ? error.message : 'An unexpected error occurred';
   181→      setErrorMessage(message);
   182→      await updateProject(mongoId, { status: 'error' });
   183→      setPhase('error');
   184→    }
   185→  }, [addProject, setActiveProjectId, advanceStage, addScenes, updateProject]);
   186→
   187→  const onRetry = useCallback(() => {
   188→    if (formValuesRef.current) {
   189→      setPhase('form');
   190→      setCurrentStage(-1);
   191→      setElapsed(0);
   192→      setErrorMessage('');
   193→    }
   194→  }, []);
   195→
   196→  const onSubmit = useCallback((values: FormValues) => {
   197→    formValuesRef.current = values;
   198→    startGeneration(values);
   199→  }, [startGeneration]);
   200→
   201→  const onBackToDashboard = useCallback(() => {
   202→    setCurrentView('dashboard');
   203→  }, [setCurrentView]);
   204→
   205→  const descriptionValue = form.watch('description') ?? '';
   206→
   207→  const formatTime = (s: number) => {
   208→    const m = Math.floor(s / 60);
   209→    const sec = s % 60;
   210→    return `${m}:${sec.toString().padStart(2, '0')}`;
   211→  };
   212→
   213→  // ─── ERROR VIEW ──────────────────────────────────────────────────────────
   214→  if (phase === 'error') {
   215→    return (
   216→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   217→        <motion.div
   218→          initial={{ opacity: 0, scale: 0.95 }}
   219→          animate={{ opacity: 1, scale: 1 }}
   220→          transition={{ duration: 0.4 }}
   221→          className="w-full max-w-md text-center space-y-6"
   222→        >
   223→          <motion.div
   224→            initial={{ scale: 0 }}
   225→            animate={{ scale: 1 }}
   226→            transition={{ type: 'spring', stiffness: 300, damping: 20, delay: 0.1 }}
   227→            className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-destructive/10"
   228→          >
   229→            <AlertCircle className="h-8 w-8 text-destructive" />
   230→          </motion.div>
   231→
   232→          <div className="space-y-2">
   233→            <h2 className="text-2xl font-bold tracking-tight">Generation Failed</h2>
   234→            <p className="text-sm text-muted-foreground">
   235→              {errorMessage || 'Something went wrong. Please try again.'}
   236→            </p>
   237→          </div>
   238→
   239→          <div className="flex flex-col gap-3">
   240→            <Button
   241→              size="lg"
   242→              onClick={onRetry}
   243→              className="w-full text-base font-semibold gap-2"
   244→            >
   245→              <RefreshCw className="h-4 w-4" />
   246→              Try Again
   247→            </Button>
   248→            <Button
   249→              variant="outline"
   250→              onClick={onBackToDashboard}
   251→              className="w-full"
   252→            >
   253→              Back to Dashboard
   254→            </Button>
   255→          </div>
   256→        </motion.div>
   257→      </div>
   258→    );
   259→  }
   260→
   261→  // ─── GENERATING VIEW ─────────────────────────────────────────────────────
   262→  if (phase === 'generating') {
   263→    return (
   264→      <div className="flex min-h-full items-center justify-center px-4 py-8 md:py-12">
   265→        <motion.div
   266→          initial={{ opacity: 0, y: 20 }}
   267→          animate={{ opacity: 1, y: 0 }}
   268→          transition={{ duration: 0.4, ease: 'easeOut' }}
   269→          className="w-full max-w-lg space-y-8"
   270→        >
   271→          {/* Header */}
   272→          <div className="text-center space-y-2">
   273→            <motion.div
   274→              animate={{ scale: [1, 1.08, 1] }}
   275→              transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   276→              className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-2xl bg-primary/10"
   277→            >
   278→              <Sparkles className="h-7 w-7 text-primary" />
   279→            </motion.div>
   280→            <h2 className="text-2xl font-bold tracking-tight">Generating Your Script</h2>
   281→            <p className="text-sm text-muted-foreground">
   282→              {formValuesRef.current?.title && `Crafting scenes for "${formValuesRef.current.title}"`}
   283→            </p>
   284→          </div>
   285→
   286→          {/* Elapsed time */}
   287→          <div className="text-center">
   288→            <span className="text-xs text-muted-foreground font-mono">
   289→              Elapsed {formatTime(elapsed)}
   290→            </span>
   291→          </div>
   292→
   293→          {/* Progress bar */}
   294→          <div className="relative h-2 w-full overflow-hidden rounded-full bg-primary/20">
   295→            <div className="absolute inset-0 animate-indeterminate-progress rounded-full bg-primary" />
   296→          </div>
   297→
   298→          {/* Stages */}
   299→          <div className="space-y-3">
   300→            {GENERATION_STAGES.map((stage, index) => {
   301→              const isCompleted = index < currentStage;
   302→              const isActive = index === currentStage;
   303→
   304→              return (
   305→                <motion.div
   306→                  key={index}
   307→                  initial={{ opacity: 0, x: -10 }}
   308→                  animate={{ opacity: 1, x: 0 }}
   309→                  transition={{ delay: index * 0.05 }}
   310→                  className="flex items-center gap-3 text-sm rounded-lg px-3 py-2 transition-colors"
   311→                  style={{
   312→                    backgroundColor: isActive ? 'hsl(var(--primary) / 0.06)' : 'transparent',
   313→                  }}
   314→                >
   315→                  <div className="flex h-6 w-6 shrink-0 items-center justify-center">
   316→                    {isCompleted ? (
   317→                      <motion.div
   318→                        initial={{ scale: 0 }}
   319→                        animate={{ scale: 1 }}
   320→                        transition={{ type: 'spring', stiffness: 500, damping: 30 }}
   321→                      >
   322→                        <Check className="h-4 w-4 text-emerald-500" />
   323→                      </motion.div>
   324→                    ) : isActive ? (
   325→                      <motion.div
   326→                        className="h-2.5 w-2.5 rounded-full bg-primary"
   327→                        animate={{ scale: [1, 1.5, 1], opacity: [0.5, 1, 0.5] }}
   328→                        transition={{ repeat: Infinity, duration: 1.2 }}
   329→                      />
   330→                    ) : (
   331→                      <div className="h-2.5 w-2.5 rounded-full bg-muted-foreground/30" />
   332→                    )}
   333→                  </div>
   334→                  <span
   335→                    className={
   336→                      isCompleted
   337→                        ? 'text-emerald-600 dark:text-emerald-400'
   338→                        : isActive
   339→                          ? 'font-medium text-foreground'
   340→                          : 'text-muted-foreground'
   341→                    }
   342→                  >
   343→                    {isCompleted ? stage.completed : stage.active}
   344→                  </span>
   345→                </motion.div>
   346→              );
   347→            })}
   348→          </div>
   349→
   350→          {/* Tip */}
   351→          <p className="text-center text-xs text-muted-foreground/60">
   352→            AI is generating narration, image prompts, and animation prompts for each scene. This may take a moment for complex topics.
   353→          </p>
   354→        </motion.div>
   355→
   356→        {/* Indeterminate progress bar animation */}
   357→        <style jsx global>{`
   358→          @keyframes indeterminate-progress {
   359→            0% { transform: translateX(-100%); width: 40%; }
   360→            50% { transform: translateX(100%); width: 60%; }
   361→            100% { transform: translateX(200%); width: 40%; }
   362→          }
   363→          .animate-indeterminate-progress {
   364→            animation: indeterminate-progress 1.8s ease-in-out infinite;
   365→          }
   366→        `}</style>
   367→      </div>
   368→    );
   369→  }
   370→
   371→  // ─── FORM VIEW (default) ─────────────────────────────────────────────────
   372→  return (
   373→    <motion.div
   374→      initial={{ opacity: 0, y: 20 }}
   375→      animate={{ opacity: 1, y: 0 }}
   376→      transition={{ duration: 0.4, ease: 'easeOut' }}
   377→      className="flex min-h-full items-start justify-center px-4 py-8 md:items-center md:py-12"
   378→    >
   379→      <Card className="w-full max-w-2xl shadow-lg">
   380→        <CardHeader className="pb-6">
   381→          <div className="flex items-center gap-3">
   382→            <Button
   383→              variant="ghost"
   384→              size="icon"
   385→              onClick={() => setCurrentView('dashboard')}
   386→              className="shrink-0"
   387→              aria-label="Back to dashboard"
   388→            >
   389→              <ArrowLeft className="h-5 w-5" />
   390→            </Button>
   391→            <div className="flex items-center gap-3">
   392→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10">
   393→                <Film className="h-5 w-5 text-primary" />
   394→              </div>
   395→              <div>
   396→                <CardTitle className="text-xl font-semibold tracking-tight">
   397→                  Create New Project
   398→                </CardTitle>
   399→                <p className="mt-0.5 text-sm text-muted-foreground">
   400→                  Transform your video idea into a production script
   401→                </p>
   402→              </div>
   403→            </div>
   404→          </div>
   405→        </CardHeader>
   406→
   407→        <CardContent className="px-6 pb-8">
   408→          <Form {...form}>
   409→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-8">
   410→              {/* ── Basic Information ────────────────────────────────── */}
   411→              <section>
   412→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   413→                  Basic Information
   414→                </h2>
   415→
   416→                <div className="rounded-lg border bg-muted/30 p-4 space-y-4">
   417→                  <FormField
   418→                    control={form.control}
   419→                    name="title"
   420→                    render={({ field }) => (
   421→                      <FormItem>
   422→                        <FormLabel>Project Title</FormLabel>
   423→                        <FormControl>
   424→                          <Input
   425→                            placeholder="e.g. The Future of Renewable Energy"
   426→                            {...field}
   427→                          />
   428→                        </FormControl>
   429→                        <FormMessage />
   430→                      </FormItem>
   431→                    )}
   432→                  />
   433→
   434→                  <FormField
   435→                    control={form.control}
   436→                    name="topic"
   437→                    render={({ field }) => (
   438→                      <FormItem>
   439→                        <FormLabel>Video Topic</FormLabel>
   440→                        <FormControl>
   441→                          <Input
   442→                            placeholder="e.g. How solar panels are changing the world"
   443→                            {...field}
   444→                          />
   445→                        </FormControl>
   446→                        <FormMessage />
   447→                      </FormItem>
   448→                    )}
   449→                  />
   450→
   451→                  <FormField
   452→                    control={form.control}
   453→                    name="description"
   454→                    render={({ field }) => (
   455→                      <FormItem>
   456→                        <div className="flex items-center justify-between">
   457→                          <FormLabel>Description</FormLabel>
   458→                          <span className="text-xs text-muted-foreground">
   459→                            {descriptionValue.length}/1000
   460→                          </span>
   461→                        </div>
   462→                        <FormControl>
   463→                          <Textarea
   464→                            placeholder="Add any extra details, context, or specific requirements for the video..."
   465→                            rows={3}
   466→                            maxLength={1000}
   467→                            {...field}
   468→                            value={field.value ?? ''}
   469→                          />
   470→                        </FormControl>
   471→                        <FormMessage />
   472→                      </FormItem>
   473→                    )}
   474→                  />
   475→                </div>
   476→              </section>
   477→
   478→              <Separator />
   479→
   480→              {/* ── Production Settings ──────────────────────────────── */}
   481→              <section>
   482→                <h2 className="mb-4 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   483→                  Production Settings
   484→                </h2>
   485→
   486→                <div className="rounded-lg border bg-muted/30 p-4">
   487→                  <div className="grid gap-4 sm:grid-cols-2">
   488→                    <FormField
   489→                      control={form.control}
   490→                      name="duration"
   491→                      render={({ field }) => (
   492→                        <FormItem>
   493→                          <FormLabel>Video Duration</FormLabel>
   494→                          <Select
   495→                            onValueChange={field.onChange}
   496→                            defaultValue={field.value}
   497→                          >
   498→                            <FormControl>
   499→                              <SelectTrigger>
   500→                                <SelectValue placeholder="Select duration" />
   501→                              </SelectTrigger>
   502→                            </FormControl>
   503→                            <SelectContent>
   504→                              {(Object.entries(DURATION_LABELS) as [VideoDuration, string][]).map(
   505→                                ([value, label]) => (
   506→                                  <SelectItem key={value} value={value}>
   507→                                    {label}
   508→                                  </SelectItem>
   509→                                )
   510→                              )}
   511→                            </SelectContent>
   512→                          </Select>
   513→                          <FormMessage />
   514→                        </FormItem>
   515→                      )}
   516→                    />
   517→
   518→                    <FormField
   519→                      control={form.control}
   520→                      name="theme"
   521→                      render={({ field }) => (
   522→                        <FormItem>
   523→                          <FormLabel>Visual Theme</FormLabel>
   524→                          <Select
   525→                            onValueChange={field.onChange}
   526→                            defaultValue={field.value}
   527→                          >
   528→                            <FormControl>
   529→                              <SelectTrigger>
   530→                                <SelectValue placeholder="Select theme" />
   531→                              </SelectTrigger>
   532→                            </FormControl>
   533→                            <SelectContent>
   534→                              {(Object.entries(THEME_LABELS) as [VideoTheme, string][]).map(
   535→                                ([value, label]) => (
   536→                                  <SelectItem key={value} value={value}>
   537→                                    {label}
   538→                                  </SelectItem>
   539→                                )
   540→                              )}
   541→                            </SelectContent>
   542→                          </Select>
   543→                          <FormMessage />
   544→                        </FormItem>
   545→                      )}
   546→                    />
   547→
   548→                    <FormField
   549→                      control={form.control}
   550→                      name="language"
   551→                      render={({ field }) => (
   552→                        <FormItem>
   553→                          <FormLabel>Language</FormLabel>
   554→                          <Select
   555→                            onValueChange={field.onChange}
   556→                            defaultValue={field.value}
   557→                          >
   558→                            <FormControl>
   559→                              <SelectTrigger>
   560→                                <SelectValue placeholder="Select language" />
   561→                              </SelectTrigger>
   562→                            </FormControl>
   563→                            <SelectContent>
   564→                              {(Object.entries(LANGUAGE_LABELS) as [VideoLanguage, string][]).map(
   565→                                ([value, label]) => (
   566→                                  <SelectItem key={value} value={value}>
   567→                                    {label}
   568→                                  </SelectItem>
   569→                                )
   570→                              )}
   571→                            </SelectContent>
   572→                          </Select>
   573→                          <FormMessage />
   574→                        </FormItem>
   575→                      )}
   576→                    />
   577→
   578→                    <FormField
   579→                      control={form.control}
   580→                      name="writingStyle"
   581→                      render={({ field }) => (
   582→                        <FormItem>
   583→                          <FormLabel>Writing Style</FormLabel>
   584→                          <Select
   585→                            onValueChange={field.onChange}
   586→                            defaultValue={field.value}
   587→                          >
   588→                            <FormControl>
   589→                              <SelectTrigger>
   590→                                <SelectValue placeholder="Select style" />
   591→                              </SelectTrigger>
   592→                            </FormControl>
   593→                            <SelectContent>
   594→                              {(Object.entries(STYLE_LABELS) as [WritingStyle, string][]).map(
