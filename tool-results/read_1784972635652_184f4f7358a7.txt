    48→    48→function formatDate(timestamp: string | number): string {
    49→    49→  if (!timestamp) return '—';
    50→    50→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    51→    51→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    52→    52→}
    53→    53→
    54→    54→function formatExpiryDate(timestamp: number): string {
    55→    55→  if (!timestamp || timestamp <= 0) return '—';

    58→    58→}
