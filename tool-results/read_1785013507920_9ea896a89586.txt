    59→    59→// ── Strip stage directions client-side (for preview) ──

    61→    61→function stripBrackets(text: string): string {
    62→    62→  return text
    63→    63→    .replace(/\[.*?\]/g, '')
    64→    64→    .replace(/\(.*?\)/g, '')
    65→    65→    .replace(/\*.*?\*/g, '')
    66→    66→    .replace(/[\[\]\(\)]/g, '')
    67→    67→    .replace(/\s{2,}/g, ' ')
    68→    68→    .trim();
    69→    69→}

    75→    75→  sceneId,

    77→    77→  label = 'Narration',
    78→    78→}: VoiceGeneratorProps) {
    79→    79→  // Voice data from API
    80→    80→  const [voices, setVoices] = useState<Voice[]>([]);

    83→    83→  const [paces, setPaces] = useState<{ value: string; label: string }[]>([]);
    84→    84→  const [accents, setAccents] = useState<{ value: string; label: string }[]>([]);
    85→    85→  const [voicesLoading, setVoicesLoading] = useState(false);

    89→    89→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
    90→    90→
    91→    91→  // Style / Pace / Accent
    92→    92→  const [selectedStyle, setSelectedStyle] = useState('');
    93→    93→  const [selectedPace, setSelectedPace] = useState('');
    94→    94→  const [selectedAccent, setSelectedAccent] = useState('');
    95→    95→  const [customInstructions, setCustomInstructions] = useState('');
    96→    96→  const [settingsOpen, setSettingsOpen] = useState(false);
    97→    97→
    98→    98→  // Playback
    99→    99→  const [generating, setGenerating] = useState(false);

   101→   101→  const [audioUrl, setAudioUrl] = useState<string | null>(null);
   102→   102→  const [audioProgress, setAudioProgress] = useState(0);
   103→   103→  const [audioDuration, setAudioDuration] = useState(0);

   114→   114→        if (parsed.style) setSelectedStyle(parsed.style);

   116→   116→        if (parsed.accent) setSelectedAccent(parsed.accent);
   117→   117→        if (parsed.customInstructions) setCustomInstructions(parsed.customInstructions);
   118→   118→      }
   119→   119→    } catch {}
   120→   120→  }, []);
   121→   121→
   122→   122→  // Load saved audio on mount
   123→   123→  useEffect(() => {

   126→   126→      setAudioUrl(narrationAudioPath);

   128→   128→  }, [narrationAudioPath]);
   129→   129→
   130→   130→  // Cleanup audio on unmount

   132→   132→    return () => {
   133→   133→      if (audioRef.current) {

   135→   135→      }
   136→   136→    };
   137→   137→  }, []);
   138→   138→
   139→   139→  const savePreference = useCallback((settings: Record<string, string>) => {

   141→   141→      sessionStorage.setItem('scriptforge_voice_settings', JSON.stringify(settings));
   142→   142→    } catch {}
   143→   143→  }, []);
   144→   144→
   145→   145→  // Fetch voices on mount

   156→   156→          setStyles(data.styles || []);

   160→   160→            setSelectedVoiceName(voiceList[0].name);
   161→   161→          }
   162→   162→        } else {
   163→   163→          const data = await res.json().catch(() => ({}));
   164→   164→          toast.error((data as Record<string, string>).error || 'Failed to load voices');
   165→   165→        }
   166→   166→      } catch {
   167→   167→        toast.error('Failed to connect to voice service');
   168→   168→      } finally {
   169→   169→        setVoicesLoading(false);
   170→   170→      }
   171→   171→    }
   172→   172→    fetchVoices();
   173→   173→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   174→   174→
   175→   175→  // Build instructions from all dropdown selections + custom text
   176→   176→  const builtInstructions = useMemo(() => {
   177→   177→    const parts: string[] = [];
   178→   178→    if (selectedStyle) parts.push(selectedStyle);
   179→   179→    if (selectedPace === 'very_slow') parts.push('speak very slowly');
   180→   180→    else if (selectedPace === 'slow') parts.push('speak slowly');
   181→   181→    else if (selectedPace === 'moderate') parts.push('speak at a moderate pace');
   182→   182→    else if (selectedPace === 'fast') parts.push('speak quickly');
   183→   183→    else if (selectedPace === 'very_fast') parts.push('speak very quickly');
   184→   184→    if (selectedAccent === 'american') parts.push('with an American accent');
   185→   185→    else if (selectedAccent === 'british') parts.push('with a British accent');
   186→   186→    else if (selectedAccent === 'australian') parts.push('with an Australian accent');

   188→   188→    if (customInstructions.trim()) parts.push(customInstructions.trim());

   195→   195→    return voices.filter(v => v.category === categoryFilter);
   196→   196→  }, [voices, categoryFilter]);
   197→   197→

   201→   201→    return descriptions.map(desc => ({
   202→   202→      description: desc,
   203→   203→      category: filteredVoices.find(v => v.description === desc)?.category || '',
   204→   204→      voices: filteredVoices.filter(v => v.description === desc),
   205→   205→    }));
   206→   206→  }, [filteredVoices]);
   207→   207→
   208→   208→  // Active filters count
   209→   209→  const activeFilterCount = useMemo(() => {
   210→   210→    let count = 0;
   211→   211→    if (selectedStyle) count++;
   212→   212→    if (selectedPace) count++;
   213→   213→    if (selectedAccent) count++;
   214→   214→    return count;
   215→   215→  }, [selectedStyle, selectedPace, selectedAccent]);
   216→   216→

   219→   219→    if (audioRef.current) {
   220→   220→      audioRef.current.pause();
   221→   221→    }
   222→   222→
   223→   223→    const audio = new Audio(url);
   224→   224→    audioRef.current = audio;
   225→   225→
   226→   226→    audio.addEventListener('loadedmetadata', () => {
   227→   227→      setAudioDuration(audio.duration);
   228→   228→    });

   230→   230→      setAudioProgress(audio.currentTime / (audio.duration || 1));
   231→   231→    });
   232→   232→    audio.addEventListener('ended', () => {
   233→   233→      setPlaying(false);
   234→   234→      setAudioProgress(0);
   235→   235→    });
   236→   236→    audio.addEventListener('error', () => {
   237→   237→      toast.error('Audio playback error');
   238→   238→      setPlaying(false);
   239→   239→    });
   240→   240→
   241→   241→    if (autoPlay) {
   242→   242→      audio.play().then(() => setPlaying(true)).catch(() => {});
   243→   243→    }
   244→   244→  }, []);
   245→   245→
   246→   246→  // Generate speech
   247→   247→  const handleGenerate = useCallback(async () => {
   248→   248→    if (!selectedVoiceName) {
   249→   249→      toast.error('Please select a voice first');
   250→   250→      return;
   251→   251→    }
   252→   252→    if (!text.trim()) {
   253→   253→      toast.error('No text to generate speech from');
   254→   254→      return;
   255→   255→    }
   256→   256→
   257→   257→    const cleanText = stripBrackets(text);
   258→   258→    if (!cleanText) {
   259→   259→      toast.error('No spoken text after removing stage directions');
   260→   260→      return;
   261→   261→    }
   262→   262→
   263→   263→    // Stop any current playback
   264→   264→    if (audioRef.current) {
   265→   265→      audioRef.current.pause();
   266→   266→      audioRef.current = null;
   267→   267→    }
   268→   268→    setPlaying(false);
   269→   269→    setAudioProgress(0);
   270→   270→    setAudioDuration(0);
   271→   271→
   272→   272→    setGenerating(true);
   273→   273→    try {
   274→   274→      const res = await fetch('/api/tts/generate', {
   275→   275→        method: 'POST',
   276→   276→        headers: { 'Content-Type': 'application/json' },
   277→   277→        body: JSON.stringify({
   278→   278→          voiceName: selectedVoiceName,
   279→   279→          text: cleanText,
   280→   280→          instructions: builtInstructions || undefined,
   281→   281→          saveAudio: sceneId !== 'preview',
   282→   282→          sceneId: sceneId !== 'preview' ? sceneId : undefined,
   283→   283→        }),
   284→   284→      });
   285→   285→
   286→   286→      if (!res.ok) {
   287→   287→        const data = await res.json().catch(() => ({}));
   288→   288→        throw new Error((data as Record<string, string>).error || 'Generation failed');
   289→   289→      }

   291→   291→      const serverAudioPath = res.headers.get('X-Audio-Path');
   292→   292→
   293→   293→      let url: string;
   294→   294→      if (serverAudioPath) {
   295→   295→        url = serverAudioPath;
   296→   296→        setHasSavedAudio(true);
   297→   297→        toast.success('Voice generated and saved!');
   298→   298→      } else {
   299→   299→        const blob = await res.blob();
   300→   300→        url = URL.createObjectURL(blob);
   301→   301→        toast.success('Voice generated!');
   302→   302→      }
   303→   303→
   304→   304→      setAudioUrl(url);
   305→   305→      setupAudioPlayer(url, true);
   306→   306→    } catch (err) {
   307→   307→      toast.error((err as Error).message || 'Failed to generate voice');
   308→   308→    } finally {
   309→   309→      setGenerating(false);
   310→   310→    }
   311→   311→  }, [selectedVoiceName, text, builtInstructions, sceneId, setupAudioPlayer]);
   312→   312→
   313→   313→  // Playback controls
   314→   314→  const togglePlayback = useCallback(() => {
   315→   315→    if (!audioUrl) return;
   316→   316→    if (audioRef.current) {
   317→   317→      if (playing) {
   318→   318→        audioRef.current.pause();
   319→   319→        setPlaying(false);
   320→   320→      } else {
   321→   321→        audioRef.current.play();
   322→   322→        setPlaying(true);
   323→   323→      }
   324→   324→    } else {
   325→   325→      setupAudioPlayer(audioUrl, true);
   326→   326→    }
   327→   327→  }, [playing, audioUrl, setupAudioPlayer]);
   328→   328→
   329→   329→  const stopPlayback = useCallback(() => {
   330→   330→    if (!audioRef.current) return;
   331→   331→    audioRef.current.pause();
   332→   332→    audioRef.current.currentTime = 0;
   333→   333→    setPlaying(false);
   334→   334→    setAudioProgress(0);
   335→   335→  }, []);
   336→   336→
   337→   337→  const handleDownload = useCallback(() => {
   338→   338→    if (!audioUrl) return;
   339→   339→    const a = document.createElement('a');
   340→   340→    a.href = audioUrl;
   341→   341→    a.download = `narration-${sceneId}.wav`;
   342→   342→    document.body.appendChild(a);
   343→   343→    a.click();
   344→   344→    document.body.removeChild(a);
   345→   345→    toast.success('Audio downloaded');
   346→   346→  }, [audioUrl, sceneId]);
   347→   347→
   348→   348→  // Handlers that save preferences
   349→   349→  const handleVoiceChange = useCallback((voiceName: string) => {
   350→   350→    setSelectedVoiceName(voiceName);
   351→   351→    savePreference({
   352→   352→      voiceName,
   353→   353→      style: selectedStyle,
   354→   354→      pace: selectedPace,
   355→   355→      accent: selectedAccent,
   356→   356→      customInstructions,
   357→   357→    });
   358→   358→  }, [selectedStyle, selectedPace, selectedAccent, customInstructions, savePreference]);
   359→   359→
   360→   360→  const handleStyleChange = useCallback((value: string) => {
   361→   361→    setSelectedStyle(value);
   362→   362→    savePreference({
   363→   363→      voiceName: selectedVoiceName,
   364→   364→      style: value,
   365→   365→      pace: selectedPace,
   366→   366→      accent: selectedAccent,
   367→   367→      customInstructions,
   368→   368→    });
   369→   369→  }, [selectedVoiceName, selectedPace, selectedAccent, customInstructions, savePreference]);
   370→   370→
   371→   371→  const handlePaceChange = useCallback((value: string) => {
   372→   372→    setSelectedPace(value);
   373→   373→    savePreference({
   374→   374→      voiceName: selectedVoiceName,
   375→   375→      style: selectedStyle,
   376→   376→      pace: value,
   377→   377→      accent: selectedAccent,
   378→   378→      customInstructions,
   379→   379→    });
   380→   380→  }, [selectedVoiceName, selectedStyle, selectedAccent, customInstructions, savePreference]);
   381→   381→
   382→   382→  const handleAccentChange = useCallback((value: string) => {
   383→   383→    setSelectedAccent(value);
   384→   384→    savePreference({
   385→   385→      voiceName: selectedVoiceName,
   386→   386→      style: selectedStyle,
   387→   387→      pace: selectedPace,
   388→   388→      accent: value,
   389→   389→      customInstructions,
   390→   390→    });
   391→   391→  }, [selectedVoiceName, selectedStyle, selectedPace, customInstructions, savePreference]);
   392→   392→
   393→   393→  const handleCustomInstructionsChange = useCallback((value: string) => {
   394→   394→    setCustomInstructions(value);
   395→   395→    savePreference({
   396→   396→      voiceName: selectedVoiceName,
   397→   397→      style: selectedStyle,
   398→   398→      pace: selectedPace,
   399→   399→      accent: selectedAccent,
   400→   400→      customInstructions: value,
   401→   401→    });
   402→   402→  }, [selectedVoiceName, selectedStyle, selectedPace, selectedAccent, savePreference]);
   403→   403→
   404→   404→  // Format time
   405→   405→  const formatTime = (seconds: number) => {
   406→   406→    const m = Math.floor(seconds / 60);
   407→   407→    const s = Math.floor(seconds % 60);
   408→   408→    return `${m}:${s.toString().padStart(2, '0')}`;
   409→   409→  };
   410→   410→
   411→   411→  const hasAudio = !!audioUrl;
   412→   412→  const noText = !text.trim();
   413→   413→
   414→   414→  return (
   415→   415→    <div className="space-y-3">
   416→   416→      {/* Voice Selection + Controls Row */}
   417→   417→      <div className="flex items-center gap-2 flex-wrap">
   418→   418→        {/* Category Filter */}
   419→   419→        <Popover>
   420→   420→          <PopoverTrigger asChild>
   421→   421→            <Button
   422→   422→              variant="outline"
   423→   423→              size="sm"
   424→   424→              className="h-8 text-xs gap-1.5 shrink-0"
   425→   425→              disabled={voicesLoading}
   426→   426→            >
   427→   427→              <Filter className="size-3" />
   428→   428→              {categoryFilter === 'all' ? 'All Voices' : categoryFilter}
   429→   429→            </Button>
   430→   430→          </PopoverTrigger>
   431→   431→          <PopoverContent className="w-48 p-2" align="start">
   432→   432→            <div className="space-y-1">
   433→   433→              <button
   434→   434→                onClick={() => setCategoryFilter('all')}
   435→   435→                className={`w-full text-left px-2 py-1.5 rounded text-xs transition-colors ${
   436→   436→                  categoryFilter === 'all' ? 'bg-primary text-primary-foreground' : 'hover:bg-muted'
   437→   437→                }`}
   438→   438→              >
   439→   439→                All Voices ({voices.length})
   440→   440→              </button>
   441→   441→              {categories.map(cat => {
   442→   442→                const count = voices.filter(v => v.category === cat).length;
   443→   443→                return (
   444→   444→                  <button
   445→   445→                    key={cat}
   446→   446→                    onClick={() => setCategoryFilter(cat)}
   447→   447→                    className={`w-full text-left px-2 py-1.5 rounded text-xs transition-colors ${
   448→   448→                      categoryFilter === cat ? 'bg-primary text-primary-foreground' : 'hover:bg-muted'
   449→   449→                    }`}
   450→   450→                  >
   451→   451→                    {cat} ({count})
   452→   452→                  </button>
   453→   453→                );
   454→   454→              })}
   455→   455→            </div>
   456→   456→          </PopoverContent>
   457→   457→        </Popover>
   458→   458→
   459→   459→        {/* Voice Dropdown */}
   460→   460→        <div className="flex-1 min-w-[200px]">
   461→   461→          <Select value={selectedVoiceName} onValueChange={handleVoiceChange} disabled={voicesLoading}>
   462→   462→            <SelectTrigger className="h-8 text-xs">
   463→   463→              <Mic className="size-3 mr-1.5 shrink-0 text-muted-foreground" />
   464→   464→              <SelectValue placeholder={voicesLoading ? 'Loading voices...' : 'Select voice'} />
   465→   465→            </SelectTrigger>
   466→   466→            <SelectContent>
   467→   467→              {groupedVoices.map(group => (
   468→   468→                <React.Fragment key={group.description}>
   469→   469→                  <div className="px-2 py-1.5 text-[10px] font-semibold text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
   470→   470→                    {group.description}
   471→   471→                    <Badge variant="secondary" className="text-[9px] px-1 py-0">
   472→   472→                      {group.category}
   473→   473→                    </Badge>
   474→   474→                  </div>
   475→   475→                  {group.voices.map(voice => (
   476→   476→                    <SelectItem key={voice.name} value={voice.name} className="text-xs">
   477→   477→                      <span className="font-medium">{voice.name}</span>
   478→   478→                    </SelectItem>
   479→   479→                  ))}
   480→   480→                </React.Fragment>
   481→   481→              ))}
   482→   482→              {filteredVoices.length === 0 && (
   483→   483→                <div className="px-2 py-3 text-xs text-muted-foreground text-center">
   484→   484→                  No voices in this category
   485→   485→                </div>
   486→   486→              )}
   487→   487→            </SelectContent>
   488→   488→          </Select>
   489→   489→        </div>
   490→   490→
   491→   491→        {/* Generate Button */}
   492→   492→        <Button
   493→   493→          size="sm"
   494→   494→          onClick={handleGenerate}
   495→   495→          disabled={generating || noText || !selectedVoiceName}
