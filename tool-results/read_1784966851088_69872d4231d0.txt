    47→function formatDate(timestamp: string | number): string {
    48→  if (!timestamp) return '—';
    49→  const date = new Date(typeof timestamp === 'number' ? timestamp : timestamp);
    50→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    51→}

    53→function formatExpiryDate(timestamp: number): string {
    54→  if (!timestamp || timestamp <= 0) return '—';
    55→  const date = new Date(timestamp);
    56→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    57→}
