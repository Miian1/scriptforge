    60→function formatDuration(seconds: number): string {
    61→  const m = Math.floor(seconds / 60);
    62→  const s = seconds % 60;
    63→  return `${m}:${s.toString().padStart(2, '0')}`;
    64→}

    66→function totalDuration(scenes: Scene[]): number {
    67→  return scenes.reduce((sum, s) => sum + s.estimatedDuration, 0);

   339→            }`}
   340→          />
   341→        ))}
   342→      </div>
   343→    </div>

   347→// ---- Main ScriptEditor ----
   348→export default function ScriptEditor() {
   349→  const projects = useAppStore((s) => s.projects);
   350→  const activeProjectId = useAppStore((s) => s.activeProjectId);
   351→  const scenes = useAppStore((s) => s.scenes);
   352→  const addScene = useAppStore((s) => s.addScene);
   353→  const addScenes = useAppStore((s) => s.addScenes);
   354→  const loadScenes = useAppStore((s) => s.loadScenes);
   355→  const loadCharacters = useAppStore((s) => s.loadCharacters);
   356→  const reorderScenes = useAppStore((s) => s.reorderScenes);
   357→  const updateProject = useAppStore((s) => s.updateProject);
   358→  const setActiveProjectId = useAppStore((s) => s.setActiveProjectId);
   359→  const generatingProjectId = useAppStore((s) => s.generatingProjectId);
   360→  const setGeneratingProjectId = useAppStore((s) => s.setGeneratingProjectId);
   361→  const router = useRouter();
   362→
   363→  const listEndRef = useRef<HTMLDivElement>(null);
   364→
   365→  // Local generating state for phases
   366→  const [isGeneratingPhase, setIsGeneratingPhase] = useState(false);
   367→  const [generatingPhaseLabel, setGeneratingPhaseLabel] = useState('');
   368→
   369→  const project = projects.find((p) => p.id === activeProjectId) ?? null;
   370→
   371→  // Computed phase info from project settings
   372→  const totalScenes = project?.settings?.totalScenes || 60;
   373→  const scenesPerPhase = project?.settings?.scenesPerPhase || SCENES_PER_PHASE;
   374→  const totalPhases = Math.ceil(totalScenes / scenesPerPhase);
   375→  const currentPhaseNumber = Math.min(
   376→    Math.ceil(scenes.length / scenesPerPhase) + (scenes.length % scenesPerPhase === 0 ? 0 : 1),
   377→    totalPhases
   378→  );
   379→  // Actually: how many full phases are done?
   380→  const completedPhases = Math.floor(scenes.length / scenesPerPhase);
   381→  const isOnPartialPhase = scenes.length % scenesPerPhase !== 0;
   382→  const activePhaseNum = isOnPartialPhase ? completedPhases + 1 : completedPhases;
   383→  const allPhasesDone = scenes.length >= totalScenes;
   384→  const hasNextPhase = !allPhasesDone && scenes.length > 0;
   385→  const nextPhaseNumber = hasNextPhase
   386→    ? (scenes.length % scenesPerPhase === 0
   387→        ? scenes.length / scenesPerPhase + 1
   388→        : Math.floor(scenes.length / scenesPerPhase) + 1)
   389→    : 0;
   390→
   391→  // Group scenes into phases for display
   392→  const phases = useMemo(() => {
   393→    const grouped: { phase: number; scenes: Scene[] }[] = [];
   394→    const sorted = [...scenes].sort((a, b) => a.sceneNumber - b.sceneNumber);
   395→    for (let i = 0; i < sorted.length; i++) {
   396→      const phaseNum = Math.floor(sorted[i].sceneNumber / scenesPerPhase) + 1;
   397→      const lastGroup = grouped[grouped.length - 1];
   398→      if (lastGroup && lastGroup.phase === phaseNum) {
   399→        lastGroup.scenes.push(sorted[i]);
   400→      } else {
   401→        grouped.push({ phase: phaseNum, scenes: [sorted[i]] });
   402→      }
   403→    }
   404→    return grouped;
   405→  }, [scenes, scenesPerPhase]);
   406→
   407→  // Load scenes and characters on mount
   408→  useEffect(() => {
   409→    if (activeProjectId) {
   410→      loadScenes(activeProjectId);
   411→      loadCharacters(activeProjectId);
   412→    }
   413→  }, [activeProjectId, loadScenes, loadCharacters]);
   414→
   415→  // DnD sensors
   416→  const sensors = useSensors(
   417→    useSensor(PointerSensor, {
   418→      activationConstraint: { distance: 8 },
   419→    }),
   420→    useSensor(KeyboardSensor)
   421→  );
   422→
   423→  // Handle drag end
   424→  const handleDragEnd = useCallback(
   425→    (event: DragEndEvent) => {
   426→      const { active, over } = event;
   427→      if (!over || active.id === over.id) return;
   428→
   429→      const oldIndex = scenes.findIndex((s) => s.id === active.id);
   430→      const newIndex = scenes.findIndex((s) => s.id === over.id);
   431→      if (oldIndex === -1 || newIndex === -1) return;
   432→
   433→      const reordered = arrayMove(scenes, oldIndex, newIndex);
   434→      reorderScenes(reordered);
   435→    },
   436→    [scenes, reorderScenes]
   437→  );
   438→
   439→  // Add new scene
   440→  const handleAddScene = useCallback(async () => {
   441→    if (!project) return;
   442→    const now = Date.now();
   443→    const sceneLen = project.settings.sceneLength || 8;
   444→    const newScene: Scene = {
   445→      id: uuidv4(),
   446→      projectId: project.id,
   447→      sceneNumber: scenes.length + 1,
   448→      title: 'New Scene',
   449→      estimatedDuration: sceneLen,
   450→      goal: '',
   451→      narration: '',
   452→      imagePrompt: '',
   453→      animationPrompt: '',
   454→      characterIds: [],
   455→      notes: {
   456→        emotion: '',
   457→        visualFocus: '',
   458→        transitionSuggestion: '',
   459→        importantDetails: '',
   460→      },
   461→      createdAt: now,
   462→      updatedAt: now,
   463→    };
   464→    await addScene(newScene);
   465→
   466→    requestAnimationFrame(() => {
   467→      listEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   468→    });
   469→
   470→    toast.success('Scene added');
   471→  }, [project, scenes.length, addScene]);
   472→
   473→  // Generate next phase
   474→  const handleGenerateNextPhase = useCallback(async () => {
   475→    if (!project || isGeneratingPhase) return;
   476→
   477→    const phaseNum = nextPhaseNumber;
   478→    const sceneStart = scenes.length + 1;
   479→    const scenesInPhase = Math.min(scenesPerPhase, totalScenes - scenes.length);
   480→    const sceneEnd = sceneStart + scenesInPhase - 1;
   481→
   482→    // Get titles of existing scenes for continuity
   483→    const existingTitles = scenes
   484→      .sort((a, b) => a.sceneNumber - b.sceneNumber)
   485→      .map((s) => s.title);
   486→
   487→    setIsGeneratingPhase(true);
   488→    setGeneratingPhaseLabel(`Phase ${phaseNum} (Scenes ${sceneStart}-${sceneEnd})`);
   489→    setGeneratingProjectId(project.id);
   490→    await updateProject(project.id, { status: 'generating' });
   491→
   492→    try {
   493→      const channelNiche = useAuthStore.getState().user?.channelNiche as ChannelNiche | undefined;
   494→
   495→      const result = await generatePhase(project, {
   496→        phaseNumber: phaseNum,
   497→        totalPhases,
   498→        sceneStart,
   499→        sceneEnd,
   500→        previousPhaseTitles: existingTitles,
   501→      }, channelNiche || null);
   502→
   503→      const scenesWithId = result.scenes.map((s) => ({ ...s, projectId: project.id }));
   504→
   505→      // Save metadata if first phase somehow missed it
   506→      if (result.metadata) {
   507→        await updateProject(project.id, {
   508→          description: result.metadata.videoDescription || project.description || '',
   509→          thumbnailPrompt: result.metadata.thumbnailPrompt || '',
   510→          tags: result.metadata.tags || [],
   511→        });
   512→      }
   513→
   514→      await addScenes(scenesWithId);
   515→      await updateProject(project.id, { status: 'completed' });
   516→      await loadScenes(project.id);
   517→      toast.success(`Phase ${phaseNum} generated! ${scenesInPhase} scenes added.`);
   518→
   519→      // Scroll to the new scenes
   520→      requestAnimationFrame(() => {
   521→        listEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   522→      });
   523→    } catch (err) {
   524→      await updateProject(project.id, { status: 'error' });
   525→      toast.error((err as Error).message || `Failed to generate Phase ${phaseNum}`);
   526→    } finally {
   527→      setIsGeneratingPhase(false);
   528→      setGeneratingPhaseLabel('');
   529→      setGeneratingProjectId(null);
   530→    }
   531→  }, [project, scenes, nextPhaseNumber, scenesPerPhase, totalScenes, totalPhases, isGeneratingPhase, addScenes, updateProject, loadScenes, setGeneratingProjectId]);
   532→
   533→  // Export all as JSON
   534→  const handleExport = useCallback(() => {
   535→    if (!project || scenes.length === 0) return;
   536→    const exportData = {
   537→      project: {
   538→        title: project.title,
   539→        topic: project.topic,
   540→        description: project.description,
   541→        settings: project.settings,
   542→        exportedAt: new Date().toISOString(),
   543→      },
   544→      scenes: scenes
   545→        .sort((a, b) => a.sceneNumber - b.sceneNumber)
   546→        .map((s) => ({
   547→          sceneNumber: s.sceneNumber,
   548→          title: s.title,
   549→          estimatedDuration: s.estimatedDuration,
   550→          goal: s.goal,
   551→          narration: s.narration,
   552→          imagePrompt: s.imagePrompt,
   553→          animationPrompt: s.animationPrompt,
   554→          notes: s.notes,
   555→        })),
   556→    };
   557→
   558→    const blob = new Blob([JSON.stringify(exportData, null, 2)], {
   559→      type: 'application/json',
   560→    });
   561→    const url = URL.createObjectURL(blob);
   562→    const a = document.createElement('a');
   563→    a.href = url;
   564→    a.download = `${project.title.replace(/[^a-zA-Z0-9]/g, '_')}_script.json`;
   565→    document.body.appendChild(a);
   566→    a.click();
   567→    document.body.removeChild(a);
   568→    URL.revokeObjectURL(url);
   569→    toast.success('Script exported');
   570→  }, [project, scenes]);
   571→
   572→  // Retry generation (regenerate Phase 1)
   573→  const handleRetry = useCallback(async () => {
   574→    if (!project) return;
   575→
   576→    setGeneratingProjectId(project.id);
   577→    setIsGeneratingPhase(true);
   578→    setGeneratingPhaseLabel('Phase 1');
   579→    await updateProject(project.id, { status: 'generating' });
   580→
   581→    try {
   582→      const channelNiche = useAuthStore.getState().user?.channelNiche as ChannelNiche | undefined;
   583→
   584→      const { scenes: newScenes, metadata } = await generateScript(project, channelNiche || null);
   585→      for (const scene of newScenes) {
   586→        await addScene(scene);
   587→      }
   588→      if (metadata) {
   589→        await updateProject(project.id, {
   590→          description: metadata.videoDescription || project.description || '',
   591→          thumbnailPrompt: metadata.thumbnailPrompt || '',
   592→          tags: metadata.tags || [],
   593→        });
   594→      }
   595→      await updateProject(project.id, { status: 'completed' });
   596→      toast.success('Script generated successfully!');
   597→      await loadScenes(project.id);
   598→    } catch (err) {
   599→      await updateProject(project.id, { status: 'error' });
   600→      toast.error((err as Error).message || 'Generation failed');
   601→    } finally {
   602→      setGeneratingProjectId(null);
   603→      setIsGeneratingPhase(false);
   604→      setGeneratingPhaseLabel('');
   605→    }
   606→  }, [project, addScene, updateProject, loadScenes, setGeneratingProjectId]);
   607→
   608→  // Back to projects
   609→  const handleBack = useCallback(() => {
   610→    setActiveProjectId(null);
   611→    router.push('/projects');
   612→  }, [setActiveProjectId, router]);
   613→
   614→  // ---- Render ----
   615→  if (!project) {
   616→    return (
   617→      <div className="flex items-center justify-center min-h-[60vh]">
   618→        <p className="text-muted-foreground">Project not found.</p>
   619→      </div>
   620→    );
   621→  }
   622→
   623→  const isGenerating = project.status === 'generating' || generatingProjectId === project.id;
   624→  const isError = project.status === 'error' && !isGenerating;
   625→  const total = totalDuration(scenes);
   626→
   627→  return (
   628→    <div className="flex flex-col min-h-full">
   629→      {/* ---- HEADER ---- */}
   630→      <header className="sticky top-0 z-30 bg-background/80 backdrop-blur-lg border-b">
   631→        <div className="max-w-4xl mx-auto px-4 sm:px-6 py-3 sm:py-4 flex items-center gap-3 sm:gap-4 flex-wrap">
   632→          <Tooltip>
   633→            <TooltipTrigger asChild>
   634→              <Button variant="ghost" size="icon" className="shrink-0" onClick={handleBack}>
   635→                <ArrowLeft className="size-4" />
   636→              </Button>
   637→            </TooltipTrigger>
   638→            <TooltipContent>Back to Projects</TooltipContent>
   639→          </Tooltip>
   640→
   641→          <div className="flex-1 min-w-0">
   642→            <h1 className="text-lg sm:text-xl font-bold truncate">{project.title}</h1>
   643→            <p className="text-xs text-muted-foreground truncate hidden sm:block">{project.topic}</p>
   644→          </div>
   645→
   646→          <Badge variant={getStatusVariant(project.status)} className="shrink-0">
   647→            {isGenerating && <Loader2 className="size-3 mr-1 animate-spin" />}
   648→            {STATUS_LABELS[project.status]}
   649→          </Badge>
   650→
   651→          {scenes.length > 0 && !isGenerating && (
   652→            <div className="flex items-center gap-3 text-xs text-muted-foreground shrink-0">
   653→              <span className="flex items-center gap-1">
   654→                <FileVideo className="size-3.5" />
   655→                {scenes.length}/{totalScenes}
   656→              </span>
   657→              <span className="flex items-center gap-1">
   658→                <Clock className="size-3.5" />
   659→                {formatDuration(total)}
   660→              </span>
   661→            </div>
   662→          )}
   663→
   664→          {!isGenerating && (
   665→            <div className="flex items-center gap-2 shrink-0">
   666→              {allPhasesDone && (
   667→                <Tooltip>
   668→                  <TooltipTrigger asChild>
   669→                    <Button variant="outline" size="sm" onClick={handleAddScene} className="gap-1.5">
   670→                      <Plus className="size-3.5" />
   671→                      <span className="hidden sm:inline">Add Scene</span>
   672→                    </Button>
   673→                  </TooltipTrigger>
   674→                  <TooltipContent>Add a new scene</TooltipContent>
   675→                </Tooltip>
   676→              )}
   677→              <Tooltip>
   678→                <TooltipTrigger asChild>
   679→                  <Button
   680→                    variant="outline"
   681→                    size="sm"
   682→                    onClick={handleExport}
   683→                    disabled={scenes.length === 0}
   684→                    className="gap-1.5"
   685→                  >
   686→                    <Download className="size-3.5" />
   687→                    <span className="hidden sm:inline">Export</span>
   688→                  </Button>
   689→                </TooltipTrigger>
   690→                <TooltipContent>Export script as JSON</TooltipContent>
   691→              </Tooltip>
   692→            </div>
   693→          )}
   694→        </div>
   695→      </header>
   696→
   697→      {/* ---- CONTENT ---- */}
   698→      <main className="flex-1 max-w-4xl w-full mx-auto px-4 sm:px-6 py-6">
   699→        {/* Generating state */}
   700→        {(isGenerating || isGeneratingPhase) && (
   701→          <GeneratingState project={project} phaseLabel={generatingPhaseLabel || 'Phase 1'} />
   702→        )}
   703→
   704→        {/* Error state */}
   705→        {isError && <ErrorState project={project} onRetry={handleRetry} />}
   706→
   707→        {/* Scene list */}
   708→        {!isGenerating && !isError && !isGeneratingPhase && (
   709→          <>
   710→            {scenes.length === 0 ? (
   711→              <EmptyState />
   712→            ) : (
   713→              <>
   714→                {/* Phase progress overview */}
   715→                <div className="mb-6 rounded-xl border bg-muted/30 p-4">
   716→                  <div className="flex items-center justify-between mb-3">
   717→                    <div className="flex items-center gap-2 text-sm font-semibold">
   718→                      <Layers className="size-4 text-primary" />
   719→                      Generation Progress
   720→                    </div>
   721→                    <Badge variant={allPhasesDone ? 'default' : 'secondary'} className="text-xs">
   722→                      {allPhasesDone ? 'All phases complete' : `${completedPhases}/${totalPhases} phases`}
   723→                    </Badge>
   724→                  </div>
   725→                  <PhaseProgressBar
   726→                    currentPhase={activePhaseNum}
   727→                    totalPhases={totalPhases}
   728→                    scenesGenerated={scenes.length}
   729→                    totalScenes={totalScenes}
   730→                  />
   731→                </div>
   732→
   733→                {/* Score, concept & metadata cards */}
   734→                <div className="space-y-4 mb-6">
   735→                  <ProjectScoreCard project={project} />
   736→                  <VideoConceptCard project={project} />
   737→                  <ProjectMetaCard project={project} />
   738→                  <CharacterManager project={project} />
   739→                </div>
   740→
   741→                {/* Scenes grouped by phase */}
   742→                <div className="space-y-6">
   743→                  {phases.map(({ phase, scenes: phaseScenes }) => (
   744→                    <div key={`phase-${phase}`}>
   745→                      {/* Phase header */}
   746→                      <div className="flex items-center gap-2 mb-3">
   747→                        <div className="flex items-center gap-2 rounded-lg bg-primary/10 px-3 py-1.5">
   748→                          <Layers className="size-3.5 text-primary" />
   749→                          <span className="text-sm font-semibold text-primary">
   750→                            Phase {phase}
   751→                          </span>
   752→                        </div>
   753→                        <span className="text-xs text-muted-foreground">
   754→                          Scenes {phaseScenes[0]?.sceneNumber}–{phaseScenes[phaseScenes.length - 1]?.sceneNumber}
   755→                          {' '}({phaseScenes.length} scenes)
   756→                        </span>
   757→                        <div className="flex-1" />
   758→                        <span className="text-xs text-muted-foreground">
   759→                          {formatDuration(phaseScenes.reduce((s, sc) => s + sc.estimatedDuration, 0))}
   760→                        </span>
   761→                      </div>
   762→
   763→                      <Separator className="mb-4" />
   764→
   765→                      {/* DnD for phase scenes */}
   766→                      <DndContext
   767→                        sensors={sensors}
   768→                        collisionDetection={closestCenter}
   769→                        onDragEnd={handleDragEnd}
   770→                      >
   771→                        <SortableContext
   772→                          items={phaseScenes.map((s) => s.id)}
   773→                          strategy={verticalListSortingStrategy}
   774→                        >
   775→                          <div className="space-y-4">
   776→                            <AnimatePresence mode="popLayout">
   777→                              {phaseScenes.map((scene) => (
   778→                                <SceneCard
   779→                                  key={scene.id}
   780→                                  scene={scene}
   781→                                  project={project}
   782→                                  totalScenes={totalScenes}
   783→                                />
   784→                              ))}
   785→                            </AnimatePresence>
   786→                          </div>
   787→                        </SortableContext>
   788→                      </DndContext>
   789→                    </div>
   790→                  ))}
   791→                </div>
   792→
   793→                {/* ---- Bottom actions ---- */}
   794→                <div className="mt-8">
   795→                  {hasNextPhase && !allPhasesDone && (
   796→                    <motion.div
   797→                      initial={{ opacity: 0, y: 12 }}
   798→                      animate={{ opacity: 1, y: 0 }}
   799→                      className="flex flex-col items-center gap-3"
   800→                    >
   801→                      <div className="rounded-xl border-2 border-dashed border-primary/30 bg-primary/[0.03] p-6 text-center w-full max-w-md space-y-4">
   802→                        <div className="flex items-center justify-center gap-2">
   803→                          <Zap className="size-5 text-primary" />
   804→                          <span className="text-sm font-semibold">
   805→                            Generate Next Phase
   806→                          </span>
   807→                        </div>
   808→                        <p className="text-xs text-muted-foreground">
   809→                          Phase {nextPhaseNumber} will generate{' '}
   810→                          {Math.min(scenesPerPhase, totalScenes - scenes.length)} more scenes
   811→                          (Scene {scenes.length + 1}–{Math.min(scenes.length + scenesPerPhase, totalScenes)}).
   812→                          AI will continue the story from where Phase {nextPhaseNumber - 1} ended.
   813→                        </p>
   814→                        <Button
   815→                          onClick={handleGenerateNextPhase}
   816→                          disabled={isGeneratingPhase}
   817→                          size="lg"
   818→                          className="gap-2"
   819→                        >
   820→                          {isGeneratingPhase ? (
   821→                            <>
   822→                              <Loader2 className="size-4 animate-spin" />
   823→                              Generating Phase {nextPhaseNumber}...
   824→                            </>
   825→                          ) : (
   826→                            <>
   827→                              <Sparkles className="size-4" />
   828→                              Generate Phase {nextPhaseNumber} of {totalPhases}
   829→                            </>
   830→                          )}
   831→                        </Button>
   832→                      </div>
   833→                    </motion.div>
   834→                  )}
   835→
   836→                  {allPhasesDone && (
   837→                    <motion.div
   838→                      initial={{ opacity: 0, y: 12 }}
   839→                      animate={{ opacity: 1, y: 0 }}
   840→                      className="flex flex-col items-center gap-3"
   841→                    >
   842→                      <div className="rounded-xl border border-emerald-500/30 bg-emerald-500/5 p-4 text-center max-w-md">
   843→                        <div className="flex items-center justify-center gap-2 text-sm font-semibold text-emerald-600 dark:text-emerald-400 mb-1">
   844→                          <Check className="size-4" />
   845→                          All {totalPhases} phases complete — {scenes.length} scenes generated
   846→                        </div>
   847→                        <p className="text-xs text-muted-foreground">
   848→                          Your full script is ready. You can add more scenes manually or export the script.
   849→                        </p>
   850→                      </div>
   851→                      <Button
   852→                        variant="outline"
   853→                        size="sm"
   854→                        onClick={handleAddScene}
   855→                        className="gap-1.5 text-muted-foreground"
   856→                      >
   857→                        <Plus className="size-3.5" />
   858→                        Add Scene
   859→                      </Button>
   860→                    </motion.div>
   861→                  )}
   862→                </div>
   863→
   864→                <div ref={listEndRef} />
   865→              </>
   866→            )}
   867→          </>
   868→        )}
   869→      </main>
   870→    </div>
   871→  );
